        parentAttributes.setNamedItem(xmlnsAttr);

        DOMNodePointer pointer = new DOMNodePointer(node, Locale.US);
        pointer.setNamespaceResolver(new org.apache.commons.jxpath.ri.NamespaceResolver()); // Set a dummy resolver

    }
