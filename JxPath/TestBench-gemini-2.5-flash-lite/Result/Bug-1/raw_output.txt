    private Map<String, Attr> attributes = new HashMap<>();
    private Map<String, String> attributeMap = new HashMap<>();
    private NodeList childNodes = new NodeList() {
        private java.util.List<Node> nodes = new java.util.ArrayList<>();
        @Override public Node item(int index) { return nodes.get(index); }
        @Override public int getLength() { return nodes.size(); }
        public void add(Node node) { nodes.add(node); }
        public void remove(Node node) { nodes.remove(node); }
    };
    private String attributeMapNS = new HashMap<>();

    public MockElement(String nodeName) {
        super(Node.ELEMENT_NODE, nodeName);
    }

    public void setAttribute(String name, String value) {
        attributeMap.put(name, value);
    }

    public void setAttributeNS(String namespaceURI, String qualifiedName, String value) {
        attributeMap.put(qualifiedName, value);
        this.setNamespaceURI(namespaceURI); // Assuming element's namespaceURI is set here
        this.setPrefix(qualifiedName.contains(":") ? qualifiedName.substring(0, qualifiedName.indexOf(':')) : null);
        this.setLocalName(qualifiedName.contains(":") ? qualifiedName.substring(qualifiedName.indexOf(':') + 1) : qualifiedName);
    }

    @Override
    public String getAttribute(String name) {
        return attributeMap.get(name);
    }

    @Override
    public String getAttributeNS(String namespaceURI, String localName) {
        String qualifiedName = localName;
        if (getPrefix() != null && getNamespaceURI().equals(namespaceURI)) {
            qualifiedName = getPrefix() + ":" + localName;
        }
        return attributeMap.get(qualifiedName);
    }

    @Override
    public NodeList getChildNodes() {
        return childNodes;
    }

    public NamedNodeMap getAttributes() {
