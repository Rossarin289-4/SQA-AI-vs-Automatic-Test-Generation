package org.jsoup.nodes;

import org.junit.Test;
import static org.junit.Assert.*;

public class FormElementTest {
    @Test
    public void testElementsInitiallyEmpty() throws Exception {
        Element parsed = Jsoup.parse("<form></form>", "http://example.com").selectFirst("form");
        FormElement form = new FormElement(parsed.tag(), parsed.baseUri(), parsed.attributes());
        assertEquals(0, form.elements().size());
    }

    @Test
    public void testAddElementReturnsFormAndRegistersElement() throws Exception {
        Element parsed = Jsoup.parse("<form><input name=x></form>", "http://example.com").selectFirst("form");
        FormElement form = new FormElement(parsed.tag(), parsed.baseUri(), parsed.attributes());
        Element input = parsed.selectFirst("input");
        assertSame(form, form.addElement(input));
        assertEquals(1, form.elements().size());
        assertSame(input, form.elements().get(0));
    }

    @Test
    public void testFormDataSkipsNonSubmittableElement() throws Exception {
        Element parsed = Jsoup.parse("<form><div name=x>v</div></form>", "http://example.com").selectFirst("form");
        FormElement form = new FormElement(parsed.tag(), parsed.baseUri(), parsed.attributes());
        form.addElement(parsed.selectFirst("div"));
        assertEquals(0, form.formData().size());
    }

    @Test
    public void testFormDataSkipsDisabledControl() throws Exception {
        Element parsed = Jsoup.parse("<form><input name=x disabled value=v></form>", "http://example.com").selectFirst("form");
        FormElement form = new FormElement(parsed.tag(), parsed.baseUri(), parsed.attributes());
        form.addElement(parsed.selectFirst("input"));
        assertEquals(0, form.formData().size());
    }

    @Test
    public void testFormDataSkipsControlWithoutName() throws Exception {

        assertEquals(0, form.formData().size());

        FormElement form = new FormElement(parsed.tag(), parsed.baseUri(), parsed.attributes());
        form.addElement(parsed.selectFirst("input"));
        assertEquals(1, form.formData().size());
    }

        FormElement form = new FormElement(parsed.tag(), parsed.baseUri(), parsed.attributes());

## 1. SOURCE CODE ANALYSIS
Tests exercise `elements()`, `addElement()`, `formData()` filtering and control-specific branches, and `submit()` URL validation.
The form-data checks cover submittability, disabled and unnamed controls, checkbox/radio handling, select fallback, and copy semantics.

## 2. TEST CASE DESIGN
- `testElementsInitiallyEmpty`: newly constructed form; expected size 0 because its control list starts empty.
- `testAddElementReturnsFormAndRegistersElement`: add one input; expected same form, one registered element, and identical element reference.
- `testFormDataSkipsNonSubmittableElement`: a `div`; expected 0 because its tag is not form-submittable.
- `testFormDataSkipsDisabledControl`: disabled input; expected 0 because disabled controls are skipped.
- `testFormDataSkipsControlWithoutName`: unnamed input; expected 0 because its name is empty.
- `testFormDataIncludesNamedTextInput`: one named input; expected one entry through the ordinary-control path.
- `testFormDataSkipsUncheckedCheckbox`: unchecked checkbox; expected 0 because checkbox data requires `checked`.
- `testFormDataCheckedCheckboxHasDefaultValue`: checked checkbox with no explicit value; expected one entry via the checked branch.
- `testFormDataCheckedRadioWithExplicitValue`: checked radio; expected one entry via the checked branch.
- `testFormDataSelectsSelectedOptions`: one selected option; expected one entry for that option.
- `testFormDataSelectWithoutSelectedOptionUsesFirstOption`: no selected option; expected one entry from the first-option fallback.
- `testFormDataReturnsIndependentList`: clear the returned list; expected a subsequent result of size 1 because each call builds a new list.
- `testSubmitUsesFormBaseUriWhenActionAbsent`: base URI supplied and no action; expected a non-null prepared connection.
- `testSubmitUsesResolvableAction`: resolvable action and base URI; expected a non-null prepared connection.
- `testSubmitRejectsMissingActionAndBaseUri`: no usable URL; expected `IllegalArgumentException` from URL validation.
