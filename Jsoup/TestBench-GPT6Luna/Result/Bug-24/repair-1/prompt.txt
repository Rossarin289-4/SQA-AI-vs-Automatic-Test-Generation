                    else

                    t.transition(ScriptDataDoubleEscaped);
            }

                    t.transition(Comment);
                    break;

                    t.transition(Data);

                t.transition(Data);
                return;

                t.advanceTransition(Data);

                t.transition(AfterDoctypeSystemKeyword);

                t.doctypePending.forceQuirks = true;
                t.advanceTransition(BogusDoctype);

                    break;

                    t.eofError(this);
                    t.doctypePending.forceQuirks = true;
                    t.emitDoctypePending();
                    t.transition(Data);
                    break;

                    t.error(this);

                case '\t':
                case '\n':
                case '\f':
                case ' ':
                    break;
                case '>':
                    t.emitDoctypePending();
                    t.transition(Data);
                    break;
                case eof:
                    t.eofError(this);
                    t.doctypePending.forceQuirks = true;
                    t.emitDoctypePending();
                    t.transition(Data);
                    break;
                default:
                    t.error(this);
                    t.transition(BogusDoctype);
                    // NOT force quirks
            }
        }
    },

        void read(Tokeniser t, CharacterReader r) {

                case '>':
                    t.emitDoctypePending();

                    break;

                    t.emitDoctypePending();

                    break;

            String data = r.consumeTo("]]>");
            t.emit(data);
            r.matchConsume("]]>");
            t.transition(Data);
        }
