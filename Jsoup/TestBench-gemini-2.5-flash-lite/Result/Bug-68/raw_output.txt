    @Test
    public void testInsertNodeToFosterParent() {
        HtmlTreeBuilder tb = new HtmlTreeBuilder();
        tb.initialiseParse(new StringReader(""), "", ParseErrorList.noParseError(), ParseSettings.htmlDefault);
        Element table = tb.insertStartTag("table"); // Create a table to act as a potential foster parent context
        tb.setFosterInserts(true);
        Node node = new Element("p");
        tb.insertNode(node);

        // If fosterInserts is true, the node should be appended to the table's parent
        if (tb.isFragmentParsing()) {
            // In fragment parsing, the foster parent is the context element's parent, or the doc.
            // With just a table, it's hard to guarantee the exact parent without more setup.
            // We check if it was appended somewhere.
            assertTrue(node.parent() != null);
        } else {
            // In document parsing, if table is not yet in document structure, it may go to doc.
            // If table has a parent, it should go there.
            if (table.parent() != null) {
                assertEquals(table.parent(), node.parent());
            } else {
                assertEquals(tb.getDocument(), node.parent());
            }
        }
    }
