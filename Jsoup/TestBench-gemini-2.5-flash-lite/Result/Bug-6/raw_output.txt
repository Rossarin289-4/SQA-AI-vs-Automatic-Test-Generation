    @Test
    public void testUnescapeEntityWithoutSemicolon() {
        // Base entities can be unescaped without a trailing semicolon
        String input = "amp"; // This should be treated as a valid entity if it's a base entity name
        // In the context of unescapePattern, if it matches "[a-zA-Z]+", and doesn't find a valid name,
        // it should append the original group(0).
        // For "amp", it will match the name, but if full.containsKey("amp") is true, it should be unescaped.
        // Let's assume it correctly maps to '&'
        // If not, it should fall back to appending the original string.
        // The regex is '&(#(x|X)?([0-9a-fA-F]+)|[a-zA-Z]+);?'
        // So "amp" will match [a-zA-Z]+, and the group(0) will be "amp".
        // The group(1) will be "amp".
        // full.containsKey("amp") is true. full.get("amp") is '&'.
        // charval will be '&'.
        // The condition `charval != -1 || charval > 0xFFFF` will be true.
        // So it should convert to '&'.
        assertEquals("&", Entities.unescape(input));
    }

    public void testUnescapeLargeCharacterValue() {
        // Test a character that might be outside the basic multilingual plane
        // For example, the smiley face 😊 is U+1F60A
        // In HTML, this is represented as &#128522; or &#x1F60A;
        String input = "&#128522;";
        // The code `charval > 0xFFFF` is checked.
        // If charval is 128522, it's greater than 0xFFFF.
        // The code `String c = Character.toString((char) charval);` will truncate.
        // Character.toString((char) 128522) would be problematic as it requires surrogate pairs.
        // However, the current logic takes `(char) charval`.
        // (char)128522 -> This is invalid as a direct char cast.
        // Let's check the behavior of Integer.valueOf(num, base) for large numbers.
        // Integer.MAX_VALUE is 2147483647.
        // 128522 is well within Integer range.
        // The crucial part is `charval > 0xFFFF`. If true, it means it's outside BMP.
        // The code then appends `m.group(0)` if `charval != -1 || charval > 0xFFFF` condition is met.
        // This means for values > 0xFFFF, it falls back to appending the original matched string.
        assertEquals("&#128522;", Entities.unescape(input));
    }
