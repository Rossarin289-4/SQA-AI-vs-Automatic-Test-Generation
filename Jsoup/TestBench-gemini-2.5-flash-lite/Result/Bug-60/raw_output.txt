    @Test
    public void parseComplexQuery() {
        Evaluator eval = QueryParser.parse("div.myClass > p#id ~ span[attr=val]");
        assertTrue(eval instanceof CombiningEvaluator.And); // div.myClass > p#id ~ span[attr=val]
        // This is a complex check, so let's check the main components are in order.
        // The order of checks in the and evaluator will be reversed from the query structure.
        List<Evaluator> andEvaluators = ((CombiningEvaluator.And) eval).getEvaluators();

        // The last evaluator added to the AND is the first in the list
