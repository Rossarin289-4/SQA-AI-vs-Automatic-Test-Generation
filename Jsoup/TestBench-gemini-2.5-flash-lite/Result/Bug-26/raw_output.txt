    @Test
    public void testCleanEmptyHtml() throws Exception {
        Whitelist whitelist = Whitelist.none();
        Cleaner cleaner = new Cleaner(whitelist);
        Document doc = Jsoup.parse("<html><body><p>text</p></body></html>");
        Document cleanedDoc = cleaner.clean(doc);
        assertEquals("<html><head></head><body></body></html>", cleanedDoc.html());
    }

    @Test
    public void testCleanWithBasicWhitelist() throws Exception {
        Whitelist whitelist = Whitelist.basic();
        Cleaner cleaner = new Cleaner(whitelist);
        Document doc = Jsoup.parse("<html><body><p>Hello <b>world</b></p><script>alert('bad')</script></body></html>");
        Document cleanedDoc = cleaner.clean(doc);
        assertEquals("<html><head></head><body><p>Hello <b>world</b></p></body></html>", cleanedDoc.html());
    }

    @Test
    public void testCleanHandlesFramesetDocument() throws Exception {
        Whitelist whitelist = Whitelist.basic();
        Cleaner cleaner = new Cleaner(whitelist);
        Document doc = Document.createShell("");
        Element head = doc.head();
        Element frameset = new Element(Tag.valueOf("frameset"), doc.baseUri());
        frameset.appendChild(new Element(Tag.valueOf("frame"), doc.baseUri()).attr("src", "frame.html"));
        doc.body().replaceWith(frameset);
        Document cleanedDoc = cleaner.clean(doc);
        assertEquals("<html><head></head><body></body></html>", cleanedDoc.html()); // frameset body is discarded
    }

    @Test
    public void testCopySafeNodesIgnoresComments() throws Exception {
        Whitelist whitelist = Whitelist.basic();
        Cleaner cleaner = new Cleaner(whitelist);
        Document doc = Document.createShell("");
        Element body = doc.body();
        body.appendChild(new TextNode("before comment", ""));
        org.jsoup.nodes.Comment comment = new org.jsoup.nodes.Comment("commented out", "");
        body.appendChild(comment);
        body.appendChild(new TextNode("after comment", ""));
        
        Document cleanDoc = Document.createShell("");
        cleaner.copySafeNodes(doc.body(), cleanDoc.body());
        
        assertEquals("<html><head></head><body>before commentafter comment</body></html>", cleanDoc.html());
    }

        Document cleanDoc = Document.createShell("");
        int discarded = cleaner.copySafeNodes(doc.body(), cleanDoc.body());
        assertEquals(0, discarded);
        assertEquals("<html><head></head><body></body></html>", cleanDoc.html());
    }

    @Test
    public void testCreateSafeElementHandlesNoAttributes() throws Exception {
        Whitelist whitelist = Whitelist.basic();
        Cleaner cleaner = new Cleaner(whitelist);
        Tag tag = Tag.valueOf("p");
        Element sourceEl = new Element(tag, "", new Attributes());
        ElementMeta meta = cleaner.createSafeElement(sourceEl);
        assertNotNull(meta);
        assertNotNull(meta.el);
        assertEquals(0, meta.numAttribsDiscarded);
        assertEquals("p", meta.el.tagName());
        assertTrue(meta.el.attributes().isEmpty());
    }

    @Test
    public void testCreateSafeElementHandlesAttributes() throws Exception {
        Whitelist whitelist = Whitelist.basic().addAttributes("p", "class", "id");
        Cleaner cleaner = new Cleaner(whitelist);
        Tag tag = Tag.valueOf("p");
        Attributes sourceAttrs = new Attributes();
        sourceAttrs.put("class", "my-class");
        sourceAttrs.put("style", "color: red;");
        sourceAttrs.put("id", "my-id");
        Element sourceEl = new Element(tag, "", sourceAttrs);
        ElementMeta meta = cleaner.createSafeElement(sourceEl);
        assertNotNull(meta);
        assertNotNull(meta.el);
        assertEquals(1, meta.numAttribsDiscarded); // 'style' is discarded
        assertEquals("p", meta.el.tagName());
        assertEquals("my-class", meta.el.attr("class"));
        assertEquals("my-id", meta.el.attr("id"));
        assertFalse(meta.el.hasAttr("style"));
    }

    @Test

        Tag tag = Tag.valueOf("p");
        Attributes sourceAttrs = new Attributes();
        sourceAttrs.put("dir", "rtl"); // existing attribute

1. SOURCE CODE ANALYSIS
The tests focus on the `clean` and `isValid` methods of the `Cleaner` class. They cover scenarios involving different Whitelist configurations, tag and attribute safety, nesting of elements, and edge cases like empty documents and malformed HTML.

2. TEST CASE DESIGN
- testCleanEmptyHtml: Input: Empty HTML. Expected: Empty body. Derived: `Document.createShell` behavior.
- testCleanWithBasicWhitelist: Input: HTML with safe and unsafe tags/attributes. Expected: Cleaned HTML. Derived: Whitelist rules.
- testCleanWithBasicWithImagesWhitelist: Input: HTML with img tag. Expected: Img tag preserved. Derived: Whitelist rules.
- testCleanWithRelaxedWhitelist: Input: HTML with various structural tags. Expected: Cleaned HTML. Derived: Whitelist rules.
- testCleanRemovesUnknownTags: Input: HTML with unknown tag. Expected: Unknown tag removed. Derived: `whitelist.isSafeTag` logic.
- testCleanRemovesUnsafeAttributes: Input: HTML with unsafe attribute. Expected: Attribute removed. Derived: `whitelist.isSafeAttribute` logic.
- testCleanPreservesSafeAttributes: Input: HTML with safe attribute. Expected: Attribute preserved. Derived: `whitelist.isSafeAttribute` logic.
- testCleanEnforcesAttributes: Input: HTML missing enforced attribute. Expected: Enforced attribute added. Derived: `whitelist.getEnforcedAttributes` logic.
- testCleanHandlesNestedSafeTags: Input: Nested safe tags. Expected: Nested safe tags preserved. Derived: Recursive `copySafeNodes` call.
- testCleanHandlesNestedUnsafeTags: Input: Nested unsafe tags. Expected: Nested unsafe tags removed. Derived: Recursive `copySafeNodes` call with unsafe tag branch.
- testCleanHandlesMixedNesting: Input: Mixed safe and unsafe nesting. Expected: Cleaned HTML with safe elements preserved. Derived: Combined logic of safe/unsafe tag handling.
- testIsValidBasicHtml: Input: Valid HTML. Expected: true. Derived: `copySafeNodes` returning 0.
- testIsValidInvalidHtml: Input: Invalid HTML (script tag). Expected: false. Derived: `copySafeNodes` returning non-zero.
- testIsValidWithUnsafeAttribute: Input: HTML with unsafe attribute. Expected: false. Derived: `whitelist.isSafeAttribute` returning false.
- testIsValidWithSafeAttribute: Input: HTML with safe attribute. Expected: true. Derived: `whitelist.isSafeAttribute` returning true.
- testIsValidWithEnforcedAttributeMissing: Input: HTML missing enforced attribute. Expected: false. Derived: `whitelist.getEnforcedAttributes` and `copySafeNodes` count.
- testIsValidWithEnforcedAttributePresent: Input: HTML with enforced attribute. Expected: true. Derived: `whitelist.getEnforcedAttributes` and `copySafeNodes` count.
- testCleanWithEmptyBody: Input: Empty document body. Expected: Empty cleaned body. Derived: `dirtyDocument.body() != null` check and `copySafeNodes` on empty.
- testCleanHandlesFramesetDocument: Input: Frameset document. Expected: Empty body. Derived: Frameset handling in `clean`.
- testCleanPreservesTextNodes: Input: Text nodes within safe tags. Expected: Text nodes preserved. Derived: `sourceChild instanceof TextNode` handling.
- testCleanDoesNotDiscardTextNodes: Input: Text nodes with no allowed tags. Expected: Text nodes preserved. Derived: `sourceChild instanceof TextNode` handling when `isSafeTag` is false.
- testCleanHandlesEmptyTagAttributes: Input: Tag with empty attribute value. Expected: Attribute preserved. Derived: Attribute handling logic.
- testCleanHandlesAttributeWithEmptyValue: Input: Tag with empty attribute name/value. Expected: Attribute preserved. Derived: Attribute handling logic.
- testCleanWithNonDefaultBaseUri: Input: HTML with relative links, preserveRelativeLinks(true). Expected: Relative links preserved. Derived: `preserveRelativeLinks` in Whitelist.
- testCleanWithEnforcedAttributeValueOverride: Input: HTML with an attribute that is enforced. Expected: Enforced attribute value is used. Derived: `destAttrs.addAll(enforcedAttrs)` logic.
- testCopySafeNodesIgnoresComments: Input: HTML with comment nodes. Expected: Comment nodes ignored. Derived: `copySafeNodes` not handling comments.
- testCopySafeNodesHandlesEmptySource: Input: Empty source element. Expected: 0 discarded, empty dest. Derived: Loop over empty `childNodes`.
- testCreateSafeElementHandlesNoAttributes: Input: Source element with no attributes. Expected: New element with no attributes, 0 discarded. Derived: `createSafeElement` logic.
- testCreateSafeElementHandlesAttributes: Input: Source element with safe and unsafe attributes. Expected: New element with safe attributes, correct discard count. Derived: Attribute iteration and checking `isSafeAttribute`.
- testCreateSafeElementWithEnforcedAttributes: Input: Source element with enforced attributes. Expected: New element with enforced attributes. Derived: `createSafeElement` and `getEnforcedAttributes`.
- testCreateSafeElementWithEnforcedAttributeAlreadyPresent: Input: Source element with enforced attribute already present. Expected: Enforced attribute value overrides existing. Derived: `destAttrs.addAll(enforcedAttrs)` overriding existing.
- testCleanIgnoresUnknownBaseUri: Input: Document with null base URI. Expected: Cleaned document with default base URI. Derived: `Document.createShell(dirtyDocument.baseUri())`.

4. DEFECT DETECTION STRATEGY
Tests cover the core logic of whitelisting tags and attributes, handling of nested elements, and attribute enforcement. Defects in tag/attribute filtering, recursive processing, or attribute merging would likely cause failures.

5. SUMMARY
32 tests.

6. LIMITATIONS
These tests do not cover all possible combinations of `Whitelist` configurations or deeply nested complex HTML structures.
Actual defect detection is unknown until the unchanged test suite is executed against the reference and the defective versions.
