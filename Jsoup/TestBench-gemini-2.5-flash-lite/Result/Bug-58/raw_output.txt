        String html = "<a href='page.html'>Link</a>";
        Whitelist whitelist = Whitelist.basic().preserveRelativeLinks(false);
        // This setting is on Whitelist, not directly on Jsoup.clean.
        // The Jsoup.clean method uses Cleaner, which in turn uses Whitelist.
        // So, the Whitelist object itself needs to be configured.
        // The `clean` method in `Jsoup` calls `new Cleaner(whitelist).clean(document)`.
        // The `Cleaner` class does not seem to directly expose `preserveRelativeLinks`.
        // This test might not be directly testable via the `Jsoup.clean` static method with a direct control for this.
        // However, the behavior of `absUrl` on `Element` might be influenced by the `baseUri` provided to `clean`.
        // Let's test the `absUrl` on the output of `clean` to infer this.
        Document dirtyDoc = Document.createShell("");
        dirtyDoc.baseUri("http://example.com/");
        Element body = dirtyDoc.body();
        body.appendElement("a").attr("href", "page.html");
        Cleaner cleaner = new Cleaner(whitelist);
        Document cleanDoc = cleaner.clean(dirtyDoc);
        // If preserveRelativeLinks is false, the `absUrl` on an element derived from the clean document
        // should resolve against the document's base URI.
        assertEquals("http://example.com/page.html", cleanDoc.select("a").first().absUrl("href"));
        // The `Jsoup.clean` static method itself does not have a parameter for preserveRelativeLinks.
        // This is handled by the Whitelist object passed to it.
    }
