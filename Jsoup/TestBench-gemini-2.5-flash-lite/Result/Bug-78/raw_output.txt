        File tempFile = File.createTempFile("jsoup_test_", ".html");
        try (OutputStream fos = new java.io.FileOutputStream(tempFile)) {

    @Test
    public void testLoadXmlFileWithDeclarationEncodingOverride() throws Exception {
        String xmlContent = "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n<data encoding=\"ISO-8859-1\">Value</data>";
        InputStream is = new ByteArrayInputStream(xmlContent.getBytes("UTF-8"));
        Document doc = DataUtil.load(is, null, "http://example.com", Parser.xmlParser());
        assertEquals("Value", doc.body().text());
        // The declaration's encoding should be used if no other charset is specified.
        // However, since it's read as UTF-8 initially, and the declaration is detected later,
        // the final charset used for parsing the reader will be determined by the parser itself.
        // The charset attribute in the XML declaration is more of a hint to the XML parser.
        // Jsoup's DataUtil specifically looks for the encoding attribute in XmlDeclaration.
        assertEquals("UTF-8", doc.outputSettings().charset().displayName());
    }
