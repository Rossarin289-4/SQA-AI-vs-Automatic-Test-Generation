    @Test
    public void testRangeEqualsHit() {
        CharacterReader reader = new CharacterReader("abcdef");
        // Manually populate cache for testing rangeEquals
        String cached = "abc";
        int index = (31 * (31 * (31 * 0 + 'a') + 'b') + 'c')) & (reader.stringCache.length - 1);
        reader.stringCache[index] = cached;
        assertTrue(reader.rangeEquals(0, 3, cached));
    }

    @Test
    public void testRangeEqualsMissLength() {
        CharacterReader reader = new CharacterReader("abcdef");
        String cached = "abc";
        int index = (31 * (31 * (31 * 0 + 'a') + 'b') + 'c')) & (reader.stringCache.length - 1);
        reader.stringCache[index] = cached;
        assertFalse(reader.rangeEquals(0, 4, cached)); // different length
    }

    @Test
    public void testRangeEqualsMissContent() {
        CharacterReader reader = new CharacterReader("abcdef");
        String cached = "abc";
        int index = (31 * (31 * (31 * 0 + 'a') + 'b') + 'c')) & (reader.stringCache.length - 1);
        reader.stringCache[index] = cached;
        assertFalse(reader.rangeEquals(1, 3, cached)); // different content starting from index 1
