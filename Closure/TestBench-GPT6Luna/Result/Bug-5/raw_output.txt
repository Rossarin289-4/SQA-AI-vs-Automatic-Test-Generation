public class InlineObjectLiteralsTest {
    @Test
    public void testEmptyScriptIsUnchanged() throws Exception {
        Compiler compiler = new Compiler();
        Node root = IR.script();
        new InlineObjectLiterals(compiler, new Supplier<String>() {
            public String get() { return "a"; }
        }).process(null, root);
        assertEquals(Token.SCRIPT, root.getType());
        assertFalse(root.hasChildren());

    @Test
    public void testGlobalObjectDeclarationRemainsUnsplit() throws Exception {
        Compiler compiler = new Compiler();
        Node declaration = IR.var(IR.name("obj"),
                IR.objectlit(IR.propdef(IR.stringKey("x"), IR.number(1))));
        Node root = IR.script(declaration);
        new InlineObjectLiterals(compiler, new Supplier<String>() {
            public String get() { return "a"; }
        }).process(null, root);
        assertEquals(1, root.getChildAtIndex(0).getType() == Token.VAR ? 1 : 0);
        assertEquals("obj", root.getFirstChild().getFirstChild().getString());
    }

    @Test
    public void testGlobalObjectPropertyUseRemainsUnchanged() throws Exception {
        Compiler compiler = new Compiler();
        Node declaration = IR.var(IR.name("obj"),
                IR.objectlit(IR.propdef(IR.stringKey("x"), IR.number(2))));
        Node use = IR.exprResult(IR.getprop(IR.name("obj"), IR.string("x")));
        Node root = IR.script(declaration, use);
        new InlineObjectLiterals(compiler, new Supplier<String>() {
            public String get() { return "b"; }
        }).process(null, root);
        assertEquals(2, root.getChildAtIndex(0).getType() == Token.VAR ? 2 : 0);
        assertEquals(Token.GETPROP, root.getChildAtIndex(1).getFirstChild().getType());
    }

        Compiler compiler = new Compiler();
        Node declaration = IR.var(IR.name("obj"));

        Node root = IR.script(declaration, assignment);
        new InlineObjectLiterals(compiler, new Supplier<String>() {
            public String get() { return "c"; }
        }).process(null, root);
        assertEquals(Token.VAR, root.getFirstChild().getType());
        assertEquals(Token.EXPR_RESULT, root.getChildAtIndex(1).getType());
        assertEquals(Token.ASSIGN, root.getChildAtIndex(1).getFirstChild().getType());
    }

    @Test
    public void testGlobalObjectPassedToMethodRemainsUnchanged() throws Exception {
        Compiler compiler = new Compiler();
        Node declaration = IR.var(IR.name("obj"),
                IR.objectlit(IR.propdef(IR.stringKey("x"), IR.number(4))));
        Node call = IR.exprResult(IR.call(IR.getprop(IR.name("obj"), IR.string("x"))));
        Node root = IR.script(declaration, call);
        new InlineObjectLiterals(compiler, new Supplier<String>() {
            public String get() { return "d"; }
        }).process(null, root);
        assertEquals(Token.VAR, root.getFirstChild().getType());
        assertEquals(Token.CALL, root.getChildAtIndex(1).getFirstChild().getType());
    }

    public void testGlobalDeletedPropertyRemainsUnchanged() throws Exception {

        Node declaration = IR.var(IR.name("obj"),

        Node deletion = IR.exprResult(new Node(Token.DELPROP,

        Node root = IR.script(declaration, deletion);
        new InlineObjectLiterals(compiler, new Supplier<String>() {
            public String get() { return "e"; }
        }).process(null, root);

        Node use = IR.exprResult(IR.getprop(IR.name("obj"), IR.string("missing")));
        Node root = IR.script(declaration, use);
        new InlineObjectLiterals(compiler, new Supplier<String>() {
            public String get() { return "f"; }

    @Test
    public void testGlobalNonObjectAssignmentRemainsUnchanged() throws Exception {

        Node root = IR.script(declaration);
        new InlineObjectLiterals(compiler, new Supplier<String>() {
            public String get() { return "g"; }
        }).process(null, root);
        assertEquals(Token.VAR, root.getFirstChild().getType());
        assertEquals(7.0, root.getFirstChild().getFirstChild()

    public void testGlobalUninitializedDeclarationRemainsPresent() throws Exception {

            public String get() { return "j"; }

        assertEquals("obj", root.getFirstChild().getFirstChild().getString());

    public void testObjectWithTwoPropertiesRemainsGlobal() throws Exception {
        Compiler compiler = new Compiler();
        Node object = IR.objectlit(

        assertEquals(2, root.getFirstChild().getFirstChild().getFirstChild()
                .getNext().getString().equals("right") ? 2 : 0);
    }
