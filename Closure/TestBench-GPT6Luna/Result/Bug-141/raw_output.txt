        assertSame(child, expr.getFirstChild());

    @Test
    public void testNewQualifiedNameSingleName() throws Exception {
        Node name = NodeUtil.newQualifiedNameNode("alpha", 2, 3);
        assertEquals(Token.NAME, name.getType());
        assertEquals("alpha", name.getString());
    }

    @Test
    public void testNewQualifiedNameTwoParts() throws Exception {
        Node qualified = NodeUtil.newQualifiedNameNode("alpha.beta", 2, 3);
        assertEquals("alpha.beta", qualified.getQualifiedName());
        assertEquals(Token.GETPROP, qualified.getType());
    }

    public void testNewQualifiedNameThreeParts() throws Exception {
        Node qualified = NodeUtil.newQualifiedNameNode("a.b.c", 0, 0);
        assertEquals("a.b.c", qualified.getQualifiedName());
        assertEquals("b.c", qualified.getLastChild().getString());
    }

    public void testNewFunctionNodeCreatesFunctionAndBody() throws Exception {
        Node body = new Node(Token.BLOCK);
        FunctionNode function = NodeUtil.newFunctionNode(

        assertEquals(Token.FUNCTION, function.getType());
        assertEquals("f", function.getFirstChild().getString());
        assertSame(body, function.getLastChild());
    }

    @Test
    public void testGetVarsDeclaredInBranchCollectsDeclarations() throws Exception {
        Node root = new Node(Token.BLOCK);
        Node var = new Node(Token.VAR, Node.newString(Token.NAME, "item"));
        root.addChildToBack(var);
        Collection<Node> vars = NodeUtil.getVarsDeclaredInBranch(root);
        assertEquals(1, vars.size());
        assertSame(var.getFirstChild(), vars.iterator().next());
    }

    @Test
    public void testGetVarsDeclaredInBranchKeepsFirstDuplicate() throws Exception {
        Node root = new Node(Token.BLOCK);
        Node firstVar = new Node(Token.VAR, Node.newString(Token.NAME, "item"));
        Node secondVar = new Node(Token.VAR, Node.newString(Token.NAME, "item"));
        root.addChildToBack(firstVar);
        root.addChildToBack(secondVar);
        Collection<Node> vars = NodeUtil.getVarsDeclaredInBranch(root);
        assertEquals(1, vars.size());
        assertSame(firstVar.getFirstChild(), vars.iterator().next());
    }

    @Test
    public void testGetVarsDeclaredInBranchDoesNotEnterFunction() throws Exception {
        Node root = new Node(Token.BLOCK);
        Node body = new Node(Token.BLOCK);
        body.addChildToBack(
                new Node(Token.VAR, Node.newString(Token.NAME, "inner")));
        Node function = new Node(Token.FUNCTION,
                Node.newString(Token.NAME, "f"),
                new Node(Token.LP),
                body);
        root.addChildToBack(
                new Node(Token.VAR, Node.newString(Token.NAME, "outer")));
        root.addChildToBack(function);
        Collection<Node> vars = NodeUtil.getVarsDeclaredInBranch(root);
        assertEquals(1, vars.size());
        assertEquals("outer", vars.iterator().next().getString());
    }
