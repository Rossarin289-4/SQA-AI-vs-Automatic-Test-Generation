    private JsDocInfoParser createParser(String jsDoc) {
        Config config = new Config(
                Sets.<String>newHashSet(),
                Sets.<String>newHashSet(),
                true, // parseJsDocDocumentation
                LanguageMode.ECMASCRIPT3,
                false);
        JsDocTokenStream stream = new JsDocTokenStream(jsDoc);
        Comment commentNode = null; // Not relevant for parseTypeString
        ErrorReporter errorReporter = NullErrorReporter.forNewRhino();
        return new JsDocInfoParser(stream, commentNode, SOURCE_NAME, config, errorReporter);
    }

    private Node parseTypeString(String typeString) {
