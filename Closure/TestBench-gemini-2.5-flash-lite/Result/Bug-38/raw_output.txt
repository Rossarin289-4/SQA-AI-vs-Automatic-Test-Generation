            @Override
            void appendOp(String op, boolean binOp) {
                append(op);

            // that we want to test.
            @Override
            void maybeEndStatement() {
                // This is a simplified version; in the real CodeConsumer, it handles
                // statement completion logic. For testing specific append behavior,
                // we can make it a no-op or add minimal logic if needed.
                // For now, let's assume it just calls append.
                if (statementNeedsEnded) {
                    append(";");
                    maybeLineBreak();
                    endLine();
                    statementNeedsEnded = false;
                }
                statementStarted = true;
            }

            @Override
            void maybeCutLine() {
                // no-op
            }

            @Override
            void endLine() {
                // no-op
            }

            @Override
            void notePreferredLineBreak() {
                // no-op
            }

            @Override
            void add(String newcode) {
                maybeEndStatement();

                if (newcode.length() == 0) {
                    return;
                }

                char c = newcode.charAt(0);
                if ((isWordChar(c) || c == '\\') &&
                    isWordChar(getLastChar())) {
                    append(" ");
                } else if (c == '/' && getLastChar() == '/') {
                    append(" ");
                }

                append(newcode);
            }
        };
