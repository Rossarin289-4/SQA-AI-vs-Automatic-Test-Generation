    // Helper method to create a minimal compiler instance for testing.
    private static AbstractCompiler createCompiler() {
        AbstractCompiler compiler = new Compiler();
        compiler.initCompilerOptionsIfTesting();
        return compiler;

    // Helper method to parse a JS string into a Node.
    private Node parseCode(String code) {
        Compiler compiler = (Compiler) createCompiler();
        return compiler.parse(code);
    }

    // Helper method to run the CollapseProperties pass.
    private void runCollapseProperties(String code, boolean collapseExterns, boolean inlineAliases) {
        Compiler compiler = (Compiler) createCompiler();
        Node root = compiler.parse(code);
        CollapseProperties cp = new CollapseProperties(compiler, collapseExterns, inlineAliases);
        cp.process(null, root);
    }

    @Test
    public void testInlineAliases() throws Exception {
        String code = "var ns = { val: 1 }; function f() { var x = ns; x.val = 2; }";
        runCollapseProperties(code, false, true); // inlineAliases = true
        // Expected: var ns = { val: 1 }; function f() { ns.val = 2; } (ns.val becomes ns$val)
        Node root = parseCode(code);
        Compiler compiler = (Compiler) createCompiler();
        CollapseProperties cp = new CollapseProperties(compiler, false, true);
        cp.process(null, root);

        Node ns_decl = root.getFirstChild();
        assertTrue(ns_decl.isVar());
        Node ns_objlit = ns_decl.getFirstChild().getFirstChild(); // ns = { val: 1 }
        assertTrue(ns_objlit.isObjectLit());
        assertTrue(ns_objlit.hasChildren()); // Should still have val: 1

        Node fn_decl = root.getLastChild();
        assertTrue(fn_decl.isFunction());
        Node fn_body = fn_decl.getFirstChild().getNext();
        assertTrue(fn_body.isBlock());
        Node var_x_decl = fn_body.getFirstChild();
        assertTrue(var_x_decl.isVar()); // x = ns;

        Node x_assign_val = var_x_decl.getNext(); // x.val = 2
        assertTrue(x_assign_val.isExprResult());
        assertTrue(x_assign_val.getFirstChild().isAssign());
        Node x_prop_access = x_assign_val.getFirstChild().getFirstChild();
        assertTrue(x_prop_access.isGetProp());
        Node x_name = x_prop_access.getFirstChild();
        assertTrue(x_name.isName());
        // After inlining, x.val should become ns.val, which then gets flattened to ns$val
        // The original 'x' declaration might be removed or kept depending on compiler optimizations.
        // We'll check for the replacement of x.val with ns.val
        assertEquals("ns", x_name.getString());
        assertEquals("val", x_prop_access.getLastChild().getString());
    }

1. SOURCE CODE ANALYSIS - The `process` method orchestrates the pass. It first calls `inlineAliases` if enabled, then builds the global namespace using `GlobalNamespace`, and finally iterates through `globalNames` to flatten references and collapse declarations. The core logic resides in `flattenReferencesToCollapsibleDescendantNames`, `flattenReferencesTo`, `flattenPrefixes`, `flattenNameRefAtDepth`, `flattenNameRef`, `collapseDeclarationOfNameAndDescendants`, `updateSimpleDeclaration`, `updateObjLitOrFunctionDeclaration`, and their helpers. These methods handle transforming qualified names and object literal properties into flattened names (e.g., `a.b.c` to `a$b$c`).
