        compiler.getOptions().setPrettyPrint(true);
        compiler.parse(code);
        compiler.getPassConfig().setSettings(compiler.getOptions());

        AbstractCompiler abstractCompiler = compiler;
        CodingConvention codingConvention = new CodingConvention.DefaultCodingConvention() {
            @Override
            public SubclassRelationship getClassesDefinedByCall(Node callNode) {
                if (callNode.isCall()) {
                    Node target = callNode.getFirstChild();
                    if (target.isName()) {
                        if ("Parent".equals(target.getString())) {
                            return new SubclassRelationship("Parent", "Object");
                        } else if ("Child".equals(target.getString())) {
                            return new SubclassRelationship("Child", "Parent");
                        }
                    }
                }
                return null;
            }
        };
        compiler.setCodingConvention(codingConvention);

        NameAnalyzer analyzer = new NameAnalyzer(compiler, true);
        analyzer.process(compiler.getExternsRoot(), compiler.getJsRoot());

        JsName jsNameParent = getJsName(analyzer, "Parent");
        assertNotNull(jsNameParent);

1. SOURCE CODE ANALYSIS - The tests focus on analyzing global names and their dependencies. Key methods tested include `NameAnalyzer.createNameInformation` for parsing names and identifying their types, `NameAnalyzer.recordReference` for building the dependency graph, and `NameAnalyzer.process` for orchestrating the analysis. Branches covered involve different types of declarations (var, function, object literal properties), assignments, function calls, and control flow structures.
2. TEST CASE DESIGN -
    * testGlobalDeclaration: var a = 1; -> a referenced
    * testGlobalDeclarationReferencedLater: var a; a = 1; -> a referenced
    * testGlobalDeclarationNotReferenced: var a; -> a not referenced
    * testFunctionDeclaration: function f() {} -> f not referenced
    * testFunctionDeclarationReferenced: function f() {} f(); -> f referenced
    * testFunctionDeclarationNotReferenced: function f() {} -> f not referenced
