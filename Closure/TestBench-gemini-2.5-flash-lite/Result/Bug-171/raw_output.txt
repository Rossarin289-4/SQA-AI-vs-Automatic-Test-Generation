    @Test
    public void testTraverseAssign() throws Exception {
        JSTypeRegistry registry = createRegistry();
        AbstractCompiler compiler = createCompiler();
        Node root = new Node(Token.SCRIPT);
        Scope globalScope = createScope(compiler, root);
        ReverseAbstractInterpreter rai = createReverseInterpreter(compiler, registry);
        Map<String, AssertionFunctionSpec> assertionFunctionsMap = Maps.newHashMap();
        TypeInference typeInference = createTypeInference(compiler, new ControlFlowGraph<>(), rai, globalScope, assertionFunctionsMap);

        Node assignNode = new Node(Token.ASSIGN, Node.newString("x"), Node.newNumber(5));
        assignNode.getFirstChild().setJSType(registry.getGlobalType(JSTypeNative.NUMBER_TYPE));
        assignNode.getLastChild().setJSType(registry.getGlobalType(JSTypeNative.NUMBER_TYPE));

        FlowScope initialScope = LinkedFlowScope.createEntryLattice(globalScope);
        FlowScope resultScope = typeInference.flowThrough(assignNode, initialScope);

        assertNotNull(assignNode.getJSType());
        assertEquals(registry.getGlobalType(JSTypeNative.NUMBER_TYPE), assignNode.getJSType());

        Var varX = globalScope.getVar("x");
        assertNotNull(varX);
        assertEquals(registry.getGlobalType(JSTypeNative.NUMBER_TYPE), varX.getType());
    }

    public void testTraverseReturn() throws Exception {

        Node functionNode = new Node(Token.FUNCTION);
        Node functionName = Node.newString("myFunc");
        Node parameters = new Node(Token.PARAM_LIST);
        Node functionBody = new Node(Token.BLOCK);
        Node returnNode = new Node(Token.RETURN, Node.newNumber(10));
        returnNode.getFirstChild().setJSType(registry.getGlobalType(JSTypeNative.NUMBER_TYPE));
        functionBody.addChildToBack(returnNode);
        functionNode.addChildToBack(functionName);
        functionNode.addChildToBack(parameters);
        functionNode.addChildToBack(functionBody);
        functionNode.setJSType(registry.createFunctionType(registry.getGlobalType(JSTypeNative.NUMBER_TYPE)));

        Node root = new Node(Token.SCRIPT, functionNode);
        Scope globalScope = createScope(compiler, root);
        ReverseAbstractInterpreter rai = createReverseInterpreter(compiler, registry);
        Map<String, AssertionFunctionSpec> assertionFunctionsMap = Maps.newHashMap();
        TypeInference typeInference = createTypeInference(compiler, new ControlFlowGraph<>(), rai, globalScope, assertionFunctionsMap);

        FlowScope initialScope = LinkedFlowScope.createEntryLattice(globalScope);
        // The flowThrough for RETURN is not directly called, it's handled by branchedFlowThrough.
        // We need to simulate entering the function.
        Scope functionScope = globalScope.createChildScope(functionNode);
        FlowScope entryLattice = LinkedFlowScope.createEntryLattice(functionScope);
        FlowScope returnedScope = typeInference.flowThrough(returnNode, entryLattice);

        assertNotNull(returnNode.getFirstChild().getJSType());
        assertEquals(registry.getGlobalType(JSTypeNative.NUMBER_TYPE), returnNode.getFirstChild().getJSType());
    }

    @Test
    public void testInferArguments() throws Exception {
        JSTypeRegistry registry = createRegistry();
        AbstractCompiler compiler = createCompiler();
        Node functionNode = new Node(Token.FUNCTION);
        Node functionName = Node.newString("myFunc");
        Node parameters = new Node(Token.PARAM_LIST, Node.newName("a"), Node.newName("b"));
        Node functionBody = new Node(Token.BLOCK);
        functionNode.addChildToBack(functionName);
        functionNode.addChildToBack(parameters);
        functionNode.addChildToBack(functionBody);

        JSDocInfo functionInfo = new JSDocInfo();
        FunctionType functionType = registry.createFunctionType(
                registry.getGlobalType(JSTypeNative.VOID_TYPE),
                registry.getGlobalType(JSTypeNative.NUMBER_TYPE),
                registry.getGlobalType(JSTypeNative.STRING_TYPE));
        functionType.setJSDocInfo(functionInfo);
        functionNode.setJSType(functionType);


        Node root = new Node(Token.SCRIPT, functionNode);
        Scope globalScope = createScope(compiler, root);
        ReverseAbstractInterpreter rai = createReverseInterpreter(compiler, registry);
        Map<String, AssertionFunctionSpec> assertionFunctionsMap = Maps.newHashMap();
        TypeInference typeInference = createTypeInference(compiler, new ControlFlowGraph<>(), rai, globalScope, assertionFunctionsMap);

        typeInference.inferArguments(globalScope); // Call the private method

        assertNotNull(functionNode.getJSType());
        FunctionType inferredFunctionType = functionNode.getJSType().toMaybeFunctionType();
        assertNotNull(inferredFunctionType);
