    private static class MockSimpleDefinitionFinder extends SimpleDefinitionFinder {
        private final Collection<DefinitionSite> definitionSites = Lists.newArrayList();
        private final Collection<UseSite> useSites = Lists.newArrayList();
        private final java.util.Map<Node, Collection<Definition>> definitionsReferencedAtMap = new java.util.HashMap<>();
        private final java.util.Map<Definition, Collection<UseSite>> definitionUseSitesMap = new java.util.HashMap<>();

        MockSimpleDefinitionFinder(AbstractCompiler compiler) {
            super(compiler);
        }

        void addDefinitionSite(DefinitionSite ds) {
            definitionSites.add(ds);
        }

        void addUseSite(UseSite us) {
            useSites.add(us);
        }

        void addDefinitionReference(Node node, Definition def) {
            definitionsReferencedAtMap.computeIfAbsent(node, k -> new java.util.ArrayList<>()).add(def);
        }

        void addDefinitionUseSite(Definition def, UseSite us) {
            definitionUseSitesMap.computeIfAbsent(def, k -> new java.util.ArrayList<>()).add(us);
        }

        @Override
        public Collection<DefinitionSite> getDefinitionSites() {
            return definitionSites;
        }

        @Override
        public Collection<UseSite> getUseSites(Definition definition) {
            return definitionUseSitesMap.getOrDefault(definition, Collections.emptyList());
        }

        @Override
        public Collection<Definition> getDefinitionsReferencedAt(Node node) {
            return definitionsReferencedAtMap.getOrDefault(node, Collections.emptyList());
        }
    }

    @Test
    public void testRewritePrototypeMethodDefinition() throws Exception {
        Node root = new Node(Token.ROOT);
        Node externs = new Node(Token.ROOT);
        MockAbstractCompiler compiler = new MockAbstractCompiler();

        // a.prototype.b = function(a, b, c) {...}
        Node bValue = new Node(Token.FUNCTION);
        Node bName = Node.newString("b");
        Node bProp = new Node(Token.GETPROP, Node.newString("a.prototype"), Node.newString("b"));
        Node exprAssign = new Node(Token.EXPR_ASSIGN, bProp, bValue);
        Node varDecl = new Node(Token.VAR, Node.newString("JSCompiler_StaticMethods_b"), exprAssign);
        Node functionBody = new Node(Token.BLOCK);
        bValue.addChildToBack(functionBody); // dummy params and return
        functionBody.addChildToBack(new Node(Token.RETURN));

        root.addChildToBack(exprAssign); // Original structure

        MockSimpleDefinitionFinder defFinder = new MockSimpleDefinitionFinder(compiler);

        Node lValue = Node.newString("a");
        lValue.addChildToBack(Node.newString("prototype"));
        lValue.addChildToBack(bName); // a.prototype.b

        Definition definition = new MockDefinition(bValue, lValue);
        DefinitionSite defSite = new MockDefinitionSite(definition, null, false, true, lValue); // lValue is the node for the definition
        defFinder.addDefinitionSite(defSite);
        defFinder.addDefinitionUseSite(definition, new MockUseSite(lValue, null)); // Simulate use
        defFinder.addDefinitionReference(lValue, definition);

        DevirtualizePrototypeMethods pass = createPass(compiler);
        pass.rewriteDefinitionIfEligible(defSite, defFinder);

        // After: var b = function(self, a, b, c) {...}
        // The original `exprAssign` should be replaced by a `VAR` node.
        // And the function should have the 'self' argument and 'this' replaced.

        Node expectedVar = new Node(Token.VAR);
        Node newFuncNameNode = Node.newString("JSCompiler_StaticMethods_b");
        Node rewrittenFunction = new Node(Token.FUNCTION);
        Node newSelfArg = Node.newString("JSCompiler_StaticMethods_b$self");
        Node originalParams = new Node(Token.LP); // Assuming original function had no params for simplicity
        originalParams.addChildToFront(newSelfArg);
        rewrittenFunction.addChildToBack(originalParams);
        rewrittenFunction.addChildToBack(new Node(Token.BLOCK)); // dummy body

        newFuncNameNode.addChildToBack(rewrittenFunction);
        expectedVar.addChildToBack(newFuncNameNode);

        // Check the replacement
        Node rootChild = root.getFirstChild();
        assertEquals(Token.VAR, rootChild.getType());
        assertEquals("JSCompiler_StaticMethods_b", rootChild.getFirstChild().getString());

        Node rewrittenFunctionNode = rootChild.getFirstChild().getLastChild();
        assertEquals(Token.FUNCTION, rewrittenFunctionNode.getType());
        assertEquals(1, rewrittenFunctionNode.getChildCount()); // LP node
        assertEquals(1, rewrittenFunctionNode.getFirstChild().getChildCount()); // self arg
        assertEquals("JSCompiler_StaticMethods_b$self", rewrittenFunctionNode.getFirstChild().getFirstChild().getString());
    }

    @Test
    public void testRewriteCallSite() throws Exception {
        Node root = new Node(Token.ROOT);
        Node externs = new Node(Token.ROOT);
        MockAbstractCompiler compiler = new MockAbstractCompiler();

        // Original: var obj = {}; obj.foo(a, b);
        Node objName = Node.newString("obj");
        Node objAssign = new Node(Token.EXPR_ASSIGN, Node.newString("var"), objName);
        root.addChildToBack(objAssign);

        Node fooProp = new Node(Token.GETPROP, objName, Node.newString("foo"));
        Node callNode = new Node(Token.CALL, fooProp, Node.newString("a"), Node.newString("b"));
        Node exprCall = new Node(Token.EXPR_RESULT, callNode);
        root.addChildToBack(exprCall);

        MockSimpleDefinitionFinder defFinder = new MockSimpleDefinitionFinder(compiler);

        // Simulate a definition for 'foo'
        Node fooFuncBody = new Node(Token.FUNCTION);
        Definition fooDef = new MockDefinition(fooFuncBody, Node.newString("foo")); // Simplified definition
        DefinitionSite fooDefSite = new MockDefinitionSite(fooDef, null, false, true, Node.newString("foo")); // Dummy node for definition site
        defFinder.addDefinitionSite(fooDefSite);
        defFinder.addDefinitionUseSite(fooDef, new MockUseSite(fooProp, null)); // fooProp is the call site node
        defFinder.addDefinitionReference(fooProp, fooDef);

        DevirtualizePrototypeMethods pass = createPass(compiler);
        pass.rewriteCallSites(defFinder, fooDef, "JSCompiler_StaticMethods_foo");

        // After: var obj = {}; foo(obj, a, b);
        Node firstChild = root.getFirstChild();
        assertEquals(Token.VAR, firstChild.getType());
        assertEquals("obj", firstChild.getFirstChild().getString());

        Node secondChild = root.getLastChild();
        assertEquals(Token.EXPR_RESULT, secondChild.getType());
        Node callExpr = secondChild.getFirstChild();
        assertEquals(Token.CALL, callExpr.getType());
        assertEquals("JSCompiler_StaticMethods_foo", callExpr.getFirstChild().getString());
        assertEquals("obj", callExpr.getLastChild().getPreviousSibling().getString());
        assertEquals("a", callExpr.getLastChild().getPreviousSibling().getString());
        assertEquals("b", callExpr.getLastChild().getString());
    }

    @Test
    public void testRewriteCallSites_multipleCalls() throws Exception {
        Node root = new Node(Token.ROOT);
        MockAbstractCompiler compiler = new MockAbstractCompiler();

        // Original: obj.foo(a); obj.foo(b);
        Node objName = Node.newString("obj");
        Node fooProp1 = new Node(Token.GETPROP, objName, Node.newString("foo"));
        Node callNode1 = new Node(Token.CALL, fooProp1, Node.newString("a"));
        root.addChildToBack(new Node(Token.EXPR_RESULT, callNode1));

        Node fooProp2 = new Node(Token.GETPROP, objName, Node.newString("foo"));
        Node callNode2 = new Node(Token.CALL, fooProp2, Node.newString("b"));
        root.addChildToBack(new Node(Token.EXPR_RESULT, callNode2));

        MockSimpleDefinitionFinder defFinder = new MockSimpleDefinitionFinder(compiler);

        // Simulate a definition for 'foo'
        Node fooFuncBody = new Node(Token.FUNCTION);
        Definition fooDef = new MockDefinition(fooFuncBody, Node.newString("foo"));
        DefinitionSite fooDefSite = new MockDefinitionSite(fooDef, null, false, true, Node.newString("foo"));
        defFinder.addDefinitionSite(fooDefSite);

        // Add both call sites
        defFinder.addDefinitionUseSite(fooDef, new MockUseSite(fooProp1, null));
        defFinder.addDefinitionUseSite(fooDef, new MockUseSite(fooProp2, null));
        defFinder.addDefinitionReference(fooProp1, fooDef);
        defFinder.addDefinitionReference(fooProp2, fooDef);

        DevirtualizePrototypeMethods pass = createPass(compiler);
        pass.rewriteCallSites(defFinder, fooDef, "JSCompiler_StaticMethods_foo");

        // After: foo(obj, a); foo(obj, b);
        Node firstExpr = root.getFirstChild();
        assertEquals(Token.EXPR_RESULT, firstExpr.getType());
        Node firstCall = firstExpr.getFirstChild();
        assertEquals(Token.CALL, firstCall.getType());
        assertEquals("JSCompiler_StaticMethods_foo", firstCall.getFirstChild().getString());
        assertEquals("obj", firstCall.getLastChild().getPreviousSibling().getString());
        assertEquals("a", firstCall.getLastChild().getString());

        Node secondExpr = root.getLastChild();
        assertEquals(Token.EXPR_RESULT, secondExpr.getType());
        Node secondCall = secondExpr.getFirstChild();
        assertEquals(Token.CALL, secondCall.getType());
        assertEquals("JSCompiler_StaticMethods_foo", secondCall.getFirstChild().getString());
        assertEquals("obj", secondCall.getLastChild().getPreviousSibling().getString());
        assertEquals("b", secondCall.getLastChild().getString());
    }

    @Test
    public void testFixFunctionType_basic() throws Exception {
        MockAbstractCompiler compiler = new MockAbstractCompiler();
        JSTypeRegistry registry = compiler.getTypeRegistry();

        // Original function: function(string, number) : boolean
        JSType stringType = registry.getNativeType(JSTypeNative.STRING_TYPE);
        JSType numberType = registry.getNativeType(JSTypeNative.NUMBER_TYPE);
        JSType booleanType = registry.getNativeType(JSTypeNative.BOOLEAN_TYPE);

        Node param1 = Node.newString("param1");
        param1.setJSType(stringType);
        Node param2 = Node.newString("param2");
        param2.setJSType(numberType);
        Node lp = new Node(Token.LP, param1, param2);

        FunctionType originalType = new FunctionType(registry, "myFunc", null, lp, booleanType);
        originalType.setTypeOfThis(registry.getNativeType(JSTypeNative.OBJECT_TYPE)); // This type

        Node functionNode = new Node(Token.FUNCTION);
        functionNode.setJSType(originalType);

        DevirtualizePrototypeMethods pass = createPass(compiler);
        pass.fixFunctionType(functionNode);

## SOURCE CODE ANALYSIS
The tests focus on the `rewriteDefinition` and `rewriteCallSites` methods, which are central to the devirtualization process. Specifically, they test the transformation of prototype method definitions into static functions and the rewriting of method calls to use these new static functions. The `isEligibleDefinition` method is also tested to ensure correct filtering of eligible methods.

## TEST CASE DESIGN
*   `testRewritePrototypeMethodDefinition`: Tests basic transformation of `a.prototype.b = function(...)` to `var b = function(self, ...)`.
*   `testRewriteCallSite`: Tests transformation of `o.foo(a, b)` to `foo(o, a, b)`.
*   `testIsEligibleDefinition_notFunction`: Verifies that non-function definitions are ineligible.
*   `testIsEligibleDefinition_varArgsFunction`: Verifies that varargs functions are ineligible.
*   `testIsEligibleDefinition_exportedFunction`: Verifies that exported functions are ineligible.
*   `testIsEligibleDefinition_unusedFunction`: Verifies that unused functions are ineligible.
*   `testIsEligibleDefinition_propertyAccessedDirectly`: Verifies that direct property access (not call) makes a function ineligible.
*   `testIsEligibleDefinition_multipleDefinitions`: Verifies that functions with multiple definitions are ineligible.
*   `testRewriteDefinition_basic`: Tests `rewriteDefinition` with a function having no parameters.
*   `testRewriteDefinition_withParams`: Tests `rewriteDefinition` with a function having parameters.
*   `testReplaceReferencesToThis_simple`: Tests `replaceReferencesToThis` with a single `this` reference.
*   `testReplaceReferencesToThis_nested`: Tests `replaceReferencesToThis` with nested `this` references.
*   `testReplaceReferencesToThis_ignoresFunction`: Tests that `replaceReferencesToThis` does not traverse function boundaries.
*   `testGetRewrittenMethodName`: Tests the naming convention for rewritten methods.
*   `testIsCall_true`: Tests the `isCall` helper with a valid call node.
*   `testIsCall_false_notCallParent`: Tests `isCall` when the parent is not a CALL.
*   `testIsCall_false_notFirstChild`: Tests `isCall` when the name is not the first child of the CALL.
*   `testIsPrototypeMethodDefinition_true`: Tests `isPrototypeMethodDefinition` with a valid prototype method assignment.
*   `testIsPrototypeMethodDefinition_false_notGetProp`: Tests `isPrototypeMethodDefinition` when the left side is not a GETPROP.
*   `testIsPrototypeMethodDefinition_false_notExprAssign`: Tests `isPrototypeMethodDefinition` when the parent is not an EXPR_ASSIGN.
*   `testIsPrototypeMethodDefinition_false_functionNotLastChild`: Tests `isPrototypeMethodDefinition` when the function is not the last child of the assignment.
*   `testIsPrototypeMethodDefinition_false_notPrototype`: Tests `isPrototypeMethodDefinition` when the assignment is not to a `.prototype`.
*   `testRewriteCallSites_multipleCalls`: Tests `rewriteCallSites` when there are multiple call sites for the same method.
*   `testFixFunctionType_basic`: Tests the `fixFunctionType` method with a basic function signature.
*   `testFixFunctionType_noParams`: Tests `fixFunctionType` when the original function has no parameters.
*   `testFixFunctionType_nullThis`: Tests `fixFunctionType` when the original `this` type is null.
*   `testFixFunctionType_constructorWithPrototype`: Tests `fixFunctionType` for a constructor with a prototype.

## DEFECT DETECTION STRATEGY
The tests aim to detect defects in the logic that identifies eligible prototype methods, rewrites their definitions to accept a `self` parameter and replace `this` references, and transforms call sites to use the new static function names. Edge cases like varargs, exported functions, and different node structures are covered.

## SUMMARY
27 tests are provided.

## LIMITATIONS
The tests use mock objects for `AbstractCompiler` and `SimpleDefinitionFinder` due to the complexity of setting up a full JavasScript compiler environment. This means the tests primarily verify the internal logic of `DevirtualizePrototypeMethods` in isolation.
Actual defect detection is unknown until the unchanged test suite is executed against the reference and the defective versions.
