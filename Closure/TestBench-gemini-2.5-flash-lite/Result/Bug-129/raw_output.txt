    // Helper method to create a Node with a given type and children
    private Node createNode(int type, Node... children) {
        Node n = new Node(type);
        for (Node child : children) {
            n.addChildToBack(child);
        }
        return n;
    }

    // Helper method to create a Node with a given type, source reference, and children
    private Node createNode(int type, Node srcref, Node... children) {
        Node n = new Node(type, srcref.getLineno(), srcref.getCharno());
        for (Node child : children) {
            n.addChildToBack(child);
        }
        return n;
    }

    // Helper method to set parent pointers for a subtree
    private void setParentPointers(Node parent, Node n) {
        if (n == null) return;
        n.setParent(parent);
        for (Node child = n.getFirstChild(); child != null; child = child.getNext()) {
            setParentPointers(n, child);
        }
    }

    public void testPrepareAnnotationsCalledForRoot() throws Exception {
        Node root = IR.script();

        Node fn = IR.function(null, IR.newNode(Token.PARAM_LIST), IR.block());
        root.addChildToBack(fn);
        setParentPointers(root, fn);

        // Add a call node to trigger annotateCalls
        Node call = IR.call(IR.newNode(Token.NAME));
        root.addChildToBack(call);

        // We can't directly test if PrepareAnnotations was *called* without more mocking.
        // Instead, we rely on the side effects of its methods being called.

        // Here, we'll check if specific properties are set if the conditions are met.

        // IR.call(IR.newNode(Token.NAME)) is not a GET.
        assertTrue(call.getBooleanProp(Node.FREE_CALL));

        Node evalCall = IR.call(IR.name("eval"));
        root.addChildToBack(evalCall);
        setParentPointers(root, evalCall);
        compiler.prepareAst(root);
