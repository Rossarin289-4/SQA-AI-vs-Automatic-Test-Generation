    @Test
    public void testIsFunctionTypeDeclaration_falseCases() throws Exception {
        JSDocInfo info = new JSDocInfo(false);
        // By default, JSDocInfo might not imply a function declaration without specific tags.
        // However, the current implementation of isFunctionTypeDeclaration returns true if parameterCount > 0.
        // To test false, we need an info object that has none of the indicators.
        // Let's assume an empty JSDocInfo means no function declaration hints.
        // BUT, the current logic `info.getParameterCount() > 0` is always true for a new JSDocInfo.
        // Let's try to create an info object that explicitly does NOT have parameters.
        // It's tricky as `JSDocInfo` initializes with parameterCount = 0.

        // A more robust test would involve adding tags that DON'T indicate a function type.
        // For example, a simple @param tag without a return or @constructor/@interface
        // might not be enough depending on how it's interpreted.
        // Let's test the provided logic directly:
        JSDocInfo emptyInfo = new JSDocInfo(false);
        // Default constructor for JSDocInfo has parameterCount = 0.
        // The check `info.getParameterCount() > 0` will be false.
        // Let's see if it also checks other properties.
        // The method is `info.getParameterCount() > 0 || info.hasReturnType() || ...`
        // So if none are present, it should be false.
        assertFalse(FunctionTypeBuilder.isFunctionTypeDeclaration(emptyInfo)); // This should be the case for a truly empty JSDocInfo.
    }
