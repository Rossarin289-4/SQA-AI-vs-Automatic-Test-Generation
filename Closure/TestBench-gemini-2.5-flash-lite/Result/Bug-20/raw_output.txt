    @Test
    public void testTryReplaceExitWithBreak_MatchingReturn() throws Exception {
        PeepholeSubstituteAlternateSyntax peephole = new PeepholeSubstituteAlternateSyntax(false);
        Node returnExpr = IR.number(1);
        Node returnNode = IR.returnNode(returnExpr);

        // Simulate a structure where the follow node is the same as the return
        // For simplicity, we'll directly create a break node as the follow
        Node breakNode = IR.breakNode();
        returnNode.getParent().replaceChild(returnNode, breakNode); // Temporarily replace
        Node cfgFollowNode = breakNode; // This is what computeFollowNode would return

        // Now, we need to ensure areMatchingExits would return true.
        // For this test, we'll assume computeFollowNode returns a node equivalent to the return.
        // In a real scenario, this would be more complex.
        // The key is that areMatchingExits(returnNode, followNode) should be true.
        // We'll simulate this by creating a return node that is equivalent to the break node
        // (which is a simplification, as they are different types).
        Node simulatedFollowNode = IR.returnNode(IR.number(1)); // Equivalent in expression

        // Manually call the internal logic for testing.
        // This is a bit of a hack to test a private method, but necessary for isolation.
        // The real logic involves ControlFlowAnalysis.computeFollowNode.
        // For this test, we are testing the areMatchingExits part.
        // Let's assume the structure leads to a situation where areMatchingExits is called.
        Node actualReturnNode = IR.returnNode(IR.number(1));
        // To make areMatchingExits true, the "follow" must be equivalent to the return.
        // We'll create a dummy "follow" node that is equivalent.
        Node dummyFollowNode = IR.returnNode(IR.number(1));
        // We need to simulate the context where tryReplaceExitWithBreak is called.
        // This is hard to do without ControlFlowAnalysis.
        // Let's focus on what `areMatchingExits` does.
        // If we pass the same return node twice, it should be true.
        assertTrue(peephole.areMatchingExits(actualReturnNode, actualReturnNode));

        // Now, testing tryReplaceExitWithBreak itself.
        // It replaces the return with a break if the follow node is the same.
        // We need to construct a scenario where this happens.
        // A simplified scenario: if (true) return 1; return 1;
        Node ifStatement = IR.ifNode(IR.trueNode(), IR.block(IR.returnNode(IR.number(1))));
        Node returnStatement = IR.returnNode(IR.number(1));
        Node block = IR.block(ifStatement, returnStatement);
        // If we call optimizeSubtree on returnStatement, it should be replaced by break.
        Node replacement = peephole.optimizeSubtree(returnStatement);
        assertEquals(Token.BREAK, replacement.getType());
    }
