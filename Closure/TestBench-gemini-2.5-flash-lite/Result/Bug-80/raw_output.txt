    // Helper to create a simple AST node.
    private Node createNode(int type, String value) {
        Node node = new Node(type);
        if (type == Token.STRING) {
            node.setString(value);
        } else if (type == Token.NUMBER) {
            node.setDouble(Double.parseDouble(value));
        } else if (type == Token.NAME) {
            node.setString(value);
        } else if (type == Token.TRUE) {
            // Representing boolean true as a node
        } else if (type == Token.FALSE) {
            // Representing boolean false as a node
        } else if (type == Token.NULL) {
            // Representing null as a node
        }
        return node;
    }

    private Node createNode(int type) {
        return new Node(type);
    }

    private Node createNumberNode(double value) {
        return Node.newNumber(value);
    }

    private Node createStringNode(String value) {
        return Node.newString(value);

        parent.addChildToBack(nameNode);
        Node funcNode = new Node(Token.FUNCTION);
        funcNode.setParent(parent);
        nameNode.setParent(funcNode); // This setup is a bit off, need to correctly associate
        // Correcting the AST structure for the test
        Node actualFuncDecl = new Node(Token.FUNCTION);
        actualFuncDecl.addChildToBack(new Node(Token.LP)); // Parameters
        actualFuncDecl.addChildToBack(new Node(Token.BLOCK)); // Body
        Node varDecl = new Node(Token.VAR,
                Node.newString(Token.NAME, "myVar").copyInformationFrom(actualFuncDecl));
        varDecl.addChildToBack(actualFuncDecl);
        varDecl.getFirstChild().setParent(varDecl); // NAME node is child of VAR
        actualFuncDecl.setParent(varDecl); // FUNCTION node is child of VAR

        assertEquals("myVar", NodeUtil.getFunctionName(actualFuncDecl));
    }

    @Test
    public void testGetFunctionName_assign() {
        Node parent = new Node(Token.ASSIGN);
        Node lhs = new Node(Token.GETPROP, new Node(Token.NAME, "obj"), new Node(Token.STRING, "method"));
        parent.addChildToBack(lhs);
        Node funcNode = new Node(Token.FUNCTION);
        funcNode.setParent(parent);
        lhs.setParent(funcNode); // Incorrect AST setup for example

        // Correcting AST setup for assignment
        Node funcBody = new Node(Token.BLOCK);
        Node funcParams = new Node(Token.LP);
        Node func = new Node(Token.FUNCTION, funcParams, funcBody);

        Node propName = Node.newString(Token.STRING, "method");
        Node objName = Node.newString(Token.NAME, "obj");
        Node getProp = new Node(Token.GETPROP, objName, propName);
        Node assign = new Node(Token.ASSIGN, getProp, func);
        assign.addChildToBack(func); // Function is the RHS
        func.setParent(assign);
        getProp.setParent(assign);
        objName.setParent(getProp);
        propName.setParent(getProp);


        assertEquals("obj.method", NodeUtil.getFunctionName(func));
    }
