    @Test
    public void testChainedAssignmentWithAliasing() throws Exception {
        Node varNodeA = new MockNode(Token.VAR);
        Node nameNodeA = new MockNode(Token.NAME, "a");
        varNodeA.addChildToBack(nameNodeA);
        root.addChildToBack(varNodeA);

        Node assignNode1 = new MockNode(Token.ASSIGN);
        Node nameNodeB = new MockNode(Token.NAME, "b");
        assignNode1.addChildToBack(nameNodeB);
        assignNode1.addChildToBack(nameNodeA); // b = a
        root.addChildToBack(assignNode1);

        Node assignNode2 = new MockNode(Token.ASSIGN);
        Node nameNodeC = new MockNode(Token.NAME, "c");
        assignNode2.addChildToBack(nameNodeC);
        assignNode2.addChildToBack(assignNode1); // c = (b = a)
        root.addChildToBack(assignNode2);

        GlobalNamespace gn = createNamespace(root, externsRoot);
        gn.ensureGenerated();

        // 'a' is set from global, and aliased by 'b'
        assertNotNull(gn.getNameIndex().get("a"));
        assertEquals(2, gn.getNameIndex().get("a").getRefs().size());
        assertEquals(GlobalNamespace.Ref.Type.SET_FROM_GLOBAL, gn.getNameIndex().get("a").getRefs().get(0).type);
        assertEquals(GlobalNamespace.Ref.Type.ALIASING_GET, gn.getNameIndex().get("a").getRefs().get(1).type);

        // 'b' is set from local (since it's assigned a value from 'a') and aliased by 'c'
        assertNotNull(gn.getNameIndex().get("b"));
        assertEquals(2, gn.getNameIndex().get("b").getRefs().size());
        assertEquals(GlobalNamespace.Ref.Type.SET_FROM_LOCAL, gn.getNameIndex().get("b").getRefs().get(0).type);
        assertEquals(GlobalNamespace.Ref.Type.ALIASING_GET, gn.getNameIndex().get("b").getRefs().get(1).type);

        // 'c' is set from global
        assertNotNull(gn.getNameIndex().get("c"));
        assertEquals(1, gn.getNameIndex().get("c").getRefs().size());
        assertEquals(GlobalNamespace.Ref.Type.SET_FROM_GLOBAL, gn.getNameIndex().get("c").getRefs().get(0).type);
    }

    // Helper for Node equality check
    private boolean nodesEqual(Node n1, Node n2) {
        if (n1 == n2) return true;
        if (n1 == null || n2 == null) return false;
        if (n1.getType() != n2.getType()) return false;
        if (n1.hasChildren() != n2.hasChildren()) return false;
        if (n1.hasChildren()) {
            Node c1 = n1.getFirstChild();
            Node c2 = n2.getFirstChild();
            while (c1 != null && c2 != null) {
                if (!nodesEqual(c1, c2)) return false;
                c1 = c1.getNext();
                c2 = c2.getNext();
            }
            return c1 == null && c2 == null;
        }
        return true;
