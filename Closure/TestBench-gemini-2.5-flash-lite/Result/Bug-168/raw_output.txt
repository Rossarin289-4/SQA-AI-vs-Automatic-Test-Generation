    @Test
    public void testOverriddenMethod() throws Exception {
        Node root = Node.newString(
            "function Parent() {} Parent.prototype.method = function() {}; " +
            "function Child() {} Child.prototype.method = function() {};");
        compiler.init(root);
        ScopeCreator creator = new TypedScopeCreator(compiler);
        Scope globalScope = creator.createScope(root, null);

        FunctionType parentCtorType = globalScope.getVar("Parent").getType().toMaybeFunctionType();
        ObjectType parentPrototype = parentCtorType.getPrototype();
        FunctionType childCtorType = globalScope.getVar("Child").getType().toMaybeFunctionType();
        ObjectType childPrototype = childCtorType.getPrototype();

        assertTrue(childPrototype.hasProperty("method"));
        assertNotNull(parentPrototype.getPropertyType("method"));
        assertNotNull(childPrototype.getPropertyType("method"));
    }

    @Test
    public void testVarInitializedWithFunctionAndAnnotation() throws Exception {
        Node root = Node.newString("/** @return {boolean} */ var x = function() {};");
        compiler.init(root);
        ScopeCreator creator = new TypedScopeCreator(compiler);
        Scope globalScope = creator.createScope(root, null);
        FunctionType xType = globalScope.getVar("x").getType().toMaybeFunctionType();
        assertNotNull(xType);
        assertEquals(compiler.getTypeRegistry().getBooleanType(), xType.getReturnType());
    }

    @Test
    public void testObjectLiteralWithEnumProperty() throws Exception {
        Node root = Node.newString("/** @enum {string} */ var Colors = { RED: 'r', GREEN: 'g' }; var obj = { color: Colors.RED };");
        compiler.init(root);
        ScopeCreator creator = new TypedScopeCreator(compiler);
        Scope globalScope = creator.createScope(root, null);
        ObjectType objType = globalScope.getVar("obj").getType().toMaybeObjectType();
        assertNotNull(objType);
        assertTrue(objType.hasProperty("color"));
        assertEquals(compiler.getTypeRegistry().getStringType(), objType.getPropertyType("color"));
    }

    @Test
    public void testGlobalVariableReassignment() throws Exception {
        Node root = Node.newString("var x = 1; x = 'hello';");
        compiler.init(root);
        ScopeCreator creator = new TypedScopeCreator(compiler);
        Scope globalScope = creator.createScope(root, null);
        // The type of 'x' will be inferred as a union of the types it's assigned.
        JSType xType = globalScope.getVar("x").getType();
        assertTrue(xType.isUnionType());
        assertTrue(xType.isNumberValueType());
        assertTrue(xType.isStringValueType());
    }
