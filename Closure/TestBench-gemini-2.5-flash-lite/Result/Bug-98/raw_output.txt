    @Test
    public void testReferenceCollection_isWellDefined_usageBeforeInit() throws Exception {
        ReferenceCollection collection = new ReferenceCollection();
        Node nameNodeDecl = new Node(Token.NAME);
        Node assignmentValue = Node.newNumber(10);
        Node parentDecl = new Node(Token.VAR); // VAR a = 10
        parentDecl.addChildToFront(nameNodeDecl);
        nameNodeDecl.addChildToBack(assignmentValue);
        NodeTraversal traversal = new NodeTraversal(COMPILER, DO_NOTHING_BEHAVIOR);
        BasicBlock dummyBlock = new BasicBlock(null, new Node(Token.BLOCK));
        Reference declRef = new Reference(nameNodeDecl, parentDecl, parentDecl.getParent(), dummyBlock, traversal.getScope(), "test.js");
        declRef.isInitializingDeclaration = true; // Simulate
        collection.add(declRef, traversal, createDummyVar("v"));

        // Simulate usage `a + 5` before assignment/declaration
        Node usageNameNode = new Node(Token.NAME);
        Node usageParent = new Node(Token.ADD);
        usageParent.addChildToFront(usageNameNode);
        Reference usageRef = new Reference(usageNameNode, usageParent, usageParent.getParent(), dummyBlock, traversal.getScope(), "test.js");
        collection.add(usageRef, traversal, createDummyVar("v"));

        // The order matters for isWellDefined, the first reference is assumed to be the initialization.
        // If declaration comes first, then `isInitializingDeclarationAt(0)` will be true.
        // If assignment comes first, then `isInitializingAssignmentAt(1)` will be true.
        // Here, the declaration `var a = 10` is added first, and it is an initializing declaration.
        // Then usage `a + 5` is added.
        // This means the initialization is at index 0 and usage is at index 1.
        // `initBlock.provablyExecutesBefore(references.get(i).getBasicBlock())` will be called.
        // Since both are in the same block, this should evaluate to true.
        assertTrue(collection.isWellDefined());
    }
