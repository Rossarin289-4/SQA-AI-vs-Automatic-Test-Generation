    private Node createGoogScopeCall(Node... bodyNodes) {
        Node googScopeName = IR.newNode(Token.CALL).setStaticSourceFile("test.js");
        googScopeName.addChildToBack(IR.newNode(Token.GETPROP).setStaticSourceFile("test.js"));
        googScopeName.getFirstChild().addChildToBack(IR.newNode(Token.NAME).setString("goog").setStaticSourceFile("test.js"));

        Node block = IR.block().setStaticSourceFile("test.js");

            block.addChildToBack(bodyNode);
        }
        anonymousFn.addChildToBack(block);

        return IR.exprResult(call);
    }

    // Helper to create a simple qualified name.
    private Node createQualifiedName(String name) {
        Node node = IR.newNode(Token.GETPROP).setStaticSourceFile("test.js");
        Node current = node;
        String[] parts = name.split("\\.");
        for (int i = 0; i < parts.length - 1; i++) {
            current.addChildToBack(IR.name(parts[i]).setStaticSourceFile("test.js"));
            current = IR.newNode(Token.GETPROP).setStaticSourceFile("test.js");
            node.getFirstChild().addChildToBack(current);
        }
        current.addChildToBack(IR.name(parts[parts.length - 1]).setStaticSourceFile("test.js"));
        return node;
    }
