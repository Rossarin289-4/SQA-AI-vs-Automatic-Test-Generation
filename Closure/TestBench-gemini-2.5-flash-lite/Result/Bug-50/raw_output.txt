    private PeepholeReplaceKnownMethods createOptimizer() {
        // The PeepholeReplaceKnownMethods class extends AbstractPeepholeOptimization, which requires an AbstractCompiler.
        // We'll use a dummy compiler implementation.
        return new PeepholeReplaceKnownMethods();

    private Node parseExpression(String js) {
        Compiler compiler = new Compiler();
        compiler.initCompiler(new com.google.javascript.jscomp.CompilerOptions());
        return compiler.parseSyntheticCode(js);
    }

    private Node replaceChild(Node parent, Node child, Node replacement) {
        parent.replaceChild(child, replacement);
        return replacement;
    }

    private void assertFold(String js, String expected) {
        PeepholeReplaceKnownMethods optimizer = createOptimizer();
        optimizer.reportCodeChange = new Runnable() {
            @Override
            public void run() {
                // Do nothing
            }
        };
        Node root = parseExpression(js);
        Node expression = getFirstStatement(root);
        Node optimized = optimizer.optimizeSubtree(expression);
        Node expectedNode = parseExpression(expected).getFirstChild();
        assertEquals(expectedNode.toStringTree(), optimized.toStringTree());
    }
