    private Node createScopeCallNode(String bodyCode) {
        Node scopeCall = new Node(Token.CALL, 1, 1);
        Node googScope = new Node(Token.CALL, 1, 1);
        Node goog = new Node(Token.NAME, "goog", 1, 1);
        Node scope = new Node(Token.STRING_KEY, "scope", 1, 1);
        goog.addChildToBack(googScope);
        googScope.addChildToBack(scope);
        scopeCall.addChildToBack(goog); // goog.scope

        Node anonFn = new Node(Token.FUNCTION, 1, 1);
        Node fnBody = new Node(Token.BLOCK, 1, 1);
        Node exprResult = new Node(Token.EXPR_RESULT, fnBody, 1, 1);
        anonFn.addChildToBack(exprResult);
        scopeCall.addChildToBack(anonFn);

        Node bodyRoot = Node.newString("var jscomp_anonymous_1 = function() {};", 1, 1);
        Node parserRoot = Node.newScript(bodyRoot);
        Node programBody = parserRoot.getLastChild();
        
        Node codeNode = Node.newString(bodyCode, 1, 1);
        Node statement = new Node(Token.EXPR_RESULT, codeNode, 1, 1);
        programBody.addChildToBack(statement);

        NodeTraversal.traverse(compiler, programBody, new NodeTraversal.Callback() {
            @Override
            public void visit(NodeTraversal t, Node n, Node parent) {
                if (n.isFunction() && parent.isEXPR_RESULT()) {
                    fnBody.replaceChild(fnBody.getFirstChild(), n);
                }
            }
        });
        
        Node script = Node.newScript(scopeCall);
        NodeTraversal.traverse(compiler, script, new NodeTraversal.ScopedCallback() {
            @Override
            public void enterScope(NodeTraversal t) {}
            @Override
            public void visit(NodeTraversal t, Node n, Node parent) {
                if (n.isFunction() && parent.isEXPR_RESULT() && t.getScopeDepth() == 2) {
                     fnBody.replaceChild(fnBody.getFirstChild(), n);
                }
            }
            @Override
            public void exitScope(NodeTraversal t) {}
            @Override
            public boolean shouldTraverse(NodeTraversal t, Node n, Node parent) { return true; }
        });

        return script;
    }

    private Node createNodeForQualifiedName(String qualifiedName) {
        Node node = null;
        String[] parts = qualifiedName.split("\\.");
        for (int i = parts.length - 1; i >= 0; i--) {
            Node current = Node.newString(parts[i]);
            if (node == null) {
                node = current;
            } else {
                Node parent = new Node(Token.GETPROP, 1, 1);
                parent.addChildToBack(node);
                parent.addChildToBack(current);
                node = parent;
            }
        }
        return node;
    }
