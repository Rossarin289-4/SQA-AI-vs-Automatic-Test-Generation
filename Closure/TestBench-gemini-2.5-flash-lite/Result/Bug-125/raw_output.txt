    @Test
    public void testVisitCase() throws Exception {
        MockCompiler compiler = new MockCompiler();
        JSTypeRegistry registry = new MockJSTypeRegistry();
        MockTypedScopeCreator typedScopeCreator = new MockTypedScopeCreator(compiler);
        MemoizedScopeCreator scopeCreator = new MockMemoizedScopeCreator(typedScopeCreator);
        Scope rootScope = new MockScope(new MockNode(Token.SCRIPT), null);
        TypeCheck typeCheck = createTypeCheck(compiler, registry, rootScope, scopeCreator);

        // Setup: Switch statement with a case
        Node switchNode = new MockNode(Token.SWITCH);
        Node caseNode = new MockNode(Token.CASE);
        Node caseValueNode = new MockNode(Token.NUMBER, 10.0);
        caseValueNode.setJSType(registry.getNativeType(JSTypeNative.NUMBER_TYPE));
        caseNode.addChildToBack(caseValueNode);
        switchNode.addChildToBack(new MockNode(Token.DEFAULT_CASE)); // Default case
        switchNode.addChildToBack(caseNode); // The case we're testing

        NodeTraversal traversal = new MockNodeTraversal(compiler, rootScope);
        Node parent = new MockNode(Token.BLOCK);
        parent.addChildToBack(switchNode);

        // Execute: Visit the CASE node
        typeCheck.visit(traversal, caseNode, switchNode);

        // Assert: The case value type should match the switch type.
        // Here, we are implicitly testing that the validator.expectSwitchMatchesCase doesn't report an error.
        // For a more explicit test, we'd need to mock validator.expectSwitchMatchesCase.
        // For now, we rely on the fact that if it failed, it would throw or report.
        // We assume a successful visit implies correct type checking.
        assertNotNull(caseNode.getJSType()); // Should have a type assigned.
    }
