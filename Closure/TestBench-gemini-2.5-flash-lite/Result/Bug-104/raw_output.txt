    @Test
    public void testGetLeastSupertypeWhenOtherIsSubtypeOfAnAlternate() {
        JSType stringType = mockRegistry.getNativeType(JSTypeNative.STRING_TYPE);
        JSType numberType = mockRegistry.getNativeType(JSTypeNative.NUMBER_TYPE);
        JSType objectType = mockRegistry.getNativeType(JSTypeNative.OBJECT_TYPE);

        Set<JSType> alternates = ImmutableSet.of(stringType, numberType);
        UnionType unionType = createUnionType(alternates);

        // objectType is a supertype of stringType and numberType.
        // In this context, it means 'this' is a subtype of 'that'.
        // The method's logic is: if any alternate is a subtype of 'that', return 'this'.
        // So, if 'that' is a supertype of one of the alternates, 'this' should be returned.
        // Here, stringType is NOT a subtype of objectType. So we expect getLeastSupertype(this, that) to be called.
        // The code seems to have a slight logic inversion in this path for clarity.
        // Let's test the case where 'that' is a subtype of 'this'.
        assertTrue(stringType.isSubtype(unionType));
        assertEquals(unionType, unionType.getLeastSupertype(stringType));
    }
