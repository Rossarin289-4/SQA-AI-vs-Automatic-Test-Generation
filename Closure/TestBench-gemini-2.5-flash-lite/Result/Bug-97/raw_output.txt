    private void assertOptimize(String js, String expected) {
        PeepholeFoldConstants peephole = createPeepholePass();
        Node root = Node.newString(js); // Simulating parsing a literal
        Node result = peephole.optimizeSubtree(root);
        // In a real scenario, we'd parse the JS, optimize, and then convert back to source.
        // For this simplified test, we'll just check the resulting node.
        // A more robust test would involve a proper parsing and code generation step.
        // For now, we'll focus on checking the types and values of the simplified nodes.

        // This approach is heavily simplified. A true test would parse `js`, optimize,
        // and then convert the resulting `Node` back to source and compare with `expected`.
        // Since we don't have a full compiler infrastructure here, we'll assert properties
        // of the resulting node based on expected transformations.
        if (expected.equals("true")) {
            assertTrue(result.getBooleanProp(Node.IS_TRUE_PROP));
        } else if (expected.equals("false")) {
            assertTrue(result.getBooleanProp(Node.IS_FALSE_PROP));
        } else if (expected.equals("null")) {
            assertEquals(Token.NULL, result.getType());
        } else if (expected.equals("undefined")) {
            assertEquals(Token.VOID, result.getType()); // Typically represented as VOID
        } else if (expected.startsWith("\"") && expected.endsWith("\"")) {
            assertEquals(Token.STRING, result.getType());
            assertEquals(expected.substring(1, expected.length() - 1), result.getString());
        } else {
            try {
                double expectedNum = Double.parseDouble(expected);
                assertEquals(Token.NUMBER, result.getType());
                assertEquals(expectedNum, result.getDouble(), 1e-9);
            } catch (NumberFormatException e) {
                // If it's not a number, string, boolean, null, or undefined,
                // we assume it's a complex node that shouldn't be folded.
                // This part needs refinement based on actual expected outputs.
                // For now, if it's not a simple literal, we can't assert much without
                // a full compiler.
            }
        }
    }

    // Helper to create a Node for testing
    private Node createNode(Object value) {
        if (value instanceof String) {
            return Node.newString((String) value);
        } else if (value instanceof Double) {
            return Node.newNumber(((Double) value).doubleValue());
        } else if (value instanceof Boolean) {
            return new Node(((Boolean) value) ? Token.TRUE : Token.FALSE);
        } else if (value == null) {
            return new Node(Token.NULL);
        } else if (value instanceof Integer) {
            return Node.newNumber(((Integer) value).doubleValue());
        }
        throw new IllegalArgumentException("Unsupported type: " + value.getClass());
    }

    @Test
    public void testFoldUnsignedRightShift() throws Exception {
        Node urshNode = new Node(Token.URSH, Node.newNumber(-5), Node.newNumber(2)); // -5 >>> 2 (depends on 32-bit representation)
        Node parent = new Node(Token.EXPR_RESULT, urshNode);
        PeepholeFoldConstants peephole = createPeepholePass();
        peephole.optimizeSubtree(urshNode);
        assertEquals(Token.NUMBER, urshNode.getParent().getType());
        // Java's behavior for unsigned right shift on negative numbers:
        // -5 in 32-bit two's complement is 1111...1011.
        // Shifting right by 2 zeros fills from left: 0011...1110.
        // This is (2^32 - 5) / 4, approximately.
        // A simpler way is to cast to unsigned int, then shift.
        long unsignedInt = -5L & 0xffffffffL;
        assertEquals(Double.valueOf(unsignedInt >>> 2), urshNode.getParent().getDouble(), 1e-9);
    }
