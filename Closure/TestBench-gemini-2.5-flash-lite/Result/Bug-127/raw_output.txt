    @Test
    public void testRemoveUselessReturn() throws Exception {
        // Input: function() { return; }
        Node returnNode = new Node(Token.RETURN);
        Node block = createNode(Token.BLOCK, returnNode);
        Node function = createNode(Token.FUNCTION, block);
        Node root = createNode(Token.SCRIPT, function);

        MockCompiler compiler = new MockCompiler();
        UnreachableCodeElimination pass = new UnreachableCodeElimination(compiler, true);

        // Mock CFG where the return node is reachable and has no outgoing edges (implicitly returns)
        pass.process = (externs, toplevel) -> {
            ControlFlowGraph<Node> cfg = new ControlFlowGraph<>(root, false, false);
            DiGraphNode<Node, Branch> returnGNode = cfg.createNode(returnNode);
            DiGraphNode<Node, Branch> implicitReturn = cfg.getImplicitReturn();
            cfg.connect(returnGNode, implicitReturn, Branch.UNCOND); // Connect return to implicit return
            cfg.setEntry(cfg.createNode(root));

            EliminationPass eliminationPass = pass.new EliminationPass(cfg);
            // In visit, for RETURN:
            // if (n.hasChildren()) { break; } // No children, so this is skipped.
            // We are looking for:
            // List<DiGraphEdge<Node, Branch>> outEdges = gNode.getOutEdges();
            // if (outEdges.size() == 1 && (n.getNext() == null || n.getNext().isFunction())) {
            //     Preconditions.checkState(outEdges.get(0).getValue() == Branch.UNCOND);
            //     Node fallThrough = computeFollowing(n);
            //     Node nextCfgNode = outEdges.get(0).getDestination().getValue();
            //     if (nextCfgNode == fallThrough && !inFinally(n.getParent(), n)) {
            //         removeNode(n);
            //     }
            // }
            // Here, outEdges.size() is 1 (to implicit return).
            // n.getNext() is null.
            // The fallThrough (computeFollowing(returnNode)) should be null.
            // The nextCfgNode is the implicit return.
            // So, nextCfgNode != fallThrough. Thus, it should NOT be removed by tryRemoveUnconditionalBranching.

            // However, the `visit` method also checks `gNode.getAnnotation() != GraphReachability.REACHABLE`.
            // If the return is truly the last statement and has no side effects,
            // and removeNoOpStatements is true, it *might* be removed.
            // Let's test the unconditional branch logic directly.
            // The `computeFollowing` logic is key here.
            // If computeFollowing(returnNode) returns the same node as the target of the branch, it's removed.
            // For a standalone return, computeFollowing returns the node after the block/function.
            // The target is the implicit return of the function. These are different.

            // This test case might be designed to ensure 'return;' itself is not removed
            // if it correctly exits the function.

            // Let's consider a case where `return` is followed by something that *could* be removed
            // but the `return` itself is not a no-op.
            // For this test, we'll assume `tryRemoveUnconditionalBranching` does NOT remove a simple `return;`
            // unless it's truly redundant (e.g., followed by another return).

            // A more appropriate test for removal of `return;` would be if it's within a block
            // that's already covered by an outer return.
            // For now, we'll assume it's not removed if it's the only statement.
            pass.codeChanged = false;
            eliminationPass.tryRemoveUnconditionalBranching(returnNode);
            assertFalse(pass.codeChanged); // Expect no code change for a valid return.
        };
        pass.process(null, root);
        assertNotNull(returnNode.getParent());
    }

            // The `nextCfgNode` is the loop condition. They are different.

    public void testComplexBranchingAroundReturn() throws Exception {
        // Input: function(x) { if (x) { return 1; } else { return 2; } return 3; }
        // The final return 3; should be removed.
        Node return1 = new Node(Token.RETURN, Node.newNumber(1));
        Node return2 = new Node(Token.RETURN, Node.newNumber(2));

            // Return1 -> ImplicitReturn
            // Return2 -> ImplicitReturn
            // Return3 -> ImplicitReturn (This node should not be reached from the IF)
            // This means return3 is reachable, but ONLY if the IF's branches do not cover all paths.
            // In this case, both branches return, so the code after the IF is indeed unreachable.
