    private TypeInference createTypeInference(String code) {
        Node root = compiler.parse(code);
        Node functionNode = root.getFirstChild(); // Assuming the code is a function
        Scope globalScope = new Scope.Builder(compiler).build();
        Scope functionScope = new SyntacticScopeCreator(compiler).createScope(functionNode, globalScope);
        ControlFlowGraph<Node> cfg = new ControlFlowGraph<>(compiler);
        cfg.computeCFG(root);
        ReverseAbstractInterpreter rai = new SemanticReverseAbstractInterpreter(registry);
        Map<String, AssertionFunctionSpec> assertionFunctionsMap = Maps.newHashMap();
        return new TypeInference(compiler, cfg, rai, functionScope, assertionFunctionsMap);
    }
