    @Test
    public void testRetraversalLogicForScript() throws Exception {
        MockCompiler compiler = new MockCompiler();
        MockOptimization opt = new MockOptimization();
        PeepholeOptimizationsPass pass = new PeepholeOptimizationsPass(compiler, opt);

        Node scriptNode = new Node(Token.SCRIPT);
        Node root = new Node(Token.ROOT, scriptNode);

        pass.beginTraversal();
        // Manually simulate a change
        pass.traversalState.peek().changed = true;
        pass.traversalState.peek().traverseChildScopes = true;

        // The traverse method calls visit, which calls optimizeSubtree.
        // If optimizeSubtree reports a change, the handler will set state.changed.
        // Then shouldRetraverse is called.
        // For a script node, if state.changed is true, it should retraverse.
        // The `traverse` method loop `do { ... } while (shouldRetraverse(node));`
        // should execute more than once.
        // This test is hard to isolate without mocking `traverse` itself.

        // A simpler test is to ensure `shouldVisit` pushes and `exitNode` pops correctly.
        pass.shouldVisit(scriptNode); // Pushes state
        assertTrue(pass.traversalState.states.size() > 1);
        pass.exitNode(scriptNode); // Pops state
        assertEquals(1, pass.traversalState.states.size());
    }

    @Test
    public void testVisitMethodIteratesUntilNoChange() throws Exception {
        MockCompiler compiler = new MockCompiler();
        // This optimization will increment the number twice.
        // After the first call, somethingChanged will be true.
        // The do-while loop in visit will run again.
        // The second optimization call will find the already incremented number and will not change it.
        AbstractPeepholeOptimization changingOptimization = new AbstractPeepholeOptimization() {
            private int callCount = 0;
            @Override
            Node optimizeSubtree(Node subtree) {
                if (callCount == 0 && subtree.getType() == Token.NUMBER) {
                    callCount++;
                    reportCodeChange(); // This will be ignored by visit's internal logic
                    return Node.newNumber(subtree.getDouble() + 1);
                }
                return subtree;
            }
            @Override void beginTraversal(AbstractCompiler compiler) {}
            @Override void endTraversal(AbstractCompiler compiler) {}
        };

        PeepholeOptimizationsPass pass = new PeepholeOptimizationsPass(compiler, changingOptimization);
        Node root = Node.newNumber(10.0);
        pass.visit(root); // Should run the loop inside visit until somethingChanged is false.

        // After the first call, subtree becomes 11.0. somethingChanged is true.
        // The loop runs again. The optimization is called with 11.0. It returns 11.0 (no change).
        // `somethingChanged` becomes false. The loop terminates.
        // The final node should be 11.0.
        assertEquals(11.0, root.getDouble(), 1e-9); // Check the node itself was modified.
    }

    // Test for state management: push/pop for nested scopes
    @Test
    public void testNestedScopesPushPop() throws Exception {
        MockCompiler compiler = new MockCompiler();
        PeepholeOptimizationsPass pass = new PeepholeOptimizationsPass(compiler);

        Node script = new Node(Token.SCRIPT);
        Node function1 = new Node(Token.FUNCTION, new Node(Token.BLOCK));
        Node function2 = new Node(Token.FUNCTION, new Node(Token.BLOCK));
        script.addChildToBack(function1);
        function1.addChildToBack(function2); // function2 is nested inside function1

        pass.beginTraversal();

        // Visit script
        assertTrue(pass.shouldVisit(script)); // Pushes state for script
        assertEquals(2, pass.traversalState.states.size()); // Initial + script

        // Visit function1
        assertTrue(pass.shouldVisit(function1)); // Pushes state for function1
        assertEquals(3, pass.traversalState.states.size()); // Initial + script + function1

        // Visit function2
        assertTrue(pass.shouldVisit(function2)); // Pushes state for function2
        assertEquals(4, pass.traversalState.states.size()); // Initial + script + function1 + function2

        // Exit function2
        pass.exitNode(function2);
        assertEquals(3, pass.traversalState.states.size()); // Back to script + function1

        // Exit function1
        pass.exitNode(function1);
        assertEquals(2, pass.traversalState.states.size()); // Back to script

        // Exit script
        pass.exitNode(script);
        assertEquals(1, pass.traversalState.states.size()); // Back to initial

        pass.endTraversal();
    }

    @Test
    public void testTraverseRevisitsFunctionIfChanged() throws Exception {
        MockCompiler compiler = new MockCompiler();
        // Optimization that changes a number node.
        AbstractPeepholeOptimization changingOpt = new AbstractPeepholeOptimization() {
            @Override
            Node optimizeSubtree(Node subtree) {
                if (subtree.isNumber()) {
                    reportCodeChange(); // This is key: signals a change happened.
                    return Node.newNumber(subtree.getDouble() + 1);
                }
                return subtree;
            }
            @Override void beginTraversal(AbstractCompiler compiler) {}
            @Override void endTraversal(AbstractCompiler compiler) {}
        };
        PeepholeOptimizationsPass pass = new PeepholeOptimizationsPass(compiler, changingOpt);

        Node numberNode = Node.newNumber(1.0);
        Node functionNode = new Node(Token.FUNCTION, numberNode);
        Node scriptNode = new Node(Token.SCRIPT, functionNode);

        // Track the number of times the `visit` method is called for the function node.
        // This is a proxy for re-traversal.
        int[] visitCallCount = {0};
        PeepholeOptimizationsPass passWithCounter = new PeepholeOptimizationsPass(compiler, new AbstractPeepholeOptimization() {
            @Override
            Node optimizeSubtree(Node subtree) {
                if (subtree == functionNode) {
                    visitCallCount[0]++;
                }
                return subtree; // Pass through to the real optimization
            }
            @Override void beginTraversal(AbstractCompiler compiler) {}
            @Override void endTraversal(AbstractCompiler compiler) {}
        }, changingOpt);


        passWithCounter.process(new Node(Token.EMPTY), scriptNode);

        // Because `changingOpt` reports a change on the numberNode within the function,
        // the `state.changed` flag for the function scope will be set.
        // `shouldRetraverse` will return true, causing the `do-while` loop in `traverse`
        // to run again for the function node.
        // The `visit` method for the function node should be called more than once.
        assertTrue("Visit count for function node should be greater than 1", visitCallCount[0] > 1);
    }
