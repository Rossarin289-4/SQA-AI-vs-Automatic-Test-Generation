    @Test
    public void testGetEnclosingFunction_nestedFunctions() throws Exception {
        compiler = new MockCompiler();
        Node innerFn = IR.function("inner", IR.paramList(), IR.block());
        Node outerFn = IR.function("outer", IR.paramList(), IR.block(innerFn));
        Node root = IR.script(outerFn);
        compiler.root = root;

        NodeTraversal t = new NodeTraversal(compiler, new MockCallback());
        t.traverse(root);

        // When visiting the inner function's body, the enclosing function should be 'outerFn'.
        // This requires a more complex callback to track. For simplicity, we'll check after traverse.
        // The getEnclosingFunction() depends on the current scope.
        // Let's simulate being inside the inner function by manually pushing its scope.
        // NOTE: This is a simplification for testing, real traversal handles scope stacking.

        // For this test, we can't directly test getEnclosingFunction inside the traversal loop.
        // We can however, test that it returns null when not in a function.
        // Testing nested function contexts is better done by observing callbacks if possible.
        // Given the current structure, let's verify it returns the correct function node when it's the current scope.
        // This implies we need to manually set the current node and push scope, which is what the traversal does.

        // After traversing, the current node will be the last one processed (the body of innerFn)
        // and the scope stack will be back to global. This is not ideal for testing getEnclosingFunction *during* traversal.

        // Let's test by manually setting up a state similar to being inside the inner function.
        t.pushScope(innerFn); // Simulate entering innerFn's scope
        assertEquals(outerFn, t.getEnclosingFunction()); // Depends on scope creation logic
        t.popScope();

        t.pushScope(outerFn); // Simulate entering outerFn's scope
        assertEquals(outerFn, t.getEnclosingFunction()); // Should be outerFn
        t.popScope();

        t.pushScope(root); // Simulate global scope
        assertNull(t.getEnclosingFunction()); // Should be null
        t.popScope();
    }

        parent.addChildToBack(child);
        compiler.root = IR.script(parent);

        MockCallback cb = new MockCallback();
        NodeTraversal t = new NodeTraversal(compiler, cb);

        // Create a mock refined scope that is different from the current global scope
        Scope globalScope = new Scope(compiler.getRoot(), compiler);
        globalScope.isGlobal = true;
        t.pushScope(globalScope); // Start in global scope

        Scope refinedScope = new Scope(child, compiler); // A scope for the child node

        t.traverseInnerNode(child, parent, refinedScope);

        assertFalse(t.getScope() == refinedScope); // The getScope() after traverseInnerNode will pop the refinedScope
        assertEquals(globalScope, t.getScope()); // Should return to the original global scope
    }
