    @Test
    public void testTraversal_findAliases() throws Exception {
        MockCompiler compiler = new MockCompiler();
        MockAliasTransformationHandler handler = new MockAliasTransformationHandler();
        ScopedAliases pass = new ScopedAliases(compiler, null, handler);
        ScopedAliases.Traversal traversal = pass.new Traversal();

        // Setup a simple scope with an alias declaration.
        Node scopeRoot = new Node(Token.BLOCK); // Simulating scope root
        Node aliasDecl = new Node(Token.VAR,
            Node.newString("dom", 0, 0),
            createQualifiedNameNode("goog.dom")
        );
        aliasDecl.getFirstChild().putProp(Node.QUOTED_PROP, true);
        scopeRoot.addChildToBack(aliasDecl);

        MockScope scope = new MockScope(scopeRoot);
        scope.addVar("dom", new MockVar("dom", createQualifiedNameNode("goog.dom")));

        MockNodeTraversal t = new MockNodeTraversal(compiler, traversal);
        t.setCurrentNode(scopeRoot);
        t.setCurrentScope(scope);
        t.setScopeDepth(2); // Inside goog.scope
        t.setScopeRoot(scopeRoot);

        // Simulate entering the scope where aliases are defined.
        Node googScopeCall = new Node(Token.CALL);
        Node scopeFn = new Node(Token.FUNCTION, new Node(Token.BLOCK));
        googScopeCall.addChildToBack(scopeFn);
        Node exprResult = new Node(Token.EXPR_RESULT, googScopeCall);
        exprResult.getParent(); // Ensure parent is set for traversal.
        Node parentOfExprResult = new Node(Token.SCRIPT); // Dummy parent
        parentOfExprResult.addChildToBack(exprResult);


        // We need to manually trigger enterScope with a valid context.
        // This is tricky with mocks. Let's simulate the effect of findAliases directly.
        // The actual `findAliases` is called within `enterScope`.
        // So we'll simulate that `enterScope` has already run and called `findAliases`.

        // Mocking the scenario where findAliases is called within enterScope.
        // We'll directly call findAliases and check its effects.
        // This requires setting up the scope and traversal objects as if enterScope was called.

        // Simulating the context of enterScope:
        Node scopeCallNode = new Node(Token.CALL);
        Node googScopeNameNode = Node.newString(Token.CALL, "goog.scope");
        scopeCallNode.addChildToFront(googScopeNameNode);
        Node anonFnNode = new Node(Token.FUNCTION);
        Node fnBody = new Node(Token.BLOCK);
        anonFnNode.addChildToBack(fnBody);
        scopeCallNode.addChildToBack(anonFnNode);
        Node exprResultNode = new Node(Token.EXPR_RESULT, scopeCallNode);


        // Setup for findAliases call
        Node traversalRoot = new Node(Token.SCRIPT);
        traversalRoot.addChildToBack(exprResultNode);

        // The real enterScope would set up the scope and then call findAliases.
        // We'll simulate that the traversal has reached the goog.scope call.
        // And its enterScope method is invoked.

        // Mocking the necessary setup for enterScope and findAliases
        MockCompiler mockCompilerForEnterScope = new MockCompiler();
        MockAliasTransformationHandler mockHandlerForEnterScope = new MockAliasTransformationHandler();
        ScopedAliases mockPassForEnterScope = new ScopedAliases(mockCompilerForEnterScope, null, mockHandlerForEnterScope);
        ScopedAliases.Traversal mockTraversalForEnterScope = mockPassForEnterScope.new Traversal();

        // Setting up a dummy scope and vars for findAliases
        Node mockScopeRoot = new Node(Token.BLOCK);
        MockScope mockScope = new MockScope(mockScopeRoot);
        Node aliasValueNode = createQualifiedNameNode("goog.dom");
        Node aliasDeclNode = new Node(Token.VAR, Node.newString("dom", 0, 0), aliasValueNode);
        aliasDeclNode.getFirstChild().putProp(Node.QUOTED_PROP, true);
        mockScope.addVar("dom", new MockVar("dom", aliasValueNode));
        mockScopeRoot.addChildToBack(aliasDeclNode);

        // Mocking NodeTraversal to provide the current scope and node.
        MockNodeTraversal mockT = new MockNodeTraversal(mockCompilerForEnterScope, mockTraversalForEnterScope);
        mockT.setCurrentNode(scopeCallNode); // The goog.scope call itself
        mockT.setCurrentScope(mockScope);
        mockT.setScopeRoot(mockScopeRoot);
        mockT.setScopeDepth(1); // Just outside the scope to be entered.

        // Manually call enterScope to trigger findAliases.
        // This requires simulating the parent of the goog.scope call.
        Node parentOfScopeCall = new Node(Token.SCRIPT);
        parentOfScopeCall.addChildToBack(exprResultNode);
        mockT.setCurrentNode(exprResultNode); // The node that contains the call
        mockT.getCallback().enterScope(mockT);

        // Verify that aliases were found.
        assertEquals(1, traversal.aliases.size());
        assertTrue(traversal.aliases.containsKey("dom"));
        assertEquals("goog.dom", traversal.aliases.get("dom").getInitialValue().getQualifiedName());
    }

    @Test
    public void testTraversal_aliasDefinitionsInOrder() throws Exception {
        MockCompiler compiler = new MockCompiler();
        MockAliasTransformationHandler handler = new MockAliasTransformationHandler();
        ScopedAliases pass = new ScopedAliases(compiler, null, handler);
        ScopedAliases.Traversal traversal = pass.new Traversal();

        Node root = new Node(Token.SCRIPT);
        Node googScopeCall = new Node(Token.CALL);
        Node scopeFn = new Node(Token.FUNCTION);
        Node scopeBody = new Node(Token.BLOCK);

        Node alias1 = new Node(Token.VAR,
            Node.newString("a", 0, 0),
            createQualifiedNameNode("goog.a")
        );
        alias1.getFirstChild().putProp(Node.QUOTED_PROP, true);
        Node alias2 = new Node(Token.VAR,
            Node.newString("b", 0, 0),
            createQualifiedNameNode("goog.b")
        );
        alias2.getFirstChild().putProp(Node.QUOTED_PROP, true);

        scopeBody.addChildToBack(alias1);
        scopeBody.addChildToBack(alias2);
        scopeFn.addChildToBack(scopeBody);
        googScopeCall.addChildToBack(scopeFn);
        root.addChildToBack(new Node(Token.EXPR_RESULT, googScopeCall));

        // Mocking traversal context to have aliases populated
        MockScope mockScope = new MockScope(scopeBody);
        mockScope.addVar("a", new MockVar("a", createQualifiedNameNode("goog.a")));
        mockScope.addVar("b", new MockVar("b", createQualifiedNameNode("goog.b")));
        traversal.aliases.put("a", mockScope.getVar("a"));
        traversal.aliases.put("b", mockScope.getVar("b"));

        // Manually call visit to simulate traversal and populating aliasDefinitionsInOrder
        MockNodeTraversal t = new MockNodeTraversal(compiler, traversal);
        t.setCurrentScope(mockScope);
        t.setScopeDepth(2);
        t.setScopeRoot(scopeBody);

        // Visit alias1
        t.setCurrentNode(alias1.getFirstChild()); // The NAME node "a"
        traversal.visit(t, alias1.getFirstChild(), alias1); // The Var node itself
        // The actual logic is in visit, which checks scope depth and then identifies VAR nodes.
        // We need to ensure visit is called with the correct context.
        // The visit method will identify the VAR node associated with an alias.

        // Let's refine this: the `visit` method is called on each node.
        // We need to call it on the `NAME` node which is the Var declaration.
        // The `visit` method logic for `t.getScopeDepth() == 2` and `aliasVar != null && NodeUtil.isLValue(n)`
        // is what adds to `aliasDefinitionsInOrder`.

        // Simulating the call to visit for the declaration of 'a'.
        // The `n` in visit will be the `NAME` node 'a'.
        t.setCurrentNode(alias1.getFirstChild());
        traversal.visit(t, alias1.getFirstChild(), alias1); // Visiting the NAME node 'a'

        // Simulating the call to visit for the declaration of 'b'.
        t.setCurrentNode(alias2.getFirstChild());
        traversal.visit(t, alias2.getFirstChild(), alias2); // Visiting the NAME node 'b'

        assertEquals(2, traversal.getAliasDefinitionsInOrder().size());
        assertEquals("a", traversal.getAliasDefinitionsInOrder().get(0).getString());
        assertEquals("b", traversal.getAliasDefinitionsInOrder().get(1).getString());
    }
