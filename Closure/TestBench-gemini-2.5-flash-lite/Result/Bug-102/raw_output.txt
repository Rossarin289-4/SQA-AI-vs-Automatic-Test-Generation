    // Mock SourceAstFactory to create Nodes
    private static class MockSourceAstFactory {
        Node createNode(String code) {
            // This is a simplified parser for testing. In a real scenario,
            // you'd use Rhino's Parser.
            // For this example, we'll just create a basic SCRIPT node.
            Node script = new Node(Token.SCRIPT);
            // A very basic attempt to parse code. This is NOT a real parser.
            // We are simulating the structure that Normalize might expect.
            if (code.contains("var")) {
                script.addChildToBack(new Node(Token.VAR, new Node(Token.NAME, 0, 0)));
            } else if (code.contains("for")) {
                script.addChildToBack(new Node(Token.FOR, new Node(Token.EMPTY), new Node(Token.EMPTY), new Node(Token.EMPTY)));
            } else if (code.contains("while")) {
                script.addChildToBack(new Node(Token.WHILE, new Node(Token.TRUE)));
            } else if (code.contains("=")) {
                Node assign = new Node(Token.ASSIGN, new Node(Token.NAME, 0, 0), new Node(Token.NUMBER, 1.0));
                script.addChildToBack(new Node(Token.EXPR_RESULT, assign));
            }
            return script;
        }
    }
