    // Helper method to create a TypeCheck instance.
    private TypeCheck createTypeCheck(AbstractCompiler compiler) {
        JSTypeRegistry registry = new JSTypeRegistry(new java.util.logging.Logger(null, null));
        Scope topScope = new Scope.mišScope(compiler.getCodingConvention()); // Mocked Scope
        return new TypeCheck(compiler, new ReverseAbstractInterpreter(registry), registry, topScope, new MemoizedScopeCreator(new TypedScopeCreator(compiler)), CheckLevel.WARNING, CheckLevel.OFF);
    }
