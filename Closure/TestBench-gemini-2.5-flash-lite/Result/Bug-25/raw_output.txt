    @Test
    public void testGetBooleanOutcomes_andCondition() {
        BooleanLiteralSet leftTrue = BooleanLiteralSet.get(true);
        BooleanLiteralSet leftFalse = BooleanLiteralSet.get(false);
        BooleanLiteralSet both = BooleanLiteralSet.BOTH;

        // left is true, condition is true -> right is evaluated
