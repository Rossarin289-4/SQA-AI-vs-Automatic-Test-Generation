        root.getFirstChild().addChildToBack(new Node(Token.LP));

        functionBody.addChildToBack(new Node(Token.VAR));
        functionBody.getFirstChild().addChildToBack(new Node(Token.NAME, "x"));
        root.getFirstChild().addChildToBack(functionBody);

        Scope globalScope = tsc.createScope(root, null);
        Scope localScope = globalScope.getInnerScopeChains().iterator().next();
        Var varX = localScope.getVar("x");

        assertNull(varX.getType());
