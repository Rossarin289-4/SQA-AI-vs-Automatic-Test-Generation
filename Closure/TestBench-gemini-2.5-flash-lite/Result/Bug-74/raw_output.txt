    private PeepholeFoldConstants getOptimiser(Node node) {
        // Need to provide a compiler instance for reportCodeChange to work
        // A better approach would be to mock AbstractCompiler more thoroughly or
        // to have peepholeFoldConstants take a Compiler instance in its constructor.
        // For now, we'll use a mock that allows reportCodeChange to be called.
        // The PeepholeFoldConstants class's reportCodeChange calls this.reportCodeChange() directly
        // which means we need to set up the PeepholeFoldConstants instance with a mock.
        // This setup is a bit indirect. A cleaner way would be to inject the compiler.
        return peepholeFoldConstants;

    private Node optimize(Node subtree) {
        codeChanged = false; // Reset for each test
        // NodeUtil.setCompiler(new MockCompiler()); // Set a mock compiler for NodeUtil if needed
        // PeepholeFoldConstants.setCompiler(new MockCompiler()); // Set a mock compiler for PeepholeFoldConstants if needed
        // The current PeepholeFoldConstants directly calls reportCodeChange() without a compiler context.
        // We've overridden reportCodeChange in the test class to track changes.

        // The actual optimization method is optimizeSubtree.
        // We need to create a parent node for subtree so that replaceChild works correctly.
        Node parent = new Node(Token.EXPR_RESULT, subtree);
        peepholeFoldConstants.optimizeSubtree(subtree);
        if (codeChanged) {
            // If code changed, the subtree might have been replaced.
            // We need to return the potentially new root of the modified subtree.
            // In this setup, optimizeSubtree replaces the node in its parent.
            // So we return the potentially new first child of the EXPR_RESULT.
            return parent.getFirstChild();
        } else {
            return subtree; // No change, return original
        }
    }

    @Test
    public void testFoldStringSubstrNegativeStart() throws Exception {
        // This test case is to check if the current implementation handles negative start index
        // According to the spec, negative start means offset from the end.
        // For "abcdef".substr(-2, 2) -> "ef"
        Node stringLiteral = Node.newString("abcdef");
        Node start = Node.newNumber(-2);
        Node length = Node.newNumber(2);
        Node substrCall = new Node(Token.CALL, new Node(Token.GETPROP, stringLiteral, Node.newString("substr")), start, length);
        Node result = optimize(substrCall);
        // The current implementation does not fold negative start indices, so it should return the original node or an unfolder version.
        // If it were to be folded correctly, it should be "ef".
        // Since it is not handled, we expect it to remain as a CALL node.
        assertEquals(Token.CALL, result.getType());
    }
