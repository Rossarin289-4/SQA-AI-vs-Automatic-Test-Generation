        // The intent is to test the logic path where isNumericResult and isBooleanResult return false.
        // A more robust test would involve mocking or a more complex AST structure.
        // For now, we rely on the fact that isNumericResult(add with unknown) is false.
        Node left = Node.newNumber(1); // Will be evaluated as numeric by isNumericResult
        Node right = Node.newString("true"); // Will be evaluated as non-boolean by isBooleanResult

        // The isNumericResult(add) check will fail because one operand is string.
        // The isBooleanResult check will fail.
        // Thus, mayBeStringHelper should return true.
        assertTrue(NodeUtil.mayBeString(add));
    }
