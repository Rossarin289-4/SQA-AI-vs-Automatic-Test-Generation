            if (column == 1 && row == 0) return 5.0;  // Category2, Series1 (first value for Category2)
            if (column == 1 && row == 1) return 15.0; // Category2, Series1 (second value for Category2) - Note: the original code implies one row per category for min/max line
            if (column == 1 && row == 2) return 2.0;  // Category2, Series1 (third value for Category2) - This indicates the dataset structure might be simplified for this test
            return null;
        });
        // Adjusting the dataset structure for how drawItem processes it: assumes one row per category for min/max
        DefaultCategoryDataset datasetForDraw = new DefaultCategoryDataset();
        datasetForDraw.addValue(10.0, "Series1", "Category1");
        datasetForDraw.addValue(5.0, "Series1", "Category2"); // Minimum for Category2
        datasetForDraw.addValue(15.0, "Series1", "Category2"); // Maximum for Category2
        // For simplicity in testing drawItem's min/max logic, we'll assume the dataset provides distinct min/max points.
        // The actual dataset structure for min/max is often a single row with multiple values.
        // The loop structure in drawItem processes row by row, and the `dataset.getRowCount() - 1 == row` condition
        // implies the last row processed for a category is where the min/max line is drawn.

        datasetSimplified.addValue(15.0, "Series1", "Category2"); // This will update max for Category2
        // The current drawItem logic assumes dataset.getRowCount() - 1 == row to draw the line. This is problematic if there are multiple series.

        // The logic `if (dataset.getRowCount() - 1 == row)` suggests the min/max line is drawn only for the last row of the *entire dataset*.
        // This seems incorrect for a category renderer. The logic `this.lastCategory == column` and then checking `dataset.getRowCount() - 1 == row` implies
        // that for each category, when the last row *of the dataset* is processed *and* it's for the current `column`, the line is drawn.
        // This is highly unusual. Let's assume a dataset with a single series and multiple categories for now.

        // Re-attempting the test with a clearer understanding of the `drawItem` logic's intended behavior for min/max.
        // The code updates `min` and `max` for the current `column`. When `dataset.getRowCount() - 1 == row`, it draws the line.
        // This means the line is drawn only when the *last row of the dataset* is processed, *if* that row belongs to the current category.
        // This implies a single series, where each row is a category.
        DefaultCategoryDataset datasetSingleSeriesCategories = new DefaultCategoryDataset();
        datasetSingleSeriesCategories.addValue(10.0, "Series1", "Category1"); // row 0, col 0
        datasetSingleSeriesCategories.addValue(5.0, "Series1", "Category2");  // row 1, col 1 (Min for Cat2)
        datasetSingleSeriesCategories.addValue(15.0, "Series1", "Category2"); // row 2, col 1 (Max for Cat2) - This value would update the max if it were the last row.
        // The current code structure implies that the min/max is tracked *per category* and drawn when the *last row of the entire dataset* is processed *and* it belongs to that category.
        // This is confusing. Let's assume the intent is to draw the min/max line for a *category* when the *last series value for that category* is processed.
        // The current implementation of `drawItem` doesn't seem to support that directly if there are multiple series or if the last row doesn't contain the max/min.

        // The line is drawn between `(x1, minY)` and `(x1, maxY)`.
        // The icons are drawn at `(x1, minY)` and `(x1, maxY)`.

        // Mocking the calls to simulate drawing for Category2, which is the last category.
        // For this test to work, we need to simulate the state that `drawItem` would have when processing the last category's last series value.
        // This is complex due to the internal state (`lastCategory`, `min`, `max`).

        // Mocking dataset.getRowCount() to return 3, and dataset.getColumnCount() to return 2.
        // The call to drawItem for (row=1, column=1) will set min=5, max=5, lastCategory=1.
        // The call to drawItem for (row=2, column=1) will update min=2, max=15.

        when(dataset.getValue(2, 1)).thenReturn(2.0);  // Third call for category 1 (updates min) - Wait, this structure is wrong.
        // The loop is `for (row) { for (column) { dataset.getValue(row, column) } }`
        // Or `for (column) { for (row) { dataset.getValue(row, column) } }`
        // The provided `drawItem` takes `row` and `column`.
        // The logic `lastCategory = column` and `dataset.getRowCount() - 1 == row` means the min/max is drawn only when the last row of the dataset is processed, for the current column.

        datasetForCategory2Last.addValue(10.0, "Series1", "Category1"); // row 0, col 0
        datasetForCategory2Last.addValue(5.0, "Series1", "Category2"); // row 0, col 1 -> min=5, max=5
        datasetForCategory2Last.addValue(20.0, "Series1", "Category2"); // row 1, col 1 -> min=5, max=20
        datasetForCategory2Last.addValue(15.0, "Series2", "Category1"); // row 2, col 0
        datasetForCategory2Last.addValue(25.0, "Series2", "Category2"); // row 3, col 1 -> min=5, max=25. And this is the last row (row 3).

        // The loop in drawItem seems to be intended for a single series.
        // The `this.lastCategory = column` reset happens *for every new column*.
        // The `if (dataset.getRowCount() - 1 == row)` means the min/max line is drawn only when the last row of the dataset is processed, and if `lastCategory` matches the `column` of that last row.
        // This is very peculiar. It means the min/max is only drawn for the category in the last row of the dataset.

        // Dataset:
        // Cat1: Series1=10, Series2=15
        // Cat2: Series1=5, Series1=20, Series2=25
        // Row indices: 0, 1, 2, 3. Column indices: 0, 1.
        // When row=3, column=1 (value 25): lastCategory was 0 (from Series2, Cat1). Now lastCategory becomes 1.
        // min is updated to 25, max is updated to 25.
        // The line is drawn for Category2. This assumes min/max are determined from the *last* series processed for that category.

        // Let's simplify the test by focusing on the properties of the renderer itself.
        // We can test the getters and setters and how they affect the object's state.
