    @Test
    public void testRemoveAgedItemsTrue() {
        TimeSeries ts = new TimeSeries("Test Series");
        ts.setMaximumItemAge(1); // Keep only items within 1 period of each other
        ts.add(new Day(1, 1, 2000), 10.0);
        ts.add(new Day(2, 1, 2000), 11.0);
        ts.add(new Day(3, 1, 2000), 12.0);

        // The difference between Day 3 and Day 1 is 2 periods, which is > maximumItemAge (1)
        // So, Day 1 should be removed.

    @Test
    public void testRemoveAgedItemsWithLatestAndNotify() {
        TimeSeries ts = new TimeSeries("Test Series");
        ts.setMaximumItemAge(1); // Keep items within 1 period

        RegularTimePeriod p1 = new Day(1, 1, 2000); // serial index 1
        RegularTimePeriod p2 = new Day(2, 1, 2000); // serial index 2
        RegularTimePeriod p3 = new Day(3, 1, 2000); // serial index 3

        ts.add(p1, 10.0);
        ts.add(p2, 11.0);
        ts.add(p3, 12.0);

        // Use a date corresponding to Day 2 (serial index 2) as the 'latest' reference
        // We need to create a Date object for Day 2. Day constructor takes month, day, year.
        // Days are 1-based. Months are 1-based. Years are AD.
        Day dayForLatest = new Day(2, 1, 2000);
        long latestMillis = dayForLatest.getFirstMillisecond();

        ts.removeAgedItems(latestMillis, true);

        // The condition is (index - getTimePeriod(0).getSerialIndex()) > this.maximumItemAge
        // For p1 (index 0, serial index 1), latest index is 2. (2 - 1) = 1. 1 is not > 1. So p1 should remain.
        // The logic in the method is a bit confusing, let's re-evaluate based on the condition.
        // (index of latest period - serial index of period at index 0) > maxItemAge
        // Here, latest period is Day 2 (index 2). Oldest period is Day 1 (index 0).
        // serial index of latest period is 2. serial index of period at index 0 is 1.
        // if latest is Day 2 (serial index 2):
        // check p1 (serial index 1): (2 - 1) = 1. Is 1 > 1? No. So p1 is NOT removed by this logic.
        // The loop condition `while ((latest - getTimePeriod(0).getSerialIndex()) > this.maximumItemAge)`
        // in the other removeAgedItems seems to imply a span.

        // Let's assume the `latest` param is a reference point and we remove items older than `latest - maximumItemAge`.
        // If latest is Day 2 (serial index 2), and maxItemAge is 1. We remove items older than index 2 - 1 = 1.
        // So, items with serial index 0 should be removed. p1 has serial index 1. p2 has 2. p3 has 3.
        // This interpretation doesn't seem right with the method signature.

        // Let's trace the method `removeAgedItems(long latest, boolean notify)` more closely.
        // It finds a `newest` period based on `latest` date.
        // Then it checks `(index - getTimePeriod(0).getSerialIndex()) > this.maximumItemAge`.
        // This is where it seems to be comparing the actual time period's serial index with the maximum age.
        // The `latest` parameter seems to be used to determine the time period class if it's null.

        // Re-testing `removeAgedItems(boolean notify)` which is more straightforward.
        // The test `testRemoveAgedItemsWithLatestAndNotify` might be testing a complex edge case
        // that is not well-defined by the current implementation's use of `latest`.

        // Let's simulate the expected behavior based on the `setMaximumItemAge` documentation:
        // "all data items more than 30 days older than the latest value are automatically dropped"
        // Here, `latest` means the latest item in the series itself, not the `latest` parameter.
        // So, `removeAgedItems(boolean notify)` is the primary mechanism.

        // The `removeAgedItems(long latest, boolean notify)` method seems to be intended to remove items
        // older than `latest - maximumItemAge`.
        // If `latest` is the millisecond representation of Day 2 (serial index 2), and `maximumItemAge` is 1.
        // `newest` period will be Day(2, 1, 2000). Its serial index is 2.
        // The `while` loop condition: `(index - getTimePeriod(0).getSerialIndex()) > this.maximumItemAge`
        // where `index` is the serial index of the `newest` period.
        // For `p1` (serial index 1), `p2` (serial index 2), `p3` (serial index 3).
        // `getTimePeriod(0)` is `p1` (serial index 1).
        // `newest.getSerialIndex()` is 2.
        // Loop: `while (getItemCount() > 0 && (2 - getTimePeriod(0).getSerialIndex()) > 1)`
        // Iteration 1: `getTimePeriod(0)` is `p1` (serial index 1). `(2 - 1) > 1` is `1 > 1`, which is false.
        // So, nothing is removed by this method if the data is sequential and exactly matches `maximumItemAge`.

        // Let's try with a wider range.
        TimeSeries ts2 = new TimeSeries("Test Series 2");
        ts2.setMaximumItemAge(1); // Max age of 1 period

        RegularTimePeriod p10 = new Day(1, 1, 2000); // serial index 1
        RegularTimePeriod p11 = new Day(11, 1, 2000); // serial index 11
        RegularTimePeriod p21 = new Day(21, 1, 2000); // serial index 21

        ts2.add(p10, 10.0);
        ts2.add(p11, 11.0);
        ts2.add(p21, 12.0);

        // latest date is Day 11 (serial index 11)
        Day dayForLatest2 = new Day(11, 1, 2000);
        long latestMillis2 = dayForLatest2.getFirstMillisecond();

        ts2.removeAgedItems(latestMillis2, true);

        // newest period is Day 11 (serial index 11).
        // oldest item is p10 (serial index 1).
        // The loop condition `while (getItemCount() > 0 && (11 - getTimePeriod(0).getSerialIndex()) > 1)`
        // Iteration 1: `getTimePeriod(0)` is `p10` (serial index 1). `(11 - 1) > 1` is `10 > 1`, which is true.
        // p10 is removed.
        // Iteration 2: `getTimePeriod(0)` is now `p11` (serial index 11). `(11 - 11) > 1` is `0 > 1`, which is false.
        // Loop terminates.
        // Expected: p10 removed. p11 and p21 remain.

        assertEquals(2, ts2.getItemCount());
        assertEquals(p11, ts2.getTimePeriod(0));
        assertEquals(11.0, ts2.getValue(0).doubleValue(), 0.0001);
