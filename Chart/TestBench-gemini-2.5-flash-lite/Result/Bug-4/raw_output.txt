    private static class DummyValueAxis extends ValueAxis {
        private Range range;
        private boolean inverted;

        public DummyValueAxis(String label, Range range, boolean inverted) {
            super(label, null); // No standardTickUnits needed for this test
            this.range = range;
            this.inverted = inverted;
        }

        @Override
        public Range getRange() {
            return this.range;
        }

        @Override
        public void setRange(Range range) {
            this.range = range;
        }

        @Override
        public boolean isInverted() {
            return this.inverted;
        }

        @Override
        public void resizeRange(double factor) {
            // Simplified implementation for testing
            double oldRange = range.getLength();
            double newRange = oldRange * factor;
            double lower = range.getLowerBound();
            double upper = range.getUpperBound();

            if (inverted) {
                upper = lower + newRange;
            } else {
                lower = upper - newRange;
            }
            setRange(new Range(lower, upper));
        }

        @Override
        public void resizeRange2(double factor, double anchor) {
            // Simplified implementation for testing
            double oldRange = range.getLength();
            double currentLower = range.getLowerBound();
            double currentUpper = range.getUpperBound();

            double newRange = oldRange * factor;

            if (anchor < currentLower || anchor > currentUpper) {
                // Anchor is outside the current range, zoom to center
                double mid = (currentLower + currentUpper) / 2.0;
                if (inverted) {
                    setRange(new Range(mid - newRange / 2.0, mid + newRange / 2.0));
                } else {
                    setRange(new Range(mid - newRange / 2.0, mid + newRange / 2.0));
                }
            } else {
                // Anchor is inside the current range
                if (inverted) {
                    // For inverted axis, higher value means lower on screen
                    setRange(new Range(anchor - newRange, anchor));
                } else {
                    setRange(new Range(anchor, anchor + newRange));
                }
            }
        }

        @Override
        public void pan(double percent) {
            // Simplified implementation for testing
            double length = range.getLength();
            double shift = length * percent;
            if (inverted) {
                setRange(new Range(range.getLowerBound() - shift, range.getUpperBound() - shift));
            } else {
                setRange(new Range(range.getLowerBound() + shift, range.getUpperBound() + shift));
            }
        }

        @Override
        public double valueToJava2D(double value, Rectangle2D dataArea, RectangleEdge edge) {
            // Dummy implementation, not crucial for most tests
            return 0.0;
        }

        @Override

            // Dummy implementation, not crucial for most tests
            return 0.0;
        }

        @Override
        public void configure() {
            // Dummy implementation
        }
    }
