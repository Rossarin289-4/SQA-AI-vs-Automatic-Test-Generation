    @Test
    public void testGetFirstMillisecond() throws Exception {
        Week week = new Week(1, 2023); // Week 1 of 2023
        Calendar cal = Calendar.getInstance(TimeZone.getTimeZone("GMT"), Locale.UK);
        cal.set(2023, Calendar.JANUARY, 1, 0, 0, 0);
        cal.set(Calendar.MILLISECOND, 0);
        // The first millisecond of week 1 of 2023 (which typically starts on Jan 1st or before)
        // We need to set the calendar to the start of the week.
        // For week 1, 2023, Jan 1st 2023 was a Sunday. If Sunday is first day of week, then Jan 1st is start.
        cal.set(Calendar.DAY_OF_WEEK, cal.getFirstDayOfWeek());
        cal.set(Calendar.HOUR_OF_DAY, 0);
        cal.set(Calendar.MINUTE, 0);
        cal.set(Calendar.SECOND, 0);
        cal.set(Calendar.MILLISECOND, 0);

        // The exact value depends on the default Calendar and TimeZone of the test environment.
        // Let's use a known date and ensure it's the start of the week.
        Calendar testCal = Calendar.getInstance(TimeZone.getTimeZone("GMT"), Locale.UK);
        testCal.set(2023, Calendar.JANUARY, 1); // Jan 1st 2023 is a Sunday
        testCal.set(Calendar.HOUR_OF_DAY, 0);
        testCal.set(Calendar.MINUTE, 0);
        testCal.set(Calendar.SECOND, 0);
        testCal.set(Calendar.MILLISECOND, 0);
        
        // If Sunday is the first day of the week
        if (testCal.getFirstDayOfWeek() == Calendar.SUNDAY) {
             assertEquals(testCal.getTimeInMillis(), week.getFirstMillisecond());
        } else {
            // If Monday is the first day, then week 1 of 2023 started on Dec 26, 2022.
            Calendar prevCal = Calendar.getInstance(TimeZone.getTimeZone("GMT"), Locale.UK);
            prevCal.set(2022, Calendar.DECEMBER, 26);
            prevCal.set(Calendar.HOUR_OF_DAY, 0);
            prevCal.set(Calendar.MINUTE, 0);
            prevCal.set(Calendar.SECOND, 0);
            prevCal.set(Calendar.MILLISECOND, 0);
            assertEquals(prevCal.getTimeInMillis(), week.getFirstMillisecond());
        }
    }
