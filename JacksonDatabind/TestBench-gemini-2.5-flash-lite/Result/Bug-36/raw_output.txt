    }

    @Test
    public void testFormatWithPlainDate() throws Exception {
        StdDateFormat sdf = new StdDateFormat();
        // Forcing the format to be plain "yyyy-MM-dd" is not directly supported by the public API
        // without altering the internal format strings, but the `parse` method handles it.
        // The `format` method by default uses ISO8601.
        // To test "plain date" formatting, we'd need to access protected members or create a custom subclass,
        // which is disallowed. However, we can test the _cloneFormat helper implicitly.
        
        // We can test the default format output, which is ISO8601.
        Calendar cal = Calendar.getInstance(TimeZone.getTimeZone("UTC"));
        cal.set(2023, Calendar.OCTOBER, 27, 0, 0, 0);

        StringBuffer buffer = new StringBuffer();
        FieldPosition fp = new FieldPosition(0);
        StringBuffer formatted = sdf.format(dateToFormat, buffer, fp);
        
        // By default, format() uses _formatISO8601 which results in yyyy-MM-dd'T'HH:mm:ss.SSSZ
        // If the time is midnight and timezone is UTC, it will be "2023-10-27T00:00:00.000Z"
        assertEquals("Default format should be ISO8601", "2023-10-27T00:00:00.000Z", formatted.toString());
    }
