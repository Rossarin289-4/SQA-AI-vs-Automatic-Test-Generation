    @Test
    public void testCollectAndResolveSubtypesByTypeIdWithRegisteredSubtypes() throws Exception {
        StdSubtypeResolver resolver = new StdSubtypeResolver();
        // Note: TypeId resolution is more about mapping names to types.
        // Here we register types without explicit names.
        resolver.registerSubtypes(new NamedType(String.class), new NamedType(Integer.class));
        MapperConfig<?> config = createMockMapperConfig();
        JavaType baseType = createJavaType(Number.class); // String is not a subtype of Number

        // For ByTypeId, we need to see how the internal _collectAndResolveByTypeId handles it.
        // With nopInstance(), no type names from annotations.
        // registeredSubtypes are considered, but they need to be subtypes of baseType.
        // Integer is a subtype.
        Collection<NamedType> subtypes = resolver.collectAndResolveSubtypesByTypeId(config, null, baseType);
        
        // _combineNamedAndUnnamed will be called.
        // byName map: {"some_name": NamedType(SomeClass)}
        // typesHandled set: {Class1, Class2}
        // It will try to combine them.
        // With nopInstance(), no names are generated.
        // The registered subtypes are checked if they are assignable. Integer is.
        // The result should contain Integer. If it has no name, it will be added without one.
        assertEquals(1, subtypes.size());
        NamedType nt = subtypes.iterator().next();
        assertEquals(Integer.class, nt.getType());
        assertNull(nt.getName());
    }

    @Test
    public void testCollectAndResolveSubtypesByTypeIdWithBaseTypeAndRegistered() throws Exception {
        StdSubtypeResolver resolver = new StdSubtypeResolver();
        resolver.registerSubtypes(new NamedType(String.class));
        MapperConfig<?> config = createMockMapperConfig();
        JavaType baseType = createJavaType(String.class); // Base type is String

        Collection<NamedType> subtypes = resolver.collectAndResolveSubtypesByTypeId(config, null, baseType);
        
        // Base type itself is returned, along with registered subtypes that are assignable.
        // String.class is registered, and it's assignable to String.class.
        // `_collectAndResolveByTypeId` is called for the base type.
        // `_combineNamedAndUnnamed` will combine results.
        // The base type itself (String.class) will be added to `typesHandled`.
        // The registered subtype (String.class) will be processed. If it has a name, it goes to `byName`.
        // If the base type has no explicit name, it might end up in `typesHandled` but not `byName`.
        // `_combineNamedAndUnnamed` then adds unnamed types from `typesHandled`.
        
        // Given `AnnotationIntrospector.nopInstance()`, String.class has no type name.
        // So `_collectAndResolveByTypeId` adds `String.class` to `typesHandled`.
        // Registered `NamedType(String.class)` is processed. It has no name.
        // So it is NOT added to `byName`.
        // `_combineNamedAndUnnamed` is called.
        // It iterates `byName` (empty).
        // It iterates `typesHandled`: `String.class`.
        // `result.add(new NamedType(String.class))` is called.
        assertEquals(1, subtypes.size());
        NamedType nt = subtypes.iterator().next();
        assertEquals(String.class, nt.getType());
        assertNull(nt.getName());
    }
