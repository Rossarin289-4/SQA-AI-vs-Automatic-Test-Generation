    // Helper to create a default SerializerFactoryConfig
    private SerializerFactoryConfig createDefaultConfig() {
        return new SerializerFactoryConfig();
    }

    // Helper to create a default SerializerProvider
    private SerializerProvider createProvider(SerializerFactory factory) throws Exception {
        ObjectMapper mapper = new ObjectMapper();
        mapper.setSerializerFactory(factory);
        return mapper.getSerializerProviderInstance();
    }
    
    // Helper to create a default SerializationConfig
    private SerializationConfig createConfig() {
        ObjectMapper mapper = new ObjectMapper();
        return mapper.getSerializationConfig();
    }

    // Helper to create a default BeanDescription
    private BeanDescription createBeanDescription(JavaType type) {
        ObjectMapper mapper = new ObjectMapper();
        return mapper.getSerializationConfig().introspect(type);
    }
    
    // Helper to create a default AnnotatedClass
    private AnnotatedClass createAnnotatedClass(Class<?> cls) {
        return new AnnotatedClass(new BasicBeanDescription(createConfig(), TypeFactory.defaultInstance().constructType(cls), cls, Collections.emptyList(), Collections.emptyList(), Collections.<String, AnnotatedMethod>emptyMap(), Collections.<String, AnnotatedField>emptyMap(), Collections.<String, AnnotatedConstructor>emptyMap()));
    }

    // Helper to create a default AnnotatedMethod
    private AnnotatedMethod createAnnotatedMethod(Method method) {
        return new AnnotatedMethod(method, AnnotatedClass.construct(method.getDeclaringClass(), TypeFactory.defaultInstance(), null), null);
    }
