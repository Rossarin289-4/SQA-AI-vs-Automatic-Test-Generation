    private JsonParser mockJsonParser(Object... tokensAndValues) throws IOException {
        JsonParser jp = mock(JsonParser.class);
        Iterator<Object> iterator = Arrays.asList(tokensAndValues).iterator();

        when(jp.nextToken()).thenAnswer(invocation -> {
            if (!iterator.hasNext()) return null;
            Object next = iterator.next();
            if (next instanceof JsonToken) {
                return (JsonToken) next;
            } else if (next instanceof String) {
                return JsonToken.VALUE_STRING; // Default to VALUE_STRING for strings
            } else if (next == null) {
                return JsonToken.VALUE_NULL;
            }
            return null; // Should not happen with valid inputs
        });

        // For cases where we need to get the text value
        when(jp.getText()).thenAnswer(invocation -> {
            // This is a bit tricky. We need to find the last string value provided.
            // A more robust mock would track the current token.
            // For simplicity, we assume the last non-token value was the one to get.
            Object lastStringValue = null;
            for (Object item : tokensAndValues) {
                if (item instanceof String) {
                    lastStringValue = item;
                }
            }
            return (String) lastStringValue;
        });

        // Mocking isExpectedStartArrayToken
        when(jp.isExpectedStartArrayToken()).thenAnswer(invocation -> {
            Object firstToken = tokensAndValues[0];
            return firstToken == JsonToken.START_ARRAY;
        });

        return jp;
    }
