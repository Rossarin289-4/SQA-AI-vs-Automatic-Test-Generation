        Base64 base64 = new Base64(10); // Chunk size of 10
        byte[] input = "This is a longer test string for chunking.".getBytes();
        String encoded = new String(base64.encode(input));
        // Expecting line breaks after 10 characters (plus padding, if any)
        // "VGhpcyBpcyBhIGxvbmdlciB0ZXN0IHN0cmluZyBmb3IgY2h1bmtpbmcu"
        // VGlzaSBpcyBhIGxvbmdlciA0IGJ5dGVz
        // VGhpcyBpcyBhIGxvbmdlcg== (10 + 2 = 12 bytes encoded)
        // This is a longer (22 bytes) -> VGhpcyBpcyBhIGxvbmdlcg==
        // test string for chunking (23 bytes) -> dGVzdCBzdHJpbmcgZm9yIGNodW5raW5nLg==

        // VGlzaSBpcyBhIGxvbmdlciA0IGJ5dGVz (12 bytes, 16 chars)
        // dGVzdCBzdHJpbmcgZm9yIGNodW5raW5nLg== (23 bytes, 32 chars)

        // Let's re-calculate for "This is a longer test string for chunking." (45 bytes)
        // Encoded length should be ceil(45/3)*4 = 15*4 = 60 characters.
        // With lineLength = 10, this means chunks of 10 chars.
        // VGhpcyBpcyBhIGxvbmdlciB0ZXN0IHN0cmluZyBmb3IgY2h1bmtpbmcu
        // Expected:
        // VGhpcyBp
        // cyBhIGxv
        // 52cg dGVzdA
        // c3RyaW5n
        // IGZvcgogY2h1
        // bmtpbmcu
        // This is too complex to calculate manually here, rely on known good example for sanity.
        // A simpler example: "12345678901234567890" (20 bytes)
        // Expected: "MTIzNDU2Nzg5MA0KMTIzNDU2Nzg5MA==" (28 chars)
        // With lineLength=10:
        // MTIzNDU2Nzg5MA0K (12 chars)
        // MTIzNDU2Nzg5MA== (16 chars)
        byte[] simpleInput = "12345678901234567890".getBytes();
        Base64 base64WithLineLength = new Base64(10);
        String encodedSimple = new String(base64WithLineLength.encode(simpleInput));
        String expectedSimple = "MTIzNDU2Nzg5MA0KMTIzNDU2Nzg5MA==";
        assertEquals(expectedSimple, encodedSimple);
    }
