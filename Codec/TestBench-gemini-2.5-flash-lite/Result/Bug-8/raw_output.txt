    @Test
    public void testBase64EncodeChunked() {
        Base64 base64 = new Base64(MIME_CHUNK_SIZE, CHUNK_SEPARATOR);
        byte[] input = new byte[100];
        for (int i = 0; i < input.length; i++) {
            input[i] = (byte) i;
        }
        byte[] encoded = base64.encode(input);
        // Expected output based on manual calculation for 100 bytes.
        // The first chunk: "AAECAwQFBgcICQoLDA0ODxAREhMUFRYXGBkaGxwdHh8gISIjJCUmJygpKissLS4vMDEyMzQ1Njc4OTo7PD0+P0BBQkNERUZH"
        // Second chunk: "SElKS0xNTk9QUVJTVFVWV1hZWVpbXF1eX1+fgA8PT4/QEFCQ0RFRkdISUpLTE1OT1BR"
        // The last 3 bytes will be padded with '='.
        // For 100 bytes input, the output should be 136 bytes.
        // 100 bytes / 3 bytes per 4 chars = 33 groups of 3, plus 1 byte remaining.
        // 33 * 4 = 132 characters for the full groups.
        // The remaining 1 byte becomes 2 base64 characters, then padding. So 2 chars + 2 padding = 4.
        // Total characters: 132 + 4 = 136.
        // With chunk size 76, there will be one newline.
        // Chunk 1: 76 chars. Chunk 2: 136 - 76 = 60 chars.
        String expected = "AAECAwQFBgcICQoLDA0ODxAREhMUFRYXGBkaGxwdHh8gISIjJCUmJygpKissLS4vMDEyMzQ1Njc4OTo7PD0+P0BBQkNERUZHS\r\nSElKS0xNTk9QUVJTVFVWV1hZWVpbXF1eX1+fgA8PT4/QEFCQ0RFRkdISUpLTE1OT1BR";
        assertEquals("Chunked encoding incorrect", expected, StringUtils.newStringUtf8(encoded));
    }
