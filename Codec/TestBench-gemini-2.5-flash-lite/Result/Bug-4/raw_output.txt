    @Test
    public void testEncodeInteger_Negative() {
        // The encodeInteger method is designed for crypto, which typically uses positive integers.
        // However, toBytes will handle it, and we expect base64 encoding of its byte representation.
        BigInteger bigInt = new BigInteger("-12345");
        byte[] encoded = Base64.encodeInteger(bigInt);
        // -12345 in two's complement might be represented in a way that encodeInteger handles.
        // Let's trace what `toIntegerBytes` does for a negative number.
        // `bigInt.toByteArray()` for -12345 returns { -49, -10, -35 } (sign-magnitude like for negative).
        // `bitlen` for -12345 is 14.
        // `((14 + 7) >> 3) << 3` -> `(21 >> 3) << 3` -> `2 << 3` -> 16. So `bitlen` is 16.
        // `bigBytes` is {-49, -10, -35}
        // `(bigInt.bitLength() % 8) != 0` is true (14 % 8 != 0)
        // `((bigInt.bitLength() / 8) + 1) == (bitlen / 8)` -> `(14/8 + 1) == (16/8)` -> `(1 + 1) == 2` -> `2 == 2` is true.
        // So `toIntegerBytes` returns `bigBytes` directly: {-49, -10, -35}.
        // Base64 encoding of {-49, -10, -35} should be "1v2tKw==".
        assertArrayEquals("1v2tKw==".getBytes(), encoded);
    }
