public class FastDateFormatTest {

    public void testFormatObjectDateAndExistingBuffer() throws Exception {
        FastDateFormat f = FastDateFormat.getInstance("yyyy-MM-dd", TimeZone.getTimeZone("UTC"), Locale.US);
        StringBuffer b = new StringBuffer("x");
        StringBuffer result = f.format(new Date(0L), b, new FieldPosition(0));

        assertEquals("x1970-01-01", b.toString());

    public void testFormatObjectLong() throws Exception {
        FastDateFormat f = FastDateFormat.getInstance("yyyy-MM-dd", TimeZone.getTimeZone("UTC"), Locale.US);

    }

        FastDateFormat f = FastDateFormat.getInstance("HH:mm", Locale.US);
        Calendar c = new GregorianCalendar(TimeZone.getTimeZone("GMT+02:00"));
        c.setTimeInMillis(0L);
        assertEquals("02:00", f.format(c, new StringBuffer()).toString());
    }

    public void testFormatObjectRejectsUnknownInput() throws Exception {

        assertEquals(0, p.getErrorIndex());

    public void testParseObjectEmptySourceAlsoResetsPosition() throws Exception {

    public void testForcedTimeZoneAccessors() throws Exception {

        assertTrue(f.getTimeZoneOverridesCalendar());

    }

    @Test
    public void testDefaultFactoryProducesUsableFormatter() throws Exception {
        FastDateFormat f = FastDateFormat.getInstance();
        assertNotNull(f.getPattern());
        assertEquals(f.getPattern(), f.getPattern());
    }
