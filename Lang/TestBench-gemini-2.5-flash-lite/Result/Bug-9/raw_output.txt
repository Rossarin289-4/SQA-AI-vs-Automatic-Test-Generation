    @Test
    public void testParseYearAbbreviatedBoundary() throws Exception {
        FastDateParser parser = new FastDateParser("yy", TimeZone.getDefault(), Locale.US);
        // Test boundary for adjustYear: 20 years after thisYear and 80 years before.
        // If thisYear = 2000, then 20 years after is 2020. So 20 should be 2020.
        // If thisYear = 2000, then 80 years before is 1920. So 19 should be 1919.
        // If thisYear = 2000, then 80 years before is 1920. So 99 should be 1999.
        // The boundary for returning trial-100 is when trial < thisYear + 20.
        // if thisYear=2000, 99 -> trial = 1999. 1999 < 2000+20, so returns 1999.
        // if thisYear=2000, 20 -> trial = 2020. 2020 >= 2000+20, so returns 2020-100 = 1920. This seems wrong.
        // Let's re-evaluate adjustYear logic:
        // int trial= twoDigitYear + thisYear - thisYear%100;
        // If thisYear = 2023: thisYear%100 = 23. trial = twoDigitYear + 2023 - 23 = twoDigitYear + 2000.
        // For input "23": trial = 23 + 2000 = 2023. 2023 < 2023 + 20 is true. Returns 2023.
        // For input "20": trial = 20 + 2000 = 2020. 2020 < 2023 + 20 is true. Returns 2020.
        // For input "21": trial = 21 + 2000 = 2021. 2021 < 2023 + 20 is true. Returns 2021.
        // For input "00": trial = 0 + 2000 = 2000. 2000 < 2023 + 20 is true. Returns 2000.
        // For input "99": trial = 99 + 2000 = 2099. 2099 < 2023 + 20 is true. Returns 2099.
        // The logic seems to be: years in the range [thisYear+20, thisYear+100) get mapped to [thisYear-80, thisYear+20).
        // This means if the calculated year is *not* in the range [thisYear+20, thisYear+100), we subtract 100.
        // int trial= twoDigitYear + thisYear - thisYear%100; -> this assumes twoDigitYear is the last two digits.
        // Let's assume thisYear is 2023. thisYear%100 is 23.
        // The base year is thisYear - (thisYear%100) = 2000.
        // So trial = twoDigitYear + 2000.
        // if twoDigitYear is 23, trial = 2023. 2023 < 2000 + 100 (thisYear+20 relative to base 2000).
        // The logic is: if trial < thisYear + 20 (so < 2043), return trial. Otherwise return trial - 100.
        // If twoDigitYear = 23, trial = 2023. 2023 < 2043. Returns 2023.
        // If twoDigitYear = 00, trial = 2000. 2000 < 2043. Returns 2000.
        // If twoDigitYear = 43, trial = 2043. 2043 is not < 2043. Returns 2043 - 100 = 1943. This is correct for years between 1943 and 2042.
        // So "23" maps to 2023 if thisYear is 2023.
        // Let's test "43" and "44" assuming thisYear is 2023.
        Date date43 = parser.parse("43"); // Expected: 1943
        Calendar cal43 = Calendar.getInstance(TimeZone.getDefault(), Locale.US);
        cal43.setTime(date43);
        assertEquals(1943, cal43.get(Calendar.YEAR));

        Date date44 = parser.parse("44"); // Expected: 2044
        Calendar cal44 = Calendar.getInstance(TimeZone.getDefault(), Locale.US);
        cal44.setTime(date44);
        assertEquals(2044, cal44.get(Calendar.YEAR));

        Date date00 = parser.parse("00"); // Expected: 2000
        Calendar cal00 = Calendar.getInstance(TimeZone.getDefault(), Locale.US);
        cal00.setTime(date00);
        assertEquals(2000, cal00.get(Calendar.YEAR));
    }

    // Test for parsing a number that exceeds the typical range for a field (e.g., month)
    @Test
    public void testParseLargeNumberForMonth() throws Exception {
        FastDateParser parser = new FastDateParser("M", TimeZone.getDefault(), Locale.US);
        // A number larger than 12 for month
        String largeMonth = "15";
        try {
            parser.parse(largeMonth);
            fail("Expected ParseException for out-of-range month number");
        } catch (IllegalArgumentException e) {
            // The NumberStrategy.setCalendar calls modify and then set.
            // If Integer.parseInt("15") succeeds, it will be set, and later Calendar might handle it or throw.
            // The current implementation in FastDateParser doesn't seem to validate the range of parsed numbers.
            // Let's check if it throws an exception. If not, check the resulting Calendar field.
            // Based on current code, it seems to allow it and Calendar will handle it (e.g., May of next year)
            // Let's re-verify with the reference source.
            // The setCalendar method in NumberStrategy just calls cal.set(field, modify(Integer.parseInt(value)));
            // Calendar.set for month 15 will roll over to the next year.
            // So, if we parse "15", it will result in Calendar.MAY of the next year.
            Date date = parser.parse(largeMonth);
            Calendar cal = Calendar.getInstance(TimeZone.getDefault(), Locale.US);
            cal.setTime(date);
            assertEquals(Calendar.MAY, cal.get(Calendar.MONTH)); // 15 = 12 + 3, so May (index 4).
        } catch (ParseException e) {
            // If it throws ParseException instead of IllegalArgumentException
             fail("Expected IllegalArgumentException, but got ParseException: " + e.getMessage());
        }
    }
