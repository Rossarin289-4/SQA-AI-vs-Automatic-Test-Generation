    @Test
    public void testParseAbbreviatedYear() throws Exception {
        // Pattern: "yy"
        FastDateParser parser = new FastDateParser("yy", TimeZone.getDefault(), Locale.US);
        // Test year 00
        Date date = parser.parse("00");
        Calendar cal = Calendar.getInstance(TimeZone.getDefault(), Locale.US);
        cal.setTime(date);
        // thisYear will be the current year. adjustYear logic for "00" will add 100 to thisYear's century.
        // e.g. if thisYear is 2023, 00 becomes 2100. If thisYear is 2020, 00 becomes 2000.
        // We are testing the logic of adjustYear. Let's assume thisYear is 2023 for now.
        int expectedYear = (2023 / 100) * 100; // current century
        if (0 < 23 + 20) { // 0 < 23 + 20, so trial = 0 + 2023 - 2023%100 = 2023.  2023 < 2023 + 20.
            expectedYear = 0 + 2023 - (2023 % 100); // trial = 2000
            if (expectedYear < 2023 + 20) {
                // expectedYear is 2000. 2000 < 2043 is true.
                // but it should be 0 + thisYear - thisYear%100, so 2000
            } else {
                expectedYear -= 100;
            }
        } else {
            expectedYear -= 100;
        }

        // The logic in adjustYear is:
        // int trial= twoDigitYear + thisYear - thisYear%100;
        // if(trial < thisYear+20) { return trial; } else { return trial-100; }
        // If twoDigitYear is 00 and thisYear is 2023:
        // trial = 0 + 2023 - 2023 % 100 = 0 + 2023 - 23 = 2000.
        // if (2000 < 2023 + 20) which is true, return 2000.
        assertEquals(2000, cal.get(Calendar.YEAR));

        // Test year 99
        date = parser.parse("99");
        cal.setTime(date);
        // If twoDigitYear is 99 and thisYear is 2023:
        // trial = 99 + 2023 - 2023 % 100 = 99 + 2023 - 23 = 2099.
        // if (2099 < 2023 + 20) which is false, return 2099 - 100 = 1999.
        assertEquals(1999, cal.get(Calendar.YEAR));
    }

        // Test edge case based on thisYear + 20 for adjustYear
        // Assume thisYear is 2023. thisYear+20 is 2043.
        // Test value that falls into the 'else' branch of adjustYear
        // A year that results in a trial value >= thisYear + 20
        // If we parse "43", trial = 43 + 2023 - 23 = 2043. 2043 is not < 2043+20. returns 2043-100 = 1943
