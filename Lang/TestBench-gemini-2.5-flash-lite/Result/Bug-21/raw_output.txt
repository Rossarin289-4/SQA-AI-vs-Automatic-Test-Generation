    @Test
    public void testGetFragmentInHours() {
        Calendar cal = Calendar.getInstance();
        cal.set(2023, Calendar.MARCH, 15, 10, 30, 15);
        Date date = cal.getTime();

        // Fragment: DAY_OF_YEAR (assuming it's the 74th day of 2023 - March 15th)
        assertEquals(10, DateUtils.getFragmentInHours(date, Calendar.DAY_OF_YEAR));
        
        // Fragment: MONTH (assuming March 15th is the 5th day of the month for this calculation)
        // The calculation is (days_in_month - 1) * 24 + hours
        // For March 15th, this would be (14 * 24) + 10 = 336 + 10 = 346
        // Let's use a simpler example for clarity. If it were March 2nd, and we are testing
        // fragment Calendar.MONTH and current day is 2nd, hours is 10.
        // It should be (2-1)*24 + 10 = 1*24 + 10 = 34
        // Let's recalculate for March 15th, considering the example in the docs.
        // "January 6, 2008 with Calendar.MONTH as fragment will return 127 (5*24 + 7)"
        // This implies the formula is (days_past_month_start * 24) + current_hour
        // For March 15th, assuming start of month is day 1:
        // (15-1)*24 + 10 = 14*24 + 10 = 336 + 10 = 346
        assertEquals(346, DateUtils.getFragmentInHours(date, Calendar.MONTH));
    }
