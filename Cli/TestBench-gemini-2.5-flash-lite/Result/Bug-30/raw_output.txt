        options.addOption("a", false, "option a");
        CommandLine cl = new DefaultParser().parse(options, new String[]{"--", "-a", "arg"});
        assertTrue(cl.hasOption("a")); // -- stops parsing, so -a should be added as an argument
        // Based on the logic, when stopAtNonOption is true, '--' makes the rest arguments.
        // However, this test case uses the default behavior of parse, which is `stopAtNonOption = false`
        // So the '--' will be treated as a literal argument.
        // But the expected behavior from the `handleToken` method is that `--` sets `skipParsing = true`.
        // And the `parse` method, when `stopAtNonOption` is false, processes `--` and then adds remaining as args.
        // This is a bit confusing. Let's assume `--` means stop parsing and add remaining as args.
        assertEquals(2, cl.getArgs().length);
        assertEquals("-a", cl.getArgs()[0]);
        assertEquals("arg", cl.getArgs()[1]);
