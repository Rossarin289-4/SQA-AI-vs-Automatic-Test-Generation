    public void testFlattenMultipleOptionsWithArgumentsEmbedded() throws Exception {
        Options options = new Options();
        options.addOption("a", true, "desc a");
        options.addOption("b", true, "desc b");

        PosixParser parser = new PosixParser();
        String[] args = {"-aaval", "-bbval"}; // This is incorrect parsing for combined options with arguments
        String[] flattened = parser.flatten(options, args, false);
        // The current implementation of burstToken does not handle arguments embedded in combined options.
        // It will try to burst -a, then -a for the second char, then it will see 'v' and try to burst it.
        // The logic in burstToken is: if options.hasOption(ch) and currentOption.hasArg() and token.length() != (i + 1),
        // then it adds token.substring(i+1).
        // For "-aaval", 'a' is an option, it has an arg, token.length() is 5, i=0, i+1=1. 5 != 1.
        // So, it adds token.substring(1) which is "aval". This means it consumes the rest of the token as an argument to -a.
        assertEquals("-a", flattened[0]);
        assertEquals("aval", flattened[1]);
        // "-bbval" will be treated as a new token. 'b' is an option, it has an arg. token.length() is 5, i=0, i+1=1. 5 != 1.
        // So, it adds token.substring(1) which is "bval".
        assertEquals("-b", flattened[2]);
        assertEquals("bval", flattened[3]);
    }
