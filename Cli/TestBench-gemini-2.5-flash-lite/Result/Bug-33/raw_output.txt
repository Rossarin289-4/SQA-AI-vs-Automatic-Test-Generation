    public void testFindWrapPosWithTab() throws Exception {
        HelpFormatter formatter = new HelpFormatter();
        String text = "word1\tword2";
        int width = 6;
        int pos = formatter.findWrapPos(text, width, 0);
        // Tab is at index 5. width is 6. pos should be 6 (pos+1 for tab).
        // Since pos+1 == width, it should return -1 as it fits.
        assertEquals(-1, pos);
        
        width = 5;
        pos = formatter.findWrapPos(text, width, 0);
        // Tab is at index 5. width is 5. pos should be 6.
        // Since pos > startPos, it should return pos, which is 5.
        assertEquals(5, pos);
    }

    public void testOptionComparator() throws Exception {
        HelpFormatter formatter = new HelpFormatter();
        Comparator comparator = formatter.getOptionComparator();

        Option opt1 = new Option("a", "alpha", false, "desc1");
        Option opt2 = new Option("b", "beta", false, "desc2");
        Option opt3 = new Option("A", "Alpha", false, "desc3"); // case-insensitive check

        assertTrue(comparator.compare(opt1, opt2) < 0); // a < b
        assertTrue(comparator.compare(opt2, opt1) > 0); // b > a
        assertEquals(0, comparator.compare(opt1, opt3)); // a == A (case-insensitive)
    }
