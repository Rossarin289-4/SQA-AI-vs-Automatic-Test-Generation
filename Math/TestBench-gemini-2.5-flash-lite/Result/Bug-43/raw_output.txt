        assertEquals(Math.sqrt(2.5), stats.getStandardDeviation(), 1e-9);

        assertEquals(2.0, stats.getPopulationVariance(), 0);

    /**
     * Tests population variance with multiple values.
     *
     * @throws Exception
     */
    @Test
    public void testPopulationVariance() throws Exception {
        SummaryStatistics stats = new SummaryStatistics();
        stats.addValue(1.0);
        stats.addValue(2.0);
        stats.addValue(3.0);
        stats.addValue(4.0);
        stats.addValue(5.0);
        // Expected population variance: sum((x_i - mean)^2) / n
        // Mean = 3.0
        // Deviations: -2, -1, 0, 1, 2
        // Squared Deviations: 4, 1, 0, 1, 4
        // Sum of Squared Deviations = 10
        // Population Variance = 10 / 5 = 2.0
        assertEquals(2.0, stats.getPopulationVariance(), 0);
    }

     * @throws Exception
     */
    @Test
    public void testBoundaryValues() throws Exception {
        SummaryStatistics stats = new SummaryStatistics();
        stats.addValue(Double.MAX_VALUE);
        stats.addValue(Double.MIN_VALUE); // Smallest positive normal value

        assertEquals(2, stats.getN());
        assertEquals(Double.MAX_VALUE, stats.getMax(), 0);
        assertEquals(Double.MIN_VALUE, stats.getMin(), 0);
        assertEquals(Double.MAX_VALUE, stats.getSum(), 0); // Sum is dominated by MAX_VALUE
        // Sum of squares will be dominated by MAX_VALUE squared which overflows to infinity
        assertEquals(Double.POSITIVE_INFINITY, stats.getSumsq(), 0);
        // Mean will be dominated by MAX_VALUE
        assertEquals(Double.MAX_VALUE, stats.getMean(), 0);
        // Variance with such disparate values should be NaN or very large.
        // Given MAX_VALUE^2 overflows, it's likely NaN or Infinity.
        // Let's check based on the implementation behavior.
        // Variance calculation involves sum of squares.
        // The sumsq is already infinity.
        // The calculation for variance involves (sumsq - n*mean^2) / (n-1)
        // (Infinity - 2 * MAX_VALUE^2) / 1 -> Infinity
        assertEquals(Double.POSITIVE_INFINITY, stats.getVariance(), 0);
        assertEquals(Double.POSITIVE_INFINITY, stats.getStandardDeviation(), 0);
        // Sum of logs will be dominated by log(MAX_VALUE)
        assertEquals(Math.log(Double.MAX_VALUE), stats.getSumOfLogs(), 0);
        // Geometric mean will be dominated by MAX_VALUE
        assertEquals(Double.MAX_VALUE, stats.getGeometricMean(), 1e-9);
        // Second moment calculation: sum of squared deviations
        // Mean = MAX_VALUE
        // Deviation of MAX_VALUE = 0
        // Deviation of MIN_VALUE = MIN_VALUE - MAX_VALUE (approximately -MAX_VALUE)
        // Squared Deviation of MIN_VALUE = (MIN_VALUE - MAX_VALUE)^2 which is Infinity
        assertEquals(Double.POSITIVE_INFINITY, stats.getSecondMoment(), 0);
        assertEquals(Double.POSITIVE_INFINITY, stats.getPopulationVariance(), 0);
    }
