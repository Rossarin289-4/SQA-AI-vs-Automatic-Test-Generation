    @Test
    public void testGetSolutionWithEqualityConstraint() {
        LinearObjectiveFunction f = new LinearObjectiveFunction(new double[]{10, 20}, 0);
        Collection<LinearConstraint> constraints = new ArrayList<>();
        constraints.add(new LinearConstraint(new double[]{1, 1}, Relationship.EQ, 5)); // Equality requires artificial
        SimplexTableau tableau = new SimplexTableau(f, constraints, GoalType.MAXIMIZE, true, 1e-6);
        // After discarding artificial variables, a basic solution can be found.
        tableau.discardArtificialVariables();
        // Expected solution: x1+x2=5, Z = 10*x1 + 20*x2. If x1=0, x2=5, Z=100. If x1=5, x2=0, Z=50.
        // The simplex method would pick one. The getSolution method might pick one based on tableau state.
        // Let's assume it finds x1=0, x2=5 for maximum objective.
        RealPointValuePair solution = tableau.getSolution();
        assertNotNull(solution);
        // The exact solution point depends on the simplex algorithm's path.
        // A simple check is to verify the point satisfies the constraints.
        // For x1=0, x2=5: 0+5=5 (eq constraint satisfied).
        // Value should be 10*0 + 20*5 = 100.
        // However, the getSolution() implementation might not reflect actual simplex iterations.
        // It relies on finding basic variables.
        // Without artificial variables, the tableau might look like:
        // Z | x1 | x2 | RHS
        // 10  20   0    0  (objective)
        // 1   1    0    5  (constraint)
        // Pivot on col x2 (20), row 1.
        // new row 0 = row 0 - 20 * row 1 => 10-20= -10, 20-20=0, 0, 0-20*5 = -100
        // Z | x1 | x2 | RHS
        // -10 0    0    -100
        // 1   1    0    5
        // Basic var is x2 in row 1. x1 is non-basic (0).
        // Solution: x1=0, x2=5. Value = 100.
        // The current getSolution logic for equality constraints might be simplified and not yield optimal.
        // For this specific constraint and maximize, the solution for getSolution() is often one where variables not in tableau are 0.
        // In the above post-discard tableau, x2 is basic in row 1.
        // x1 is not basic. getBasicRowForSolution(0) would return null.
        // coefficients[0] (for x1) = (basicRow == null ? 0 : getEntry(basicRow, getRhsOffset())) = 0.
        // coefficients[1] (for x2) = getEntry(1, getRhsOffset()) = 5.
        // Value = f.getValue({0, 5}) = 10*0 + 20*5 = 100.
        assertArrayEquals(new double[]{0.0, 5.0}, solution.getPoint(), 1e-9);
        assertEquals(100.0, solution.getValue(), 1e-9);
    }

    @Test
    public void testGetNegativeDecisionVariableOffsetWhenRestricted() {
        LinearObjectiveFunction f = new LinearObjectiveFunction(new double[]{10, 20}, 0);
        Collection<LinearConstraint> constraints = new ArrayList<>();
        constraints.add(new LinearConstraint(new double[]{1, 1}, Relationship.LEQ, 5));
        SimplexTableau tableau = new SimplexTableau(f, constraints, GoalType.MAXIMIZE, true, 1e-6); // Restricted to non-negative
        // The method getNegativeDecisionVariableOffset should still return a valid offset,
        // but it won't be used in calculations if restrictToNonNegative is true.
        // The logic might still calculate it based on numDecisionVariables, which would be 2.
        // The `getOriginalNumDecisionVariables` returns numDecisionVariables (2) if restricted.
        // So offset = numObjectiveFunctions (1) + numDecisionVariables (2) = 3.
        // However, this method is specific to the "x-" variable, which only exists when not restricted.
        // The implementation `getNumObjectiveFunctions() + getOriginalNumDecisionVariables()`
        // when restrictToNonNegative is true, `getOriginalNumDecisionVariables()` returns `numDecisionVariables`.
        // So for restrictToNonNegative=true, numDecisionVariables = originalNumDecisionVariables.
        // numObjectiveFunctions is 1. Offset = 1 + 2 = 3.
        // This seems to be a slight confusion in naming or intended usage if this offset is expected to be valid/meaningful when not restricted.
        // Let's check the actual implementation detail.
        // The offset is `getNumObjectiveFunctions() + getOriginalNumDecisionVariables()`.
        // If `restrictToNonNegative` is true, `getOriginalNumDecisionVariables()` returns `numDecisionVariables`.
        // `numDecisionVariables` is `getNumVariables() + (restrictToNonNegative ? 0 : 1)`.
        // So if `restrictToNonNegative` is true, `numDecisionVariables` = `getNumVariables()`.
        // Offset = `getNumObjectiveFunctions()` + `numDecisionVariables`.
        // In this case: 1 + 2 = 3.
        assertEquals(3, tableau.getNegativeDecisionVariableOffset());
    }
