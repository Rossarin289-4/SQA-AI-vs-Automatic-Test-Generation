    @Test
    public void testTaylor() {
        DSCompiler compiler = DSCompiler.getCompiler(1, 2);
        double[] ds = {2.0, 3.0, 1.0}; // f(x) = 2 + 3x + x^2
        double delta = 1.0; // taylor expansion at x + delta = 1 + 1 = 2
        double result = compiler.taylor(ds, 0, delta);
        // Taylor expansion: f(x0) + f'(x0)*delta + f''(x0)/2! * delta^2
        // Here x0=1, delta=1. f(1)=2+3+1=6. f'(1)=3+2*1=5. f''(1)=2.
        // Result = 6 + 5*1 + 2/2 * 1^2 = 6 + 5 + 1 = 12.
        // Direct evaluation: f(2) = 2 + 3*2 + 2^2 = 2 + 6 + 4 = 12.
        assertEquals(12.0, result, 1e-9);
    }

    @Test
    public void testTaylorMultiParam() {
        DSCompiler compiler = DSCompiler.getCompiler(2, 1); // f, df/dx, df/dy
        double[] ds = {5.0, 2.0, 3.0}; // f(x,y) = 5 + 2x + 3y
        double deltaX = 1.0;
        double deltaY = 1.0;
        double result = compiler.taylor(ds, 0, deltaX, deltaY);
        // f(x0+dx, y0+dy) = f(x0,y0) + df/dx(x0,y0)*dx + df/dy(x0,y0)*dy
        // Here x0=0, y0=0, dx=1, dy=1.
        // Result = 5 + 2*1 + 3*1 = 10.
        assertEquals(10.0, result, 1e-9);
    }
