    public void testComparatorConstructor() throws Exception {

            @Override
            public int compare(Comparable<?> o1, Comparable<?>) {
                // Simplified reverse comparison for demonstration
                if (o1 instanceof Integer && o2 instanceof Integer) {
                    return ((Integer) o2).compareTo((Integer) o1);
                }
                return 0;
            }
        };
        Frequency freq = new Frequency(reverseComparator);

        freq.addValue(2);
        freq.addValue(3);
        freq.addValue(2);
        
        Iterator<Comparable<?>> it = freq.valuesIterator();
        assertEquals(3, it.next()); // Should iterate in reverse order

        assertEquals(1, it.next());

        // Due to the custom comparator, getCumFreq behavior might be tricky.
