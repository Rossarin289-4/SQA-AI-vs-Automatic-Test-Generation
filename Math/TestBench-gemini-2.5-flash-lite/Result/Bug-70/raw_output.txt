        // If min=0, f(min)=-2.5. If m=2.5, f(m)=0. Then fm*fmin will be 0, so it goes to else branch.
        // Let's construct a case where fm is negative and fmin is positive.
        f = new UnivariateRealFunction() {
            public double value(double x) {
                return -(x - 2.5); // root at 2.5
