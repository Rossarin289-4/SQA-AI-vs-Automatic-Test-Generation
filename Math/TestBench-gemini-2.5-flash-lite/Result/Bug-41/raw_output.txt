        variance.increment(5.0);

        // Expected unbiased variance: sum((x_i - mean)^2) / (n - 1)
        // Mean = 3
        // Deviations: -2, -1, 0, 1, 2
        // Squared deviations: 4, 1, 0, 1, 4. Sum = 10

    public void testEvaluateWithWeightsMultipleValuesWeighted() throws Exception {

        double[] weights = {1.0, 2.0, 3.0}; // Weights are 1, 2, 3
        // Weighted Mean = (1*1 + 2*2 + 3*3) / (1+2+3) = (1 + 4 + 9) / 6 = 14 / 6 = 7/3

        // Sum of weights = 1 + 2 + 3 = 6
        // Variance (unbiased) = (10/3) / (6 - 1) = (10/3) / 5 = 10/15 = 2/3
        assertEquals(2.0 / 3.0, variance.evaluate(values, weights), 1e-9);
