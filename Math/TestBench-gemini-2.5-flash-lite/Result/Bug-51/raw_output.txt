     * Test with AllowedSolution.BELOW_SIDE.
     */
    @Test
    public void testSolveBelowSide() throws Exception {
        // f(x) = x^2 - 4, roots at -2 and 2.
        UnivariateRealFunction f = new UnivariateRealFunction() {
            @Override
            public double value(double x) {
                return x * x - 4.0;
            }
        };
        // Using RegulaFalsiSolver
        RegulaFalsiSolver solver = new RegulaFalsiSolver();
        // Interval [-5, 5]. Allowed solutions to be below side (f(x) <= 0).
        // Both -2 and 2 are valid. The solver might pick either depending on initial guesses.
        // Let's choose an interval that favors one.
        double root = solver.solve(100, f, -5.0, 0.0, AllowedSolution.BELOW_SIDE);
        // In [-5, 0], root is -2. f(-5)=21, f(0)=-4. Root is expected to be negative.
        // If the solver returns a positive value, its function value would be negative, satisfying BELOW_SIDE.
        // But the root itself must be within [-5, 0].
        assertEquals(-2.0, root, 1e-6);

        root = solver.solve(100, f, 0.0, 5.0, AllowedSolution.BELOW_SIDE);
        // In [0, 5], root is 2. f(0)=-4, f(5)=21. Root is expected to be positive.
        // If the solver returns a negative value, its function value would be positive.
        // However, the problem is to find a root within [0,5].
        // The condition BELOW_SIDE checks if f(root) <= 0.
        // For x^2-4 in [0,5], only x=2 satisfies f(x)=0.
        // If the solver returns something close to 2, f(x) will be close to 0.
        assertEquals(2.0, root, 1e-6);
    }

        // Interval [-10.0, -0.1]. Root is -10.

        // f(-0.1) = 0.01 - 100 = -99.99
        root = solver.solve(100, f, -10.0, -0.1);
