        SubLine subLine1 = new SubLine(line1, new Interval(-2, 2));
        Line line2 = new Line(new Vector2D(0, 0), new Vector2D(0, 1)); // Intersects at (0,0)
        SplitSubHyperplane<Euclidean2D> split = subLine1.split(line2);

        assertNotNull(split.getPlus());
        assertNotNull(split.getMinus());

        // Check subLine on the plus side of line2
        Line plusLine = (Line) split.getPlus().getHyperplane();
        IntervalsSet plusRegion = (IntervalsSet) split.getPlus().getRemainingRegion();
        assertEquals(1, plusRegion.asList().size());
        assertEquals(0.0, plusRegion.asList().get(0).getInf(), 1e-9);
        assertEquals(2.0, plusRegion.asList().get(0).getSup(), 1e-9);
        assertTrue(plusLine.isParallelTo(line1));
        assertTrue(plusLine.sameOrientationAs(line1));

        // Check subLine on the minus side of line2
        Line minusLine = (Line) split.getMinus().getHyperplane();
        IntervalsSet minusRegion = (IntervalsSet) split.getMinus().getRemainingRegion();
        assertEquals(1, minusRegion.asList().size());
        assertEquals(-2.0, minusRegion.asList().get(0).getInf(), 1e-9);
        assertEquals(0.0, minusRegion.asList().get(0).getSup(), 1e-9);
