    @Test
    public void testCovariancesSimple() throws Exception {
        // Mocking a simple scenario where J^T J is invertible.
        // Let's assume a 2x2 jacobian for simplicity.
        // jacobian:
        // | 1  2 |
        // | 3  4 |
        // jacobian = {1.0, 2.0, 3.0, 4.0} (flattened, row-major for updateJacobian, but loop is column-major based on 'cols')
        // updateJacobian fills jacobian from measurements and parameters.
        // For simplicity, let's assume we can mock the jacobian directly for testing getCovariances.

        // A simple case for J^T J
        // J = [ 1 ]
        //     [ 2 ]
        // J^T = [1 2]
        // J^T J = [1*1 + 2*2] = [5]
        // J^T J inverse = [1/5]

        // Mocking is difficult due to dependencies. Let's create a concrete scenario.
        // We need a problem with measurements and unbound parameters.

        // Dummy setup for a simple estimation problem
        EstimatedParameter p1 = new DummyEstimatedParameter("p1", 0.0);

        WeightedMeasurement wm1 = new DummyWeightedMeasurement(1.0, 1.0, 1.0) {
            @Override
            public double getPartial(EstimatedParameter parameter) {
                return (parameter == p1) ? 1.0 : 0.0; // partial w.r.t p1 is 1.0
            }
            @Override
            public double getTheoreticalValue() { return 0.0; } // Not used by getCovariances
        };
        WeightedMeasurement wm2 = new DummyWeightedMeasurement(1.0, 2.0, 2.0) {
            @Override
            public double getPartial(EstimatedParameter parameter) {
                return (parameter == p1) ? 2.0 : 0.0; // partial w.r.t p1 is 2.0
            }
            @Override
            public double getTheoreticalValue() { return 0.0; } // Not used by getCovariances
        };
        WeightedMeasurement[] measurements = {wm1, wm2};
        EstimationProblem problem = new DummyEstimationProblem(measurements, unboundParams);

        AbstractEstimator estimator = new AbstractEstimator() {
            // Implement abstract method estimate - not needed for getCovariances test
            @Override
            public void estimate(EstimationProblem problem) throws EstimationException {}

            // Need to make fields accessible for testing setup if not using public setters
            // For this test, we will use the actual initialization logic by calling a protected method.
            // Initialize is protected, so we'll need to call it from a concrete subclass or override it.
            // The actual jacobian update is in updateJacobian().
        };

        // We need to set up the state that updateJacobian uses.
        // This means calling initializeEstimate and setting measurements and parameters.
        estimator.initializeEstimate(problem);

        // Now call updateJacobian which populates the jacobian field.
        estimator.updateJacobian(); // This will increment jacobianEvaluations.

        // The jacobian will be:
        // wm1: weight=1, residual=1, partial(p1)=1.0 => sqrt(1) * 1.0 = 1.0

        // jacobian = [1.0, 2.0]

        // J = [1]
        //     [2]
        // J^T J = [1 2] * [1] = [1*1 + 2*2] = [5]
        //                 [2]
        // Invertible J^T J: [1/5]

        double[][] covariances = estimator.getCovariances(problem);
        assertEquals(1, covariances.length);
        assertEquals(1, covariances[0].length);
        assertEquals(0.2, covariances[0][0], 1e-9); // 1/5

    @Test
    public void testUpdateResidualsAndCost() throws Exception {
        AbstractEstimator estimator = new LevenbergMarquardtEstimator();
        estimator.setMaxCostEval(5);

        EstimatedParameter p1 = new DummyEstimatedParameter("p1", 0.0);
        WeightedMeasurement wm1 = new DummyWeightedMeasurement(1.0, 1.0, 1.0);
        WeightedMeasurement wm2 = new DummyWeightedMeasurement(2.0, 2.0, 2.0);
        EstimationProblem problem = new DummyEstimationProblem(
                new WeightedMeasurement[]{wm1, wm2},
                new EstimatedParameter[]{p1}
        );
        estimator.initializeEstimate(problem);

        estimator.updateResidualsAndCost();

        // Expected cost calculation:
        // cost = sqrt(wm1.weight * residual1^2 + wm2.weight * residual2^2)
        // cost = sqrt(1.0 * 1.0^2 + 2.0 * 2.0^2) = sqrt(1.0 + 8.0) = sqrt(9.0) = 3.0
        assertEquals(3.0, estimator.cost, 1e-9);

        // Residuals array:
        // residuals[0] = sqrt(wm1.weight) * residual1 = sqrt(1.0) * 1.0 = 1.0
        // residuals[1] = sqrt(wm2.weight) * residual2 = sqrt(2.0) * 2.0
        assertEquals(1.0, estimator.residuals[0], 1e-9);
        assertEquals(Math.sqrt(2.0) * 2.0, estimator.residuals[1], 1e-9);

        // Cost evaluations counter
        assertEquals(1, estimator.getCostEvaluations());
    }

    @Test
    public void testUpdateJacobian() throws Exception {
        AbstractEstimator estimator = new LevenbergMarquardtEstimator();
        estimator.setMaxCostEval(10); // Needed to initialize estimate

        EstimatedParameter p1 = new DummyEstimatedParameter("p1", 0.0);
        EstimatedParameter p2 = new DummyEstimatedParameter("p2", 0.0);
        EstimatedParameter[] unboundParams = {p1, p2};

        WeightedMeasurement wm1 = new DummyWeightedMeasurement(1.0, 1.0, 1.0) {
            @Override
            public double getPartial(EstimatedParameter parameter) {
                if (parameter == p1) return 0.5;
                if (parameter == p2) return 1.0;
                return 0.0;
            }
            @Override
            public double getTheoreticalValue() { return 0.0; }
        };
        WeightedMeasurement wm2 = new DummyWeightedMeasurement(4.0, 2.0, 2.0) {
            @Override
            public double getPartial(EstimatedParameter parameter) {
                if (parameter == p1) return 1.5;
                if (parameter == p2) return 2.0;
                return 0.0;
            }
            @Override
            public double getTheoreticalValue() { return 0.0; }
        };
        WeightedMeasurement[] measurements = {wm1, wm2};
        EstimationProblem problem = new DummyEstimationProblem(measurements, unboundParams);

        estimator.initializeEstimate(problem);
        estimator.updateJacobian();

        // Jacobian rows = measurements.length = 2
        // Jacobian cols = parameters.length = 2
        // jacobian size = 2 * 2 = 4

        // Factor for wm1: -sqrt(1.0) = -1.0
        // Factor for wm2: -sqrt(4.0) = -2.0

        // jacobian[index++] = factor * wm.getPartial(parameters[j]);

        // For wm1 (i=0):
        // j=0 (p1): -1.0 * 0.5 = -0.5
        // j=1 (p2): -1.0 * 1.0 = -1.0
        // jacobian[0] = -0.5, jacobian[1] = -1.0

        // For wm2 (i=1):
        // j=0 (p1): -2.0 * 1.5 = -3.0
        // j=1 (p2): -2.0 * 2.0 = -4.0
        // jacobian[2] = -3.0, jacobian[3] = -4.0

        assertEquals(-0.5, estimator.jacobian[0], 1e-9);
        assertEquals(-1.0, estimator.jacobian[1], 1e-9);
        assertEquals(-3.0, estimator.jacobian[2], 1e-9);
        assertEquals(-4.0, estimator.jacobian[3], 1e-9);

        assertEquals(2, estimator.getJacobianEvaluations());
    }

    // Dummy implementations for testing purposes
    private static class DummyWeightedMeasurement extends WeightedMeasurement {
        private static final long serialVersionUID = 1L;
        private final double weight;
        private final double measuredValue;
        private final double residual;
        private final double theoreticalValue;

        public DummyWeightedMeasurement(double weight, double measuredValue, double residual) {
            super(weight, measuredValue);
            this.weight = weight;
            this.measuredValue = measuredValue;
            this.residual = residual;
            this.theoreticalValue = measuredValue - residual; // calculated theoretical value
        }

        @Override
        public double getTheoreticalValue() {
            return theoreticalValue;
        }

        @Override
        public double getPartial(EstimatedParameter parameter) {
            // Default implementation: assume partial derivative is 0 unless overridden
            return 0.0;
        }

        @Override
        public double getWeight() { return weight; }
        @Override
        public double getMeasuredValue() { return measuredValue; }
        @Override
        public double getResidual() { return residual; }
    }
