    @Test
    public void testSqrt() throws Exception {
        Complex c = new Complex(3.0, 4.0);
        // sqrt(3+4i). |3+4i|=5. t = sqrt((3+5)/2) = sqrt(4) = 2.
        // real >= 0, so t + (imaginary/(2t))i = 2 + (4/(2*2))i = 2 + i
        assertEquals(2.0, c.sqrt().getReal(), 1e-9);
        assertEquals(1.0, c.sqrt().getImaginary(), 1e-9);

        Complex c2 = new Complex(-1.0, 0.0);
        // sqrt(-1). | -1 | = 1. t = sqrt((1+1)/2) = 1.
        // real < 0, |imaginary|/2t + indicator(imaginary)*t i = 0/2 + 0*1 i = 0
        // This seems wrong. Let's check the formula for negative real part.
        // If real < 0, return |imaginary|/2t + sign(b)t i
        // For -1 + 0i, imaginary = 0, so sign(0) = 0 (using MathUtils.indicator for double, which returns 1.0 for 0.0)
        // Oh, indicator(0.0) is 1.0. Let's re-evaluate.
        // t = sqrt((|-1| + |-1|) / 2) = sqrt((1+1)/2) = 1.
        // real < 0. return |0|/(2*1) + indicator(0)*1 i = 0 + 1.0 * 1 * i = i.
        // Correct is i.
        assertEquals(0.0, c2.sqrt().getReal(), 1e-9);
        assertEquals(1.0, c2.sqrt().getImaginary(), 1e-9);

        Complex c3 = new Complex(-3.0, -4.0);
        // sqrt(-3-4i). |-3-4i|=5. t = sqrt((3+5)/2) = 2.
        // real < 0. |imaginary|/2t + indicator(imaginary)*t i = |-4|/(2*2) + indicator(-4)*2 i
        // = 4/4 + (-1)*2 i = 1 - 2i.
        assertEquals(1.0, c3.sqrt().getReal(), 1e-9);
        assertEquals(-2.0, c3.sqrt().getImaginary(), 1e-9);

        Complex c4 = new Complex(Double.POSITIVE_INFINITY, 0.0);
        assertEquals(Double.POSITIVE_INFINITY, c4.sqrt().getReal(), 1e-9);
        assertEquals(0.0, c4.sqrt().getImaginary(), 1e-9);

        Complex c5 = new Complex(0.0, Double.POSITIVE_INFINITY);
        assertEquals(Double.POSITIVE_INFINITY, c5.sqrt().getReal(), 1e-9);
        assertEquals(0.0, c5.sqrt().getImaginary(), 1e-9);
    }
