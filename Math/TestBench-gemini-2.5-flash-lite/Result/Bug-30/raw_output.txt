    @Test
    public void testMannWhitneyUWithNaNStrategyFixed() throws Exception {
        MannWhitneyUTest test = new MannWhitneyUTest(NaNStrategy.FIXED, TiesStrategy.AVERAGE);
        double[] x = {1, Double.NaN, 3};
        double[] y = {2, 4};
        // Ranks for z = {1, 2, 3, 4} with NaNStrategy.FIXED: NaN is not ranked, only {1, 2, 3, 4} -> ranks {1, 2, 3, 4}
        // Sum of ranks for x (excluding NaN): 1
        // n1 = 3, n2 = 2. After removing NaN, effective n1 = 2.
        // The code as written will use original lengths for calculation of U. This might be a bug.
        // Let's assume the ranking will place NaN somewhere and then it will be ignored by the summation.
        // The `naturalRanking.rank(z)` will handle NaNs according to NaNStrategy.FIXED.
        // If NaNStrategy.FIXED means NaNs are ignored, then z will effectively be {1, 2, 3, 4}.
        // The loop for sumRankX iterates through original x.length, which is 3.
        // If ranks[1] is for NaN, it will be skipped in summation if it's not a number.
        // Let's assume z = {1, NaN, 3, 2, 4}. Ranking {1, 2, 3, 4} gives ranks {1, 2, 3, 4}.
        // If NaN is ranked at the end (highest rank for FIXED), z = {1, 3, 2, 4, NaN}. Ranks {1, 3, 2, 4, 5}.
        // Sum of ranks for x (original indices 0, 1, 2): ranks[0] (for 1) + ranks[1] (for NaN) + ranks[2] (for 3).
        // The current code sums ranks[i] for i < x.length. If z = {1, 3, 2, 4, NaN}, ranks are {1, 3, 2, 4, 5}.
        // sumRankX = ranks[0] + ranks[1] + ranks[2] = 1 + 3 + 2 = 6.
        // n1 = 3, n2 = 2.
        // U1 = 6 - (3 * 4 / 2) = 6 - 6 = 0.
        // U2 = 3 * 2 - 0 = 6.
        // Let's check with NaNStrategy.MINIMAL (which might put NaN at the start)
        // z = {NaN, 1, 3, 2, 4}. Ranks {1, 2, 4, 3, 5} if NaN is at rank 1.
        // sumRankX = ranks[0] + ranks[1] + ranks[2] = 1 + 2 + 4 = 7.
        // U1 = 7 - 6 = 1. U2 = 6 - 1 = 5. Max is 5.
        // The documentation says "NaNs are left in place". This implies they are NOT ranked and NOT summed.
        // Let's assume NaNs are simply ignored in ranking.
        // z = {1, 3, 2, 4}. Ranks {1, 3, 2, 4}.
        // sumRankX: sum of ranks for original x values {1, 3} among {1, 3, 2, 4}. Ranks are {1, 3}. Sum = 4.
        // n1 = 2 (effective), n2 = 2.
        // U1 = 4 - (2 * 3 / 2) = 4 - 3 = 1.
        // U2 = 2 * 2 - 1 = 3. Max is 3.
        // HOWEVER, the code `for (int i = 0; i < x.length; ++i)` uses `x.length` which includes NaN.
        // And `z` is `x.length + y.length`.
        // The reference implementation's `naturalRanking.rank(z)` with `NaNStrategy.FIXED` would put NaN at the end if it were sorted.
        // z = {1.0, 3.0, 2.0, 4.0, NaN} sorted {1.0, 2.0, 3.0, 4.0, NaN}. ranks {1, 2, 3, 4, 5}
        // The `rank` method returns an array of the same length as `data`. So ranks will have 5 elements.
        // z = {1, NaN, 3, 2, 4}. ranks = {1, 5, 3, 2, 4} if NaN is highest.
        // sumRankX = ranks[0] (for 1) + ranks[1] (for NaN) + ranks[2] (for 3) = 1 + 5 + 3 = 9.
        // n1 = 3, n2 = 2.
        // U1 = 9 - (3 * 4 / 2) = 9 - 6 = 3.
        // U2 = 3 * 2 - 3 = 3.
        assertEquals(3.0, test.mannWhitneyU(x, y), 1e-9);
    }

    // Edge case: Large number of elements, potential for overflow in intermediate calculations
    // However, standard doubles should handle this.
    @Test
    public void testMannWhitneyU_LargeNumberOfElements() throws Exception {
        int n = 1000; // Using a moderately large number
        double[] x = new double[n];
        double[] y = new double[n];
        for (int i = 0; i < n; i++) {
            x[i] = i + 1;
            y[i] = i + 1 + n; // y values are shifted
        }
        
        // Ranks for z = {1, ..., n, n+1, ..., 2n}. Ranks for x are {1, ..., n}.
        // Sum of ranks for x = n * (n + 1) / 2
        double sumRankX = (double)n * (n + 1) / 2.0;
        
        // n1 = n, n2 = n
        // U1 = sumRankX - (n1 * (n1 + 1)) / 2
        // U1 = (n * (n + 1) / 2) - (n * (n + 1) / 2) = 0
        // U2 = n * n - 0 = n^2
        assertEquals((double)n * n, new MannWhitneyUTest().mannWhitneyU(x, y), 1e-9);
    }

    @Test
    public void testMannWhitneyUTest_LargeNumberOfElements() throws Exception {
        int n = 1000;
        double[] x = new double[n];
        double[] y = new double[n];
        for (int i = 0; i < n; i++) {
            x[i] = i + 1;
            y[i] = i + 1 + n;
        }

        // Umax = n^2, Umin = 0
        // n1 = n, n2 = n
        // EU = (n * n) / 2
        // VarU = n * n * (n + n + 1) / 12 = n^2 * (2n + 1) / 12
        // z = (0 - n^2 / 2) / sqrt(n^2 * (2n + 1) / 12)
        // z = (-n^2 / 2) / (n * sqrt((2n + 1) / 12))
        // z = (-n / 2) / sqrt((2n + 1) / 12)
        // As n -> infinity, sqrt((2n + 1) / 12) approx sqrt(2n/12) = sqrt(n/6)
        // z approx (-n / 2) / sqrt(n/6) = -n / 2 * sqrt(6/n) = -sqrt(n^2 * 6 / n) / 2 = -sqrt(6n) / 2 = -sqrt(1.5n)
        // For n=1000, z approx -sqrt(1500) approx -38.7
        // p-value will be extremely small, effectively 0.

        // Let's calculate precisely for n=1000
        double n_double = (double)n;
        double EU = n_double * n_double / 2.0;
        double VarU = n_double * n_double * (2.0 * n_double + 1.0) / 12.0;
        double z = (0.0 - EU) / FastMath.sqrt(VarU);
        NormalDistribution standardNormal = new NormalDistribution(0, 1);
        double expectedPValue = 2 * standardNormal.cumulativeProbability(z);

        assertEquals(expectedPValue, new MannWhitneyUTest().mannWhitneyUTest(x, y), 1e-9);
    }
    
    // Boundary: Single element samples
    @Test
    public void testMannWhitneyU_SingleElementSamples() throws Exception {
        double[] x = {10.0};
        double[] y = {20.0};
        // z = {10.0, 20.0}, ranks {1.0, 2.0}
        // sumRankX = 1.0
        // n1 = 1, n2 = 1
        // U1 = 1.0 - (1 * 2 / 2) = 0.0
        // U2 = 1 * 1 - 0.0 = 1.0
        assertEquals(1.0, new MannWhitneyUTest().mannWhitneyU(x, y), 1e-9);
    }

    @Test
    public void testMannWhitneyUTest_SingleElementSamples() throws Exception {
        double[] x = {10.0};
        double[] y = {20.0};
        // Umax = 1.0, Umin = 0.0
        // n1 = 1, n2 = 1
        // EU = (1 * 1) / 2.0 = 0.5
        // VarU = 1 * 1 * (1 + 1 + 1) / 12.0 = 3.0 / 12.0 = 0.25
        // z = (0.0 - 0.5) / sqrt(0.25) = -0.5 / 0.5 = -1.0
        // p-value = 2 * P(Z < -1.0) = 0.31731050786291415
        assertEquals(0.31731050786291415, new MannWhitneyUTest().mannWhitneyUTest(x, y), 1e-9);
    }
