    @Test
    public void testLog10() {
        DfpField field = new DfpField(10);
        // A number with exponent 1 (4 digits) and most significant digit > 1000
        Dfp num1 = field.newInstance("10000"); // exp = 1
        // log10(10000) = 4. exp*4 = 4. mantissa[last] = 0, so exp*4 - 4 = 0
        // Let's re-check the logic of log10().
        // If mant[last] > 1000 (i.e., > 10^3), then exp * 4 - 1
        // If mant[last] > 100, then exp * 4 - 2
        // If mant[last] > 10, then exp * 4 - 3
        // Else, exp * 4 - 4
        // For 10000, exp = 1. mant[last] = 1. This falls into the last case. 1*4 - 4 = 0.
        assertEquals(0, num1.log10());

        // A number with exponent 1 (4 digits) and most significant digit > 1000
        Dfp num2 = field.newInstance("12340000"); // exp = 2. mant[last] = 1234. exp*4 - 1 = 2*4 - 1 = 7.
        assertEquals(7, num2.log10());

        // A number with exponent 1 and mantissa 1000.
        Dfp num3 = field.newInstance("10000"); // exp = 1. mant[last] = 1. This is exp*4 - 4 = 0.
        assertEquals(0, num3.log10());

        // A number just above 1000 in mantissa.
        Dfp num4 = field.newInstance("10010000"); // exp = 2. mant[last]=1001. exp*4 - 1 = 2*4 - 1 = 7.
        assertEquals(7, num4.log10());

        // A number with fractional exponent
        Dfp num5 = field.newInstance("12345"); // exp = 2. mant[last]=1234. exp*4 - 1 = 2*4 - 1 = 7.
        assertEquals(7, num5.log10());
    }
