    /**
     * Test case for the simple Rosenbrock function.
     * This function is a common benchmark for optimization algorithms.
     */
    @Test
    public void testRosenbrockFunction() {
        final double[] startPoint = {0, 0};
        final double[] expected = {1, 1};
        final double epsilon = 1e-10; // Tolerance for convergence

        CMAESOptimizer optimizer = new CMAESOptimizer();
        optimizer.setGoalType(GoalType.MINIMIZE);
        optimizer.setMaxIterations(10000); // Increased iterations for better convergence
        optimizer.setConvergenceChecker(new SimpleValueChecker(epsilon, epsilon)); // Set convergence checker

        // Define the Rosenbrock function
        MultivariateFunction rosenbrock = new MultivariateFunction() {
            @Override
            public double value(double[] point) {
                double x1 = point[0];
                double x2 = point[1];
                return Math.pow(1 - x1, 2) + 100 * Math.pow(x2 - x1 * x1, 2);
            }
        };

        // Set start point and optimize
        optimizer.setStartPoint(startPoint);
        PointValuePair result = optimizer.optimize(rosenbrock, GoalType.MINIMIZE, new double[]{0,0}, new double[]{2,2}); // Provide bounds

        // Assert that the result is close to the expected minimum
        double[] point = result.getPoint();
        for (int i = 0; i < expected.length; i++) {
            assertEquals(expected[i], point[i], epsilon);
        }
        assertEquals(0.0, result.getValue(), epsilon);
    }

    @Test
    public void testInitializeCmaWithBoundsAndInputSigma() {
        int lambda = 40;
        double[] inputSigma = {0.5, 0.2};
        double[] guess = {0.0, 0.0};
        double[] lowerBounds = {0.0, 0.0};
        double[] upperBounds = {1.0, 1.0};

        CMAESOptimizer optimizer = new CMAESOptimizer(lambda, inputSigma);
        optimizer.setStartPoint(guess);
        optimizer.setLowerBound(lowerBounds);
        optimizer.setUpperBound(upperBounds);
        optimizer.initializeCMA(guess);

        assertEquals(lambda, optimizer.lambda);
        assertEquals(guess.length, optimizer.dimension);
        assertEquals(1.0, optimizer.sigma, 1e-9); // sigma is max(inputSigma / range) = max(0.5/1.0, 0.2/1.0) = 0.5. Wait, no, it's initial sigma.
                                                 // The code says: sigma = max(insigma); insigma = inputSigma / range
                                                 // For guess={0,0}, range={1,1}. So insigma = {0.5, 0.2}. max(insigma) = 0.5.
                                                 // However, the default sigma in the code is 0.3. Let's check the initialization logic again.
                                                 // `sigma = max(insigma);` is correct.
                                                 // The issue might be in how inputSigma is used or interpreted in the reference source.
                                                 // The default is 0.3 * range. Here, it seems to be derived from inputSigma.
                                                 // Let's re-evaluate based on the source:
                                                 // `double range = (boundaries == null) ? 1.0 : boundaries[1][i] - boundaries[0][i];`
                                                 // `sigmaArray[i][0] = ((inputSigma == null) ? 0.3 : inputSigma[i]) / range;`
                                                 // `sigma = max(insigma);`
                                                 // For {0.5, 0.2} and bounds {0,1}, range is {1,1}.
                                                 // sigmaArray[0][0] = 0.5 / 1.0 = 0.5
                                                 // sigmaArray[1][0] = 0.2 / 1.0 = 0.2
                                                 // insigma = {{0.5}, {0.2}}
                                                 // sigma = max(insigma) = 0.5.
        assertEquals(0.5, optimizer.sigma, 1e-9);

        // Check initial diagD
        double[] expectedDiagD = {0.5 / 0.5, 0.2 / 0.5}; // inputSigma / sigma
        assertArrayEquals(expectedDiagD, optimizer.diagD.getColumn(0), 1e-9);
    }

    @Test

        double[] lowerBounds = {0.0, -1.0, 1.0};
        double[] upperBounds = {1.0, 1.0, 2.0};

        CMAESOptimizer optimizer = new CMAESOptimizer();

        optimizer.setUpperBound(upperBounds);
        optimizer.setStartPoint(new double[3]);

        CMAESOptimizer.FitnessFunction ff = optimizer.new FitnessFunction();
        double[] repairedAndDecoded = ff.repairAndDecode(point);

        // Expected repaired and decoded values:
        // point[0] = -0.5 becomes 0.0 (repaired)
        // point[1] = 1.5 becomes 1.0 (repaired)
        // point[2] = 0.5 stays 0.5
        // Then decode these repaired values:
        // repaired[0] = 0.0 => lower[0] + 0.0 * range[0] = 0.0 + 0.0 * 1.0 = 0.0
        // repaired[1] = 1.0 => lower[1] + 1.0 * range[1] = -1.0 + 1.0 * 2.0 = -1.0 + 2.0 = 1.0
        // repaired[2] = 0.5 => lower[2] + 0.5 * range[2] = 1.0 + 0.5 * 1.0 = 1.0 + 0.5 = 1.5
        double[] expected = {0.0, 1.0, 1.5};
        assertArrayEquals(expected, repairedAndDecoded, 1e-9);
    }

    /**
     * Test `FitnessFunction.setValueRange` and its effect on penalty.
     */
    @Test
    public void testSetValueRangeAndPenalty() {
        double[] point = {-0.5, 0.5}; // Normalized point outside bounds
        double[] lowerBounds = {0.0, 0.0};
        double[] upperBounds = {1.0, 1.0};

        CMAESOptimizer optimizer = new CMAESOptimizer();
        optimizer.setLowerBound(lowerBounds);
        optimizer.setUpperBound(upperBounds);
        optimizer.setStartPoint(new double[2]);

        CMAESOptimizer.FitnessFunction ff = optimizer.new FitnessFunction();
        ff.setValueRange(10.0); // Set a custom value range

        // Use a simple objective function that returns 0
        MultivariateFunction zeroObjective = new MultivariateFunction() {
            @Override
            public double value(double[] p) {
                return 0.0;
            }
        };
        optimizer.computeObjectiveValue = zeroObjective;

        double fitness = ff.value(point);

        // The point -0.5 becomes 0.0 after repair.
        // Penalty = abs(point[0] - repaired[0]) * valueRange
        // Penalty = abs(-0.5 - 0.0) * 10.0 = 0.5 * 10.0 = 5.0
        // Total fitness = 0 (objective) + 5.0 (penalty) = 5.0
        assertEquals(5.0, fitness, 1e-9);
    }
