    // This is a utility for deriving expected values for density, not a test method itself.
    private double calculateExponentTerm(final double[] means, final RealMatrix covarianceMatrixInverse, final double[] values) {
        final double[] centered = new double[values.length];
        for (int i = 0; i < centered.length; i++) {
            centered[i] = values[i] - means[i];
        }
        final double[] preMultiplied = covarianceMatrixInverse.preMultiply(centered);
        double sum = 0;
        for (int i = 0; i < preMultiplied.length; i++) {
            sum += preMultiplied[i] * centered[i];
        }
        return FastMath.exp(-0.5 * sum);
    }

    // Helper to get covarianceMatrixInverse for deriving expected values, NOT FOR DIRECT TESTING OF PRIVATE FIELDS
    private RealMatrix getCovarianceMatrixInverseForDerivation(final double[] means, final double[][] covariances) {
        final RealMatrix covMatrix = new Array2DRowRealMatrix(covariances);
        final EigenDecomposition covMatDec = new EigenDecomposition(covMatrix);
        return covMatDec.getSolver().getInverse();
    }
    
    // Helper to get covarianceMatrixDeterminant for deriving expected values
    private double getCovarianceMatrixDeterminantForDerivation(final double[] means, final double[][] covariances) {
        final RealMatrix covMatrix = new Array2DRowRealMatrix(covariances);
        final EigenDecomposition covMatDec = new EigenDecomposition(covMatrix);
        return covMatDec.getDeterminant();
    }
    
    // Helper to get dimension for deriving expected values

    @Test
    public void testDensity1DWithOffsetAndScale() {
        final double[] means = {1.0};
        final double[][] covariances = {{4.0}}; // Std dev = 2
        final MultivariateNormalDistribution dist = new MultivariateNormalDistribution(means, covariances);
        
        final double x = 3.0; // This is 1 std dev away from the mean (3.0 - 1.0 = 2.0)
        final double dim = getDimension(means);
        final double det = getCovarianceMatrixDeterminantForDerivation(means, covariances);
        final RealMatrix invCov = getCovarianceMatrixInverseForDerivation(means, covariances);
        
        final double exponentTerm = calculateExponentTerm(means, invCov, new double[]{x});
        
        final double expectedDensity = FastMath.pow(2 * FastMath.PI, -0.5 * dim) *
                                       FastMath.pow(det, -0.5) *
                                       exponentTerm;
        
        assertEquals(expectedDensity, dist.density(new double[]{x}), 1e-9);
    }

        // expectedDensity = (1 / (2 * pi * sqrt(det))) * exp(0)
        // sqrt(det) = sqrt(1e20) = 1e10
        // expectedDensity = 1 / (2 * pi * 1e10)
        final double expectedDensity = 1.0 / (2 * FastMath.PI * 1e10);
        
        assertEquals(expectedDensity, dist.density(vals), 1e-9);
    }
    
    @Test
    public void testCovarianceWithLargeOffDiagonal() {
        final double[] means = {0.0, 0.0};
        // A covariance matrix where off-diagonal is large but still positive definite.
        // Example: [[2, 1.9], [1.9, 2]] -> determinant = 4 - 1.9*1.9 = 4 - 3.61 = 0.39
        final double[][] covariances = {{2.0, 1.9}, {1.9, 2.0}};
        final MultivariateNormalDistribution dist = new MultivariateNormalDistribution(means, covariances);
        
        final double[] vals = {0.0, 0.0};
        final double dim = getDimension(means);
        final double det = getCovarianceMatrixDeterminantForDerivation(means, covariances); // 0.39
        final RealMatrix invCov = getCovarianceMatrixInverseForDerivation(means, covariances);
        
        final double exponentTerm = calculateExponentTerm(means, invCov, vals); // 1.0
        
        final double expectedDensity = FastMath.pow(2 * FastMath.PI, -0.5 * dim) *
                                       FastMath.pow(det, -0.5) *
                                       exponentTerm;
        
        assertEquals(expectedDensity, dist.density(vals), 1e-9);
    }
