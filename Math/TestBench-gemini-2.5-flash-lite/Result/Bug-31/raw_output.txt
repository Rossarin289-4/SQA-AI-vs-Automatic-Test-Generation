    @Test
    public void testEvaluatePrecisionEquals() throws Exception {
        // This test focuses on the Precision.equals checks.
        // The small threshold is 1e-50.
        class PrecisionCheckFraction extends ContinuedFraction {
            private int callCount = 0;
            @Override
            protected double getA(int n, double x) {
                if (n == 0) return 1.0;
                return 1.0;
            }
            @Override
            protected double getB(int n, double x) {
                if (n == 1 && callCount == 0) {
                    callCount++;
                    return 1e-55; // Very small b, to test dN, cN near zero.
                }
                return 1.0;
            }
        }
        ContinuedFraction cf = new PrecisionCheckFraction();
        // n=0: hPrev = 1.0
        // n=1: a=1, b=1e-55. dPrev=0, cPrev=1.0.
        // dN = a + b*dPrev = 1 + 1e-55*0 = 1.0. Not close to 0.
        // cN = a + b/cPrev = 1 + 1e-55/1.0 = 1 + 1e-55. This is close to 1.0.
        // The code checks: if (Precision.equals(dN, 0.0, small)) { dN = small; }
        // The code checks: if (Precision.equals(cN, 0.0, small)) { cN = small; }
        // Neither dN nor cN are near zero in this case.

        // Let's try to make dN or cN near zero.
        class PrecisionCheckFraction2 extends ContinuedFraction {
            private int callCount = 0;
            @Override
            protected double getA(int n, double x) {
                if (n == 0) return 1.0;
                if (n == 1) return 1.0;
                if (n == 2) return 1.0; // To make the calculation proceed
                return 1.0;
            }
            @Override
            protected double getB(int n, double x) {
                if (n == 1 && callCount == 0) {
                    callCount++;
                    return -1.0; // This can lead to small dN or cN.
                }
                return 1.0;
            }
        }
        ContinuedFraction cf2 = new PrecisionCheckFraction2();
        // n=0: hPrev = 1.0
        // n=1: a=1, b=-1. dPrev=0, cPrev=1.
        // dN = 1 + (-1)*0 = 1.0. Not near 0.
        // cN = 1 + (-1)/1 = 0.0.
        // Precision.equals(cN, 0.0, small) is true. cN becomes small (1e-50).
        // dN = 1 / 1.0 = 1.0
        // deltaN = cN * dN = 1e-50 * 1.0 = 1e-50.
        // hN = hPrev * deltaN = 1.0 * 1e-50 = 1e-50.
        // abs(1e-50 - 1.0) is large.
        // dPrev = 1, cPrev = 1e-50, hPrev = 1e-50.
        // n=2: a=1, b=1.
        // dN = 1 + 1*1 = 2.0. Not near 0.
        // cN = 1 + 1/1e-50 = 1 + 1e50. Not near 0.
        // dN = 1/2 = 0.5.
        // deltaN = cN * dN = (1+1e50) * 0.5 approx 0.5e50.
        // hN = hPrev * deltaN = 1e-50 * 0.5e50 = 0.5.
        // abs(deltaN - 1.0) is large.
        // Let's assume it converges to some value.
        // We can assert that the result is finite and not NaN.
        double result = cf2.evaluate(0.0, DEFAULT_EPSILON);
        assertTrue(Double.isFinite(result));
    }

    @Test
    public void testEvaluateWithEpsilonJustAboveDelta() throws Exception {
        ContinuedFraction cf = new GoldenRatioContinuedFraction();
        // We calculated for GR:
        // n=1: deltaN=2, |deltaN-1|=1
        // n=2: deltaN=0.75, |deltaN-1|=0.25
        // n=3: deltaN=10/9 approx 1.11, |deltaN-1| approx 0.11
        // n=4: deltaN=24/25 = 0.96, |deltaN-1| = 0.04
        // If epsilon is just slightly larger than 0.04, it should stop at n=4.
        // Let epsilon = 0.04000000000000001
        double epsilon = 0.04000000000000001;
        // hPrev = 1.5 (after n=2)
        // n=3: a=1, b=1. dPrev=0.5, cPrev=1.5. dN=1.5, cN=5/3. dN=2/3. deltaN=(5/3)*(2/3)=10/9. hN=(3/2)*(10/9)=5/3. |10/9-1|=1/9 approx 0.111. 0.111 is NOT < epsilon.
        // dPrev=2/3, cPrev=5/3, hPrev=5/3.
        // n=4: a=1, b=1. dN=1+1*(2/3)=5/3. cN=1+1/(5/3)=8/5. dN=3/5. deltaN=(8/5)*(3/5)=24/25=0.96. hN=(5/3)*(24/25)=8/5=1.6. |0.96-1|=0.04.
        // Is 0.04 < epsilon (0.04000000000000001)? Yes. Converged.
        // Result hN = 1.6.
        assertEquals(1.6, cf.evaluate(0.0, epsilon), epsilon);
    }

                return 1.0;
            }

            }
        }
        ContinuedFraction cf = new SmallCoeffsFraction();
        // n=0: hPrev = 1e-10. Precision.equals(hPrev, 0.0, small) is false.
        // n=1: a=1, b=1e-10. dPrev=0, cPrev=1e-10.
        // dN = 1 + 1e-10 * 0 = 1.0.
        // cN = 1 + 1e-10 / 1e-10 = 1 + 1 = 2.0.
        // dN = 1/1 = 1.0.
        // deltaN = cN * dN = 2.0 * 1.0 = 2.0.
        // hN = hPrev * deltaN = 1e-10 * 2.0 = 2e-10.
        // abs(2.0 - 1.0) = 1.0.
        // dPrev = 1.0, cPrev = 2.0, hPrev = 2e-10.
        // n=2: a=1, b=1.
        // dN = 1 + 1*1 = 2.0.
        // cN = 1 + 1/2.0 = 1.5.
        // dN = 1/2 = 0.5.
        // deltaN = 1.5 * 0.5 = 0.75.
        // hN = hPrev * deltaN = 2e-10 * 0.75 = 1.5e-10.
        // abs(0.75 - 1.0) = 0.25.
        // This seems to be converging to a small value, close to 0.
        // The actual value of 1e-10 + 1e-10 / (1 + 1/(1 + ...)) = 1e-10 + 1e-10 / phi = 1e-10 * (1 + 1/phi) = 1e-10 * (1 + phi - 1) = 1e-10 * phi = 1e-10 * 1.618...
        // Let's re-verify manual calculations for hN.
        // hN = hPrev * deltaN.
        // n=1: hN = 1e-10 * 2 = 2e-10.
        // n=2: hN = 2e-10 * 0.75 = 1.5e-10.
        // n=3: a=1, b=1. dPrev=0.5, cPrev=1.5. dN=1.5. cN=1+1/1.5=5/3. dN=1/1.5=2/3. deltaN=(5/3)*(2/3)=10/9. hN=1.5e-10 * (10/9) = (3/2)e-10 * (10/9) = 30/18 e-10 = 5/3 e-10 approx 1.666e-10.
        // The value is approximately 1.618 * 1e-10.
        assertEquals(1.6180339887e-10, cf.evaluate(0.0, DEFAULT_EPSILON), DEFAULT_EPSILON * 1e-10);
    }

    @Test
    public void testEvaluateConvergenceExceptionWithSpecificX() throws Exception {
        // Test that the exception message includes the evaluation point x.
        // Use a fraction and x that will definitely diverge.
        class DivergingFraction extends ContinuedFraction {
            @Override
            protected double getA(int n, double x) {
                return 1.0;
            }
            @Override
            protected double getB(int n, double x) {
                // This will cause a_n + b_n * dPrev to grow.
                // If b_n is large, it might cause overflow or divergence.
                return 100.0; // Large b coefficient
            }
        }
        ContinuedFraction cf = new DivergingFraction();
        double x = 5.0; // A specific evaluation point.
        try {
            cf.evaluate(x, DEFAULT_EPSILON, 100); // Limit iterations to ensure it throws
            fail("Expected ConvergenceException");
        } catch (ConvergenceException e) {
            // Check if the exception message contains the evaluation point.
            // The LocalizedFormats.NON_CONVERGENT_CONTINUED_FRACTION pattern takes maxIterations and x.
            // The LocalizedFormats.CONTINUED_FRACTION_INFINITY_DIVERGENCE pattern takes x.
            // The LocalizedFormats.CONTINUED_FRACTION_NAN_DIVERGENCE pattern takes x.
            // The DivergingFraction might lead to Infinity.
            assertTrue(e instanceof ConvergenceException);
            // Depending on the exact divergence, it might be Infinity or NaN or MaxCountExceeded.
            // If it leads to Infinity, the message should contain x.
            // Let's assume it leads to Infinity or NaN for this test.
            // If it leads to MaxCountExceeded, it won't test the x in ConvergenceException directly.

            // Let's force infinity.
            class InfinityDivergingFraction extends ContinuedFraction {
                @Override
                protected double getA(int n, double x) {
                    if (n == 1) return 1.0;
                    return 1.0;
                }
                @Override
                protected double getB(int n, double x) {
                    if (n == 1) return Double.POSITIVE_INFINITY; // Force infinity
                    return 1.0;
                }
            }
            try {
                new InfinityDivergingFraction().evaluate(x, DEFAULT_EPSILON, 100);
                fail("Expected ConvergenceException for Infinity");
            } catch (ConvergenceException eInf) {
                assertEquals(LocalizedFormats.CONTINUED_FRACTION_INFINITY_DIVERGENCE, eInf.getPattern());
                // Check if the argument to the exception constructor is x.
                Object[] args = eInf.getArguments();
                assertNotNull(args);
                assertTrue(args.length >= 1);
                assertEquals(x, (Double) args[0], 0.0);
            }
        }
    }
