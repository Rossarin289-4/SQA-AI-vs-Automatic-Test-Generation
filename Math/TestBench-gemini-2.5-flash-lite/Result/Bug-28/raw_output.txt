        LinearObjectiveFunction f = new LinearObjectiveFunction(new double[]{-1, -2}, 0);
        Collection<LinearConstraint> constraints = new ArrayList<>();

        // After initial setup, if we try to pivot on column 1 (x2):
        // Row 1 (s1): rhs=2, entry=2 -> ratio = 1
        // Row 2 (s2): rhs=2, entry=1 -> ratio = 2
        // Minimum ratio is 1 for row 1 (index 1 of the data part, which corresponds to height=2, so row 1).
        // Let's create a scenario where there is a tie.
        constraints.clear();
        constraints.add(new LinearConstraint(new double[]{1, 1}, Relationship.LEQ, 2)); // x1 + x2 <= 2
        constraints.add(new LinearConstraint(new double[]{2, 2}, Relationship.LEQ, 4)); // 2x1 + 2x2 <= 4 (redundant)
        tableau = new SimplexTableau(f, constraints, GoalType.MAXIMIZE, true, 1e-6, 10);
        // Objective row: -1, -2, 0, 0, 0 (x1, x2, s1, a1, rhs)
        // Constraint 1: 1, 1, 1, 0, 2
        // Constraint 2: 2, 2, 0, 1, 4
        // If we pivot on col 1 (x2):
        // Row 1 (s1): entry=1, rhs=2 -> ratio = 2
        // Row 2 (s2): entry=2, rhs=4 -> ratio = 2
        // Tie in ratios. Bland's rule should pick the row with the smaller basic variable index.
        // We need to inspect the basic variable indices for these rows.
        // Assuming s1 is basic variable for row 1, and s2 for row 2.
        // The current tableau structure might not directly reveal this without deeper inspection.
        // Let's try a more direct test setup.

        // Setup for a tie that forces Bland's rule
        LinearObjectiveFunction f2 = new LinearObjectiveFunction(new double[]{-1, -1}, 0);
        Collection<LinearConstraint> constraints2 = new ArrayList<>();
        constraints2.add(new LinearConstraint(new double[]{1, 0}, Relationship.LEQ, 1)); // x1 <= 1
        constraints2.add(new LinearConstraint(new double[]{0, 1}, Relationship.LEQ, 1)); // x2 <= 1
        SimplexTableau tableau2 = new SimplexTableau(f2, constraints2, GoalType.MAXIMIZE, true, 1e-6, 10);
        // Objective: -1 -1 0 0 0 (x1 x2 s1 s2 rhs)
        // Row 1 (s1): 1 0 1 0 1
        // Row 2 (s2): 0 1 0 1 1

        // Assume we pivot on column 0 (x1).
        // Row 1: rhs=1, entry=1 -> ratio = 1
        // Row 2: rhs=1, entry=0 -> ratio is undefined (entry <= 0)
        // So we pivot on row 1.

        // Assume we pivot on column 1 (x2).
        // Row 1: rhs=1, entry=0 -> ratio is undefined
        // Row 2: rhs=1, entry=1 -> ratio = 1
        // So we pivot on row 2.

        // To get a tie, let's try a different problem structure.
        // Example that leads to tie in MRT and Bland's rule
        // Maximize x1 + x2 subject to:

        constraints3.add(new LinearConstraint(new double[]{1, 2}, Relationship.LEQ, 2));
        constraints3.add(new LinearConstraint(new double[]{2, 1}, Relationship.LEQ, 2));
        SimplexTableau tableau3 = new SimplexTableau(f3, constraints3, GoalType.MAXIMIZE, true, 1e-6, 10);
        // Objective: -1 -1 0 0 0 (x1 x2 s1 s2 rhs)
        // Row 1 (s1): 1 2 1 0 2
        // Row 2 (s2): 2 1 0 1 2

        // Pivot column: getPivotColumn will pick col 0 (x1) as it's the first most negative.
        Integer pivotCol = tableau3.getPivotColumn(); // Should be 0
        // MRT for col 0:
        // Row 1 (s1): rhs=2, entry=1 -> ratio = 2
        // Row 2 (s2): rhs=2, entry=2 -> ratio = 1
        // Min ratio is 1, row is 1 (index of data, so tableau row 2).
        assertEquals(Integer.valueOf(0), pivotCol);
        assertEquals(Integer.valueOf(1), tableau3.getPivotRow(pivotCol.intValue())); // Row 1 (index 1) has min ratio.

        // Now let's force a tie.
        // Maximize x1 + x2 subject to:
        // x1 + x2 <= 2

        // Row 1 (s1): 1 1 1 0 2
        // Row 2 (s2): 1 1 0 1 3

        Integer pivotCol4 = tableau4.getPivotColumn(); // Should be 0
        // MRT for col 0:
        // Row 1 (s1): rhs=2, entry=1 -> ratio = 2
        // Row 2 (s2): rhs=3, entry=1 -> ratio = 3
        // Min ratio is 2, row is 1.
        assertEquals(Integer.valueOf(0), pivotCol4);
        assertEquals(Integer.valueOf(0), tableau4.getPivotRow(pivotCol4.intValue())); // Row 0 (index 0) has min ratio.

        // Let's force a tie by having identical constraints (not very realistic, but for testing)
        // Maximize x1 subject to:
        // x1 + x2 <= 2
        // x1 + x2 <= 2
        LinearObjectiveFunction f5 = new LinearObjectiveFunction(new double[]{1, 0}, 0);
        Collection<LinearConstraint> constraints5 = new ArrayList<>();
        constraints5.add(new LinearConstraint(new double[]{1, 1}, Relationship.LEQ, 2));
        constraints5.add(new LinearConstraint(new double[]{1, 1}, Relationship.LEQ, 2));
        SimplexTableau tableau5 = new SimplexTableau(f5, constraints5, GoalType.MAXIMIZE, true, 1e-6, 10);
        // Objective: -1 0 0 0 0 (x1 x2 s1 s2 rhs)
        // Row 1 (s1): 1 1 1 0 2
        // Row 2 (s2): 1 1 0 1 2

        // Pivot column: pick col 0 (x1)
        Integer pivotCol5 = tableau5.getPivotColumn(); // Should be 0
        // MRT for col 0:
        // Row 1 (s1): rhs=2, entry=1 -> ratio = 2
        // Row 2 (s2): rhs=2, entry=1 -> ratio = 2
        // Tie in ratios. Bland's rule should pick the row with the smallest index.
        // Row 1 (basic var s1) vs Row 2 (basic var s2). The basic variable indices need to be determined.
        // In the default tableau construction, slack variables are basic.
        // s1 is for constraint 1, s2 is for constraint 2.
        // The getBasicRow(col) method is used to find the row for a given column (basic variable).
        // Let's assume s1 corresponds to column 2, and s2 to column 3.
        // For row 0 (which is the objective function row), getBasicRow(0) is null.
        // For row 1, getBasicRow(2) should return 0 (index of this row in the data).
        // For row 2, getBasicRow(3) should return 1 (index of this row in the data).
        // The loop in getPivotRow iterates through minRatioPositions. If it finds a tie, it checks for artificial variables.
        // If not, it applies Bland's rule: "take the row for which the corresponding basic variable has the smallest index".
        // In this case, the basic variables are s1 (column 2) and s2 (column 3).
        // The indices of these variables are 2 and 3. Smallest is 2, which corresponds to row 1 (index 0 in `minRatioPositions`).
        // So, it should return the first row (index 0) of the minRatioPositions list, which is row 1 (index 0 in tableau data).
        assertEquals(Integer.valueOf(0), pivotCol5);
        assertEquals(Integer.valueOf(0), tableau5.getPivotRow(pivotCol5.intValue())); // Row 0 (index 0) should be chosen by Bland's rule.
    }

    @Test
    public void testPivotRowSelectionNoFeasible() {
        // Test getPivotRow when no positive entry exists in the pivot column.
        // This indicates an unbounded solution.
        LinearObjectiveFunction f = new LinearObjectiveFunction(new double[]{-1, -1}, 0);
        Collection<LinearConstraint> constraints = new ArrayList<>();
        // Constraint: -x1 - x2 <= -2  => x1 + x2 >= 2
        // With Slack: -x1 - x2 + s1 = -2.  We need a way to get this into standard form where rhs is positive.
        // Let's reformulate:
        // Maximize x1 + x2 subject to:
        // x1 + x2 <= 1
        // -x1 <= 0
        // -x2 <= 0
        // This problem is bounded.

        // To trigger "no positive entry", we need a column where all entries in rows below objective are <= 0.
        // Consider a problem where we try to pivot on a column that only has negative entries in constraint rows.
        LinearObjectiveFunction f_test = new LinearObjectiveFunction(new double[]{-1, -1}, 0);
        Collection<LinearConstraint> constraints_test = new ArrayList<>();
        constraints_test.add(new LinearConstraint(new double[]{-1, 1}, Relationship.LEQ, 0)); // -x1 + x2 <= 0
        constraints_test.add(new LinearConstraint(new double[]{1, -1}, Relationship.LEQ, 0)); // x1 - x2 <= 0
        SimplexTableau tableau_test = new SimplexTableau(f_test, constraints_test, GoalType.MAXIMIZE, true, 1e-6, 10);

        // Objective row: -1, -1, 0, 0, 0 (x1, x2, s1, s2, rhs)
        // Row 1 (s1): -1, 1, 1, 0, 0
        // Row 2 (s2): 1, -1, 0, 1, 0

        // Pivot column: getPivotColumn should return 0 (x1)
        Integer pivotCol = tableau_test.getPivotColumn();
        assertEquals(Integer.valueOf(0), pivotCol);

        // Now, getPivotRow for column 0:
        // Row 1 (s1): rhs=0, entry=-1. Entry is not > 0.
        // Row 2 (s2): rhs=0, entry=1. Entry is > 0. Ratio = 0 / 1 = 0.
        // So, minRatioPositions = [1] (index of row 2 in data).
        // The method should return 1.
        // This does NOT trigger the "no positive entry" condition.

        // Let's construct a scenario for unboundedness.
        // Maximize x1 subject to:
        // x1 - x2 <= 1
        // -x1 + x2 <= 1
        // This is unbounded.
        LinearObjectiveFunction f_unbounded = new LinearObjectiveFunction(new double[]{1, 0}, 0);
        Collection<LinearConstraint> constraints_unbounded = new ArrayList<>();
        constraints_unbounded.add(new LinearConstraint(new double[]{1, -1}, Relationship.LEQ, 1));
        constraints_unbounded.add(new LinearConstraint(new double[]{-1, 1}, Relationship.LEQ, 1));
        SimplexTableau tableau_unbounded = new SimplexTableau(f_unbounded, constraints_unbounded, GoalType.MAXIMIZE, true, 1e-6, 10);

        // Objective row: -1, 0, 0, 0, 0 (x1, x2, s1, s2, rhs) - Note: the `f` coefficients are negated for maximization.
        // The tableau is constructed from `getFunction()`, which returns a `LinearObjectiveFunction` with coefficients negated if goal is MAXIMIZE.
        // So, for `f = {1, 0}`, tableau objective row would start as `{-1, 0, ...}`.
        // Let's assume getPivotColumn correctly identifies a pivot column, e.g., column 0 (x1).
        // Row 1 (s1): 1, -1, 1, 0, 1
        // Row 2 (s2): -1, 1, 0, 1, 1

        // Pivot column: getPivotColumn finds column 0 (-1 is the most negative).
        Integer pivotCol_unbounded = tableau_unbounded.getPivotColumn(); // Should be 0
        assertEquals(Integer.valueOf(0), pivotCol_unbounded);

        // getPivotRow for column 0:
        // Row 1 (s1): rhs=1, entry=1. Entry > 0. Ratio = 1 / 1 = 1.
        // Row 2 (s2): rhs=1, entry=-1. Entry is NOT > 0.
        // So, only row 1 has a positive entry in the pivot column. The min ratio is 1.
        // The method returns row 1 (index 0).

        // The `UnboundedSolutionException` is thrown when `pivotRow` is null.
        // This happens if *no* row has a positive entry in the pivot column.
        // Let's try to construct such a case.
        // Maximize x1 subject to:
        // -x1 + x2 <= 1
        // -x1 - x2 <= 1
        LinearObjectiveFunction f_unbounded2 = new LinearObjectiveFunction(new double[]{1, 0}, 0);
        Collection<LinearConstraint> constraints_unbounded2 = new ArrayList<>();
        constraints_unbounded2.add(new LinearConstraint(new double[]{-1, 1}, Relationship.LEQ, 1));
