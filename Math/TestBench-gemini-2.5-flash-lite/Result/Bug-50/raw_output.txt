    @Test
    public void testAllowedSolutionLeftSide() throws Exception {
        // Test with AllowedSolution.LEFT_SIDE. The root is 2.
        final RegulaFalsiSolver solver = new RegulaFalsiSolver();
        final UnivariateRealFunction f = new UnivariateRealFunction() {
            public double value(double x) {
                return x * x - 4; // Root at x = 2
            }
        };
        // If the found root is greater than the actual root, it should still be returned if it's on the LEFT_SIDE of the solution.
        // For x*x-4, roots are -2 and 2. If we search between 0 and 5, root is 2.
        // If allowed is LEFT_SIDE, and the result is slightly more than 2, it's not on LEFT_SIDE, so it might return 0.
        // However, since 2 is the exact root, it will be returned.
        // Let's consider a function where the root is not exactly at the midpoint.
        final UnivariateRealFunction g = new UnivariateRealFunction() {
            public double value(double x) {
                return x - 2.5; // Root at x = 2.5
            }
        };
        double result = solver.solve(100, g, 0.0, 5.0, AllowedSolution.LEFT_SIDE);
        // The solver is guaranteed to find a root within the interval.
        // If the solution is 2.5, and allowed is LEFT_SIDE, it should return the closest value to 2.5 from the left.
        // However, the logic in doSolve() returns x1 if f1 is close to ftol, and then applies allowed.
        // If x1 is returned, and it's indeed the root, it might satisfy LEFT_SIDE for the interval [0, 5].
        // Let's check the behavior for a function where the root is guaranteed to be on one side.
        final UnivariateRealFunction h = new UnivariateRealFunction() {
            public double value(double x) {
                return x - 1; // Root at x = 1
            }
        };
        // Search interval [0, 5]. Root is 1.
        // For LEFT_SIDE, we expect a value <= 1.
        result = solver.solve(100, h, 0.0, 5.0, AllowedSolution.LEFT_SIDE);
        assertTrue(result <= 1.0 + solver.getAbsoluteAccuracy()); // Allow for minor inaccuracies
    }
