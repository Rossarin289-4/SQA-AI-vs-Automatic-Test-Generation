    @Test
    public void testHashCodeChangesIfKeyHashCodeChanges() {
        // This tests the constraint that keys should not change after creation.
        // If a key's hashCode changes, the equals() method will fail while hashCode() remains the same.
        // This test verifies that the cached hashCode is based on the *initial* hashCodes.
        // We create a mutable key that changes its hashCode.
        class MutableKey {
            private String value;
            private int hash;

            public MutableKey(String value) {
                this.value = value;
                this.hash = value.hashCode();
            }

            public void changeValue(String newValue) {
                this.value = newValue;
                this.hash = newValue.hashCode(); // Recalculate hash
            }

            @Override
            public int hashCode() {
                return hash;
            }

            @Override
            public boolean equals(Object obj) {
                if (this == obj) return true;
                if (obj == null || getClass() != obj.getClass()) return false;
                MutableKey other = (MutableKey) obj;
                return Objects.equals(value, other.value); // Compare by value
            }
        }

        MutableKey mutableKey1 = new MutableKey("initial");
        MultiKey mk = new MultiKey(mutableKey1, "key2");
        int initialHashCode = mk.hashCode();

        // Change the underlying value and thus the hashCode of mutableKey1
        mutableKey1.changeValue("changed");

        // The hashCode of the MultiKey should NOT change because it was cached based on the initial hash.
        // This is important for Map behavior.
        assertEquals(initialHashCode, mk.hashCode());
        
        // However, equals() *will* now fail if compared to a MultiKey created with the changed key.
        MultiKey mkChanged = new MultiKey(mutableKey1, "key2");
        assertNotEquals(mk.hashCode(), mkChanged.hashCode()); // This assertion is on the *new* mkChanged hashcode, not mk's.
        assertFalse(mk.equals(mkChanged)); // This is the critical part - equals should detect the change.
    }

    public void testReadResolveRecalculatesHashCode() throws IOException, ClassNotFoundException {
        Object key1 = "key1";

        MultiKey mkOriginal = new MultiKey(key1, key2);
        
        // Manually calculate the expected hash code from the keys array
        int expectedHashCode = 0;
        for (Object key : mkOriginal.getKeys()) {
            if (key != null) {
                expectedHashCode ^= key.hashCode();
            }
        }
        assertEquals(expectedHashCode, mkOriginal.hashCode());

        // Simulate deserialization by calling readResolve
        // In a real scenario, this would be called by ObjectInputStream
        Object resolvedMk = mkOriginal.readResolve();

        // Ensure it's the same instance (as per readResolve contract)
        // and that the hashCode is still the same as it was not changed by readResolve in this case
        assertTrue(resolvedMk instanceof MultiKey);
        MultiKey mkResolved = (MultiKey) resolvedMk;
        assertEquals(mkOriginal.hashCode(), mkResolved.hashCode()); // Should be same as no change occurred
        
        // To truly test readResolve's purpose of recalculating hashCode if it *could* change,
        // we'd need a scenario where hashCode is unstable or changed.
        // The current implementation of calculateHashCode is stable for given keys.
        // The comment in readResolve mentions "hash codes based on the system hash code are only stable for the same process".
        // This implies a potential issue if an Object's hashCode changes between serialization and deserialization.
        // Our current keys are immutable strings, so their hashCodes are stable.
        // For the purpose of this test, we can assert that readResolve doesn't break anything.
    }
