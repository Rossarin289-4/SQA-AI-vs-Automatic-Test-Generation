    @Test
    public void testWriteEntryHeaderWithStarMode() throws Exception {
        TarArchiveEntry entry = new TarArchiveEntry("large_file.bin");
        entry.setSize(Long.MAX_VALUE); // Very large size
        entry.setUserId(Integer.MAX_VALUE + 1L); // Large UID
        entry.setGroupId(Integer.MAX_VALUE + 1L); // Large GID

        byte[] outbuf = new byte[512];
        entry.writeEntryHeader(outbuf, TarUtils.DEFAULT_ENCODING, true); // starMode = true

        // With starMode=true, large values should be written as binary.
        // Verifying binary representation is complex without full understanding of TarUtils' binary formatting.
        // We can check if the octal representation would have overflowed if starMode was false,
        // and assume that binary is correctly used if the value is large.
        
        // The key is that writeEntryHeaderField in the reference code handles this.
        // We can check if the field for size has values that suggest it's not a simple octal overflow.
