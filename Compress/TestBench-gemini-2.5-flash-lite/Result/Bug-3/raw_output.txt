    @Test

        int recordSize = taos.buffer.getRecordSize();
        byte[] data = new byte[recordSize / 2];
        for (int i = 0; i < data.length; i++) {
            data[i] = (byte) i;
        }
        try (TarArchiveOutputStream taos = new TarArchiveOutputStream(out)) {
            TarArchiveEntry entry = new TarArchiveEntry("partial");
            entry.setSize(data.length);
            taos.putArchiveEntry(entry);
            taos.write(data, 0, data.length);
            taos.closeArchiveEntry();
            assertFalse(taos.haveUnclosedEntry);
            assertEquals(data.length, taos.currBytes);
            // Check that the record was padded with zeros
            byte[] writtenData = Arrays.copyOf(out.toByteArray(), data.length);
            assertArrayEquals(data, writtenData);
            // The padding bytes should be 0
            byte[] padding = Arrays.copyOfRange(out.toByteArray(), data.length, recordSize);
            for (byte b : padding) {
                assertEquals(0, b);
            }
        }

    @Test
    public void testWriteWithPartialRecordAssembly() throws Exception {
        OutputStream out = new ByteArrayOutputStream();
        int recordSize = taos.buffer.getRecordSize();
        byte[] chunk1 = new byte[recordSize - 5];
        byte[] chunk2 = new byte[10];
        for (int i = 0; i < chunk1.length; i++) {
            chunk1[i] = (byte) (i + 1);
        }
        for (int i = 0; i < chunk2.length; i++) {
            chunk2[i] = (byte) (i + 101);
        }

        try (TarArchiveOutputStream taos = new TarArchiveOutputStream(out)) {
            TarArchiveEntry entry = new TarArchiveEntry("assembly");
            entry.setSize(chunk1.length + chunk2.length);
            taos.putArchiveEntry(entry);

            taos.write(chunk1, 0, chunk1.length); // This should fill the assemBuf and trigger a partial record write when the next chunk is written.
            taos.write(chunk2, 0, chunk2.length); // This should cause the first record to be written and then assembly to start for the second record.

            taos.closeArchiveEntry();

            assertEquals(chunk1.length + chunk2.length, taos.currBytes);

            // Verify the first record content
            byte[] firstRecord = Arrays.copyOfRange(out.toByteArray(), 0, recordSize);
            byte[] expectedFirstRecord = new byte[recordSize];
            System.arraycopy(chunk1, 0, expectedFirstRecord, 0, chunk1.length);
            // The remaining bytes in the first record should be from chunk2
            System.arraycopy(chunk2, 0, expectedFirstRecord, chunk1.length, recordSize - chunk1.length);
            assertArrayEquals(expectedFirstRecord, firstRecord);

            // Verify the second record content
            byte[] secondRecord = Arrays.copyOfRange(out.toByteArray(), recordSize, recordSize + chunk2.length);
            byte[] expectedSecondRecord = new byte[chunk2.length];
            System.arraycopy(chunk2, recordSize - chunk1.length, expectedSecondRecord, 0, chunk2.length);
            assertArrayEquals(expectedSecondRecord, secondRecord);
        }
    }
