    @Test
    public void testRead_exceptionFromUnderlyingStream() {
        Checksum checksum = new CRC32();
        InputStream failingStream = new ByteArrayInputStream(new byte[] {1, 2, 3}) {
            @Override
            public int read() throws IOException {
                throw new IOException("Simulated IO Exception");
            }
            @Override
            public int read(byte[] b, int off, int len) throws IOException {
                throw new IOException("Simulated IO Exception");
            }
        };
        ChecksumCalculatingInputStream cci = new ChecksumCalculatingInputStream(checksum, failingStream);
        try {
            cci.read();
            fail("Expected IOException");
        } catch (IOException e) {
            assertEquals("Simulated IO Exception", e.getMessage());
        }
        // Checksum should not be updated if underlying stream throws during read()
        // The implementation updates after a successful read, so if read() throws, update isn't called.
        // However, the current implementation of skip() calls read() which can throw.
        // For this test, we focus on read() itself throwing.
        // If read() returns >= 0, checksum is updated. If it throws, it's not.
        // The behavior here depends on the exact sequence of operations that cause the exception.
        // For a direct read() call that throws, no update should occur.
        assertEquals(0, checksum.getValue()); // Assuming no update happened before exception.
    }
