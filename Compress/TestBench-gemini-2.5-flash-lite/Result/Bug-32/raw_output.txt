        // Example: "26 path=new_name\n"
        String paxPathKey = "path";
        String paxPathValue = "new_name";
        String paxHeaderLine = "26 " + paxPathKey + "=" + paxPathValue + "\n"; // 26 is length of "path=new_name\n"
        byte[] paxHeaderBytes = paxHeaderLine.getBytes(CharsetNames.UTF_8);

        // Actual entry header follows pax header
        byte[] entryHeader = new byte[TarConstants.DEFAULT_RCDSIZE];
        String entryName = "actual_entry"; // This name should be overwritten by pax
        ArchiveUtils.matchAsciiBuffer(entryName.getBytes(), entryHeader, TarConstants.NAMEOFFSET, TarConstants.NAMELEN);
        setTarHeaderField(entryHeader, TarConstants.LF_SIZE, 50);
        byte[] entryContent = new byte[50];

        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        baos.write(header); // Header indicating Pax
        baos.write(paxHeaderBytes);
        baos.write(entryHeader);
        baos.write(entryContent);
        InputStream is = new ByteArrayInputStream(baos.toByteArray());

        TarArchiveInputStream tais = new TarArchiveInputStream(is, TarConstants.DEFAULT_BLKSIZE, TarConstants.DEFAULT_RCDSIZE, CharsetNames.UTF_8);
        
        // The first getNextTarEntry() should process the Pax header and return the actual entry
        TarArchiveEntry entry = tais.getNextTarEntry();
        assertNotNull(entry);
        assertEquals(paxPathValue, entry.getName()); // Name should be from Pax header
        assertEquals(50, entry.getSize()); // Size from actual entry header
    }

    // Helper method to set a field in a Tar header byte array
    private void setTarHeaderField(byte[] header, int offset, long value) {
        // Tar headers use 8-byte fields, null-padded, often right-aligned.
        // For simplicity here, we'll set the value as a string representation.
        String valueStr = Long.toString(value);
        byte[] valueBytes = valueStr.getBytes(CharsetNames.US_ASCII);
        
        // Pad with nulls to the right, up to the field size
        byte[] fieldBytes = new byte[TarConstants.SYMTLENGTH]; // e.g., 100 for size
        System.arraycopy(valueBytes, 0, fieldBytes, 0, valueBytes.length);
        
        // Copy to header, right-aligned and null-padded
        int fieldSize = TarConstants.SYMTLENGTH; // e.g., 100 for LF_SIZE
        int start = offset + fieldSize - 1;
        for (int i = fieldSize - 1; i >= 0; i--) {
            if (i < valueBytes.length) {
                header[offset + i] = valueBytes[valueBytes.length - 1 - i];
            } else {
                header[offset + i] = 0; // Null padding
            }
        }
    }

    // Helper method to create a ByteArrayOutputStream with header and content
    private ByteArrayOutputStream baosWithHeaderAndContent(byte[] header, byte[] content) throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        baos.write(header);
        baos.write(content);
        return baos;
    }

    // Mock InputStream that implements Closeable and tracks close calls
    private static class MockCloseableInputStream extends InputStream {
        private final InputStream delegate;
        private boolean closed = false;

        public MockCloseableInputStream(InputStream delegate) {
            this.delegate = delegate;
        }

        public boolean isClosed() {
            return closed;
        }

        @Override
        public int read() throws IOException {
            ensureOpen();
            return delegate.read();
        }

        @Override
        public int read(byte[] b, int off, int len) throws IOException {
            ensureOpen();
            return delegate.read(b, off, len);
        }

        @Override
        public void close() throws IOException {
            delegate.close();
            closed = true;
        }
        
        private void ensureOpen() throws IOException {
            if (closed) {
                throw new IOException("Stream closed");
            }
        }
    }

    // Mock InputStream that simulates a ByteArrayInputStream but allows setting data
    private static class ByteArrayInputStream extends java.io.ByteArrayInputStream {
        public ByteArrayInputStream(byte[] buf) {
            super(buf);
        }
    }
