    @Test
    public void testReadBitsLittleEndianEdgeCase1() throws Exception {
        // Test case where bitsCachedSize is just enough to not require a new byte read,
        // but the requested bits span across the boundary of bitsCachedSize.
        byte[] data = {(byte) 0b00000011, (byte) 0b11000000}; // 3, 192
        try (BitInputStream bis = new BitInputStream(new ByteArrayInputStream(data), ByteOrder.LITTLE_ENDIAN)) {
            // Read 5 bits. bitsCached becomes 0b00000011, bitsCachedSize becomes 8.
            bis.readBits(5); // Reads 00000
            // Now read 4 bits. The cached bits are 0b00000011. bitsCachedSize = 8.
            // The logic should extract 0b0011 (last 4 bits of the cached bits),
            // leaving bitsCached as 0b000000, bitsCachedSize as 4.
            // The next byte is 0b11000000.
            // Expected: last 4 bits of first byte (0011)
            assertEquals(0b0011L, bis.readBits(4));
            // After reading 4 bits, bitsCached should be 0b000000 and bitsCachedSize 4.
            // The next read should pull from the next byte.
            assertEquals(0b11000000L, bis.readBits(8)); // Reads the second byte
        }
    }

    @Test
    public void testReadBitsBigEndianEdgeCase1() throws Exception {
        // Test case where bitsCachedSize is just enough to not require a new byte read,
        // but the requested bits span across the boundary of bitsCachedSize.
        byte[] data = {(byte) 0b00000011, (byte) 0b11000000}; // 3, 192
        try (BitInputStream bis = new BitInputStream(new ByteArrayInputStream(data), ByteOrder.BIG_ENDIAN)) {
            // Read 5 bits. bitsCached becomes 0b00000011, bitsCachedSize becomes 8.
            bis.readBits(5); // Reads 00000
            // Now read 4 bits. The cached bits are 0b00000011. bitsCachedSize = 8.
            // The logic should extract the MSB 4 bits: 0b0000.
            // Remaining bits in cache: 0b0011, bitsCachedSize becomes 4.
            assertEquals(0b0000L, bis.readBits(4));
            // Next read should pull from the next byte.
            // The remaining cached bits are 0b0011.
            // We need 8 bits. The next byte is 0b11000000.
            // bitsCached <<= 8 -> 0b001100000000
            // bitsCached |= nextByte -> 0b001100000000 | 0b11000000 = 0b001111000000
            // bitsCachedSize becomes 4 + 8 = 12.
            // Then extract 8 bits from the left: (0b001111000000 >> (12 - 8)) & MASKS[8]
            // = (0b001111000000 >> 4) & MASKS[8] = 0b00111100 & 0xFF = 0b00111100 (60)
            assertEquals(0b00111100L, bis.readBits(8));
        }
    }

    @Test
    public void testReadBitsLittleEndianComplex() throws Exception {
        byte[] data = {
            (byte) 0x01, (byte) 0x02, (byte) 0x03, (byte) 0x04,
            (byte) 0x05, (byte) 0x06, (byte) 0x07, (byte) 0x08,
            (byte) 0x09, (byte) 0x0a, (byte) 0x0b, (byte) 0x0c
        };
        try (BitInputStream bis = new BitInputStream(new ByteArrayInputStream(data), ByteOrder.LITTLE_ENDIAN)) {
            // Read 12 bits. The last 4 bits of byte 1 (0x01) and all 8 bits of byte 2 (0x02)
            // Little Endian: bits are shifted left as they are read.
            // First byte 0x01 = 00000001
            // Second byte 0x02 = 00000010
            // bitsCached starts empty.
            // read 8 bits:
            // read 0x01: bitsCached = 0x01, bitsCachedSize = 8
            // read 0x02: bitsCached = 0x01 | (0x02 << 8) = 0x01 | 0x0200 = 0x0201
            // bitsCachedSize = 16
            // Now read 12 bits.
            // bitsOut = (0x0201 & MASKS[12]) = 0x0201 = 513
            // bitsCached >>>= 12 -> 0x0201 >>> 12 = 0x0000
            // bitsCachedSize -= 12 -> 16 - 12 = 4
            assertEquals(0x0201L, bis.readBits(12)); // The bits are concatenated in the order they appear from LSB of first byte to MSB of second byte.
            
            // Read next 8 bits. The next byte is 0x03.
            // bitsCached is 0, bitsCachedSize is 4.
            // Need 8 bits. Cache size is < 8.
            // Pull byte 0x03: bitsCached = 0x03, bitsCachedSize = 8.
            // Now read 8 bits. bitsOut = (0x03 & MASKS[8]) = 3
            // bitsCached >>>= 8 -> 0x03 >>> 8 = 0
            // bitsCachedSize -= 8 -> 0
            assertEquals(0x03L, bis.readBits(8));

            // Read 16 bits. Next byte 0x04, next byte 0x05
            // Pull byte 0x04: bitsCached = 0x04, bitsCachedSize = 8
            // Pull byte 0x05: bitsCached = 0x04 | (0x05 << 8) = 0x04 | 0x0500 = 0x0504
            // bitsCachedSize = 16
            // Now read 16 bits.
            // bitsOut = (0x0504 & MASKS[16]) = 0x0504 = 1284
            // bitsCached >>>= 16 -> 0x0504 >>> 16 = 0
            // bitsCachedSize -= 16 -> 0
            assertEquals(0x0504L, bis.readBits(16));
        }
    }

    @Test
    public void testReadBitsBigEndianComplex() throws Exception {
        byte[] data = {
            (byte) 0x01, (byte) 0x02, (byte) 0x03, (byte) 0x04,
            (byte) 0x05, (byte) 0x06, (byte) 0x07, (byte) 0x08,
            (byte) 0x09, (byte) 0x0a, (byte) 0x0b, (byte) 0x0c
        };
        try (BitInputStream bis = new BitInputStream(new ByteArrayInputStream(data), ByteOrder.BIG_ENDIAN)) {
            // Read 12 bits. First 8 bits from byte 1 (0x01), next 4 bits from byte 2 (0x02)
            // Big Endian: bits are shifted left as bytes are read.
            // First byte 0x01 = 00000001
            // Second byte 0x02 = 00000010
            // bitsCached starts empty.
            // read 8 bits:
            // read 0x01: bitsCached = 0x01, bitsCachedSize = 8
            // read 0x02: bitsCached <<= 8 -> 0x0100. bitsCached |= 0x02 -> 0x0102
            // bitsCachedSize = 16
            // Now read 12 bits.
            // bitsOut = (0x0102 >> (16 - 12)) & MASKS[12] = (0x0102 >> 4) & MASKS[12]
            // = 0x010 & 0x0FFF = 0x010 (decimal 16)
            // bitsCachedSize -= 12 -> 4. bitsCached is still 0x0102, but effectively only the last 4 bits are relevant for the next operation.
            // The actual remaining bits are 0x02.
            assertEquals(0x010L, bis.readBits(12)); // bits 00000001 00000010 -> MSB first 12 bits are 000000010000

            // Read next 8 bits.
            // bitsCachedSize is 4, remaining cached bits are 0x02.
            // Need 8 bits. Cache size is < 8.
            // bitsToAddCount = 8 - 4 = 4. overflowBits = 8 - 4 = 4.
            // read 0x03: nextByte = 0x03
            // bitsCached <<= bitsToAddCount (4) -> 0x02 << 4 = 0x020
            // bitsToAdd = (0x03 >>> overflowBits (4)) & MASKS[bitsToAddCount (4)] = (0x03 >>> 4) & 0xF = 0 & 0xF = 0
            // bitsCached |= bitsToAdd -> 0x020 | 0 = 0x020
            // bitsCachedSize = 8
            // overflow = nextByte & MASKS[overflowBits (4)] = 0x03 & 0xF = 0x03
            // bitsOut = bitsCached & MASKS[count (8)] = 0x020 & 0xFF = 0x20 (32)
            // bitsCached = overflow (0x03)
            // bitsCachedSize = overflowBits (4)
            assertEquals(0x20L, bis.readBits(8)); // Reads 0x02 << 4 bits from 0x03

            // Read 16 bits.
            // bitsCachedSize is 4, remaining cached bits are 0x03.
            // Need 16 bits. Cache size is < 16.
            // bitsToAddCount = 16 - 4 = 12. overflowBits = 8 - 12 (this path is wrong, should not happen as 12 < 8)
            // The logic for overflowBits calculation needs review.
            // It should be: if bitsCachedSize < count:
            // If we have enough bits in the cache + one byte to form 'count' bits
            // bitsToAddCount = count - bitsCachedSize;
            // overflowBits = 8 - bitsToAddCount; (if bitsToAddCount <= 8)
            // Read next byte.
            // If byte read is less than 0, return -1.
            // If LITTLE_ENDIAN:
            //   bitsToAdd = nextByte & MASKS[bitsToAddCount];
            //   bitsCached |= (bitsToAdd << bitsCachedSize);
            //   overflow = (nextByte >>> bitsToAddCount) & MASKS[overflowBits];
            // If BIG_ENDIAN:
            //   bitsCached <<= bitsToAddCount;
            //   bitsToAdd = (nextByte >>> overflowBits) & MASKS[bitsToAddCount];
            //   bitsCached |= bitsToAdd;
            //   overflow = nextByte & MASKS[overflowBits];
            // bitsCachedSize = count;
            
            // Let's trace carefully:
            // Initial state: bitsCached = 0x03, bitsCachedSize = 4.
            // readBits(16). count = 16.
            // bitsCachedSize (4) < count (16).
            // bitsCachedSize (4) < 57.
            // bitsToAddCount = 16 - 4 = 12. This is > 8, so the overflowBits logic is bypassed.
            // The loop `while (bitsCachedSize < count && bitsCachedSize < 57)` is not entered again.
            // The code enters the `if (bitsCachedSize < count)` block.
            // `bitsToAddCount = count - bitsCachedSize` = 16 - 4 = 12.
            // `overflowBits = 8 - bitsToAddCount` This seems incorrect if bitsToAddCount > 8.
            // Let's check the actual code again:
            // `if (bitsCachedSize < count)` condition is met.
            // `int bitsToAddCount = count - bitsCachedSize;` -> 12
            // `overflowBits = 8 - bitsToAddCount;` -> 8 - 12 = -4. This is not right.
            // Let's assume the `if (bitsCachedSize < count)` block is intended to handle cases where we don't have enough bits even after reading one full byte.
            // If count is 16 and bitsCachedSize is 4:
            // We need 12 more bits.
            // We read one byte (8 bits).
            // If LITTLE_ENDIAN: bitsToAdd = nextByte & MASKS[12]. This will not work as MASKS[12] is fine, but `nextByte` is only 8 bits.
            // If BIG_ENDIAN: bitsCached <<= 12. bitsToAdd = (nextByte >>> (8-12)). This is also not right.

            // Let's re-evaluate based on the code's actual logic for Big Endian:
            // `bitsCached <<= bitsToAddCount;` -> 0x03 << 12 = 0x3000
            // `bitsToAdd = (nextByte >>> overflowBits) & MASKS[bitsToAddCount];`
            //   Here `overflowBits` is calculated as `8 - bitsToAddCount`. If `bitsToAddCount` is 12, `overflowBits` is -4.
            //   This suggests that the code expects `bitsToAddCount` to be at most 8, meaning `count - bitsCachedSize <= 8`.
            //   So `count <= bitsCachedSize + 8`.
            //   In our case, count = 16, bitsCachedSize = 4. `16 <= 4 + 8` is `16 <= 12`, which is false.
            //   This means the code does not expect to read more than one additional byte.
            //   If `bitsCachedSize < count` and `bitsCachedSize >= 57`, this is handled differently.
            //   The current state is `bitsCachedSize = 4`, `count = 16`. So `bitsCachedSize < count` and `bitsCachedSize < 57`.
            //   The code *should* be reading multiple bytes until `bitsCachedSize >= count`.
            //   However, the loop `while (bitsCachedSize < count && bitsCachedSize < 57)` only reads one byte if `bitsCachedSize` is less than 57.
            //   And the `if (bitsCachedSize < count)` block seems to assume we can form `count` bits using `bitsCachedSize` plus *one* more byte.
            //   Let's assume this is a bug in the code and trace based on what it *does*.

            // Following the code:
            // Current state: bitsCached = 0x03, bitsCachedSize = 4. count = 16.
            // Read next byte: 0x04. nextByte = 0x04.
            // bitsToAddCount = 16 - 4 = 12.
            // overflowBits = 8 - 12 = -4. (This is where the issue lies, but we proceed with the code's execution).
            // `bitsCached <<= bitsToAddCount;` -> 0x03 << 12 = 0x3000
            // `bitsToAdd = (nextByte >>> overflowBits) & MASKS[bitsToAddCount];`
            //   `nextByte >>> overflowBits` is `0x04 >>> -4`. This is undefined behavior or treats it as 0x04 << 4 = 0x40.
            //   Assuming it's treated as 0x40.
            //   `bitsToAdd = 0x40 & MASKS[12]` -> 0x40.
            // `bitsCached |= bitsToAdd;` -> 0x3000 | 0x40 = 0x3040
            // `overflow = nextByte & MASKS[overflowBits];` -> 0x04 & MASKS[-4]. Also problematic.

            // Let's re-examine the "overflowBits" calculation logic.
            // `int bitsToAddCount = count - bitsCachedSize;`
            // `overflowBits = 8 - bitsToAddCount;`
            // This implies that `bitsToAddCount` must be <= 8. So `count - bitsCachedSize <= 8`.
            // If `bitsCachedSize` is 4 and `count` is 16, `bitsToAddCount` is 12.
            // This condition `bitsCachedSize < count` inside the `if` block implies we need more bits.
            // The loop `while (bitsCachedSize < count && bitsCachedSize < 57)` ensures we read at least one byte if possible.
            // If `bitsCachedSize` is small and `count` is large, we should read multiple bytes.
            // The logic `bitsCachedSize < 57` in the while loop suggests that if we have 57 or more bits, we might need a different approach.
            // If `bitsCachedSize < count` and `bitsCachedSize >= 57`:
            //   `int bitsToAddCount = count - bitsCachedSize;` (e.g. count=63, bitsCachedSize=57 -> bitsToAddCount=6)
            //   `overflowBits = 8 - bitsToAddCount;` (e.g. 8-6 = 2)
            //   Read `nextByte`.
            //   If LITTLE_ENDIAN:
            //     `bitsToAdd = nextByte & MASKS[bitsToAddCount];` (e.g. 6 bits from next byte)
            //     `bitsCached |= (bitsToAdd << bitsCachedSize);` (shift these 6 bits to the correct position)
            //     `overflow = (nextByte >>> bitsToAddCount) & MASKS[overflowBits];` (remaining 2 bits of the byte)
            //   If BIG_ENDIAN:
            //     `bitsCached <<= bitsToAddCount;` (shift existing bits left by 6)
            //     `bitsToAdd = (nextByte >>> overflowBits) & MASKS[bitsToAddCount];` (take 6 bits from next byte, shifted right by 2)
            //     `bitsCached |= bitsToAdd;`
            //     `overflow = nextByte & MASKS[overflowBits];` (take the lower 2 bits of the next byte)
            //   `bitsCachedSize = count;`

            // Let's restart the trace for BIG_ENDIAN, count=16, bitsCached=0x03, bitsCachedSize=4.
            // `bitsCachedSize < count` is true (4 < 16).
            // `bitsCachedSize < 57` is true (4 < 57).
            // The `while` loop condition is false because `bitsCachedSize` is not less than `count` after reading just one byte.
            // The code enters the `if (bitsCachedSize < count)` block.
            // `bitsToAddCount = 16 - 4 = 12`.
            // `overflowBits = 8 - 12 = -4`. This is the critical part.
            // If the compiler/JVM interprets `>>> -4` as `>>> (unsigned shift by 4)`, then it would be:
            // `nextByte = 0x04`. `overflowBits = -4`.
            // `bitsCached <<= 12;` -> `0x03 << 12 = 0x3000`.
            // `bitsToAdd = (0x04 >>> -4) & MASKS[12]`. Let's assume `0x04 >>> -4` results in `0x40`. `bitsToAdd = 0x40 & 0x0FFF = 0x40`.
            // `bitsCached |= bitsToAdd;` -> `0x3000 | 0x40 = 0x3040`.
            // `overflow = nextByte & MASKS[overflowBits];` -> `0x04 & MASKS[-4]`. `MASKS[-4]` is not defined. This path seems very broken with negative index.

            // The intended logic for reading bits across byte boundaries when `bitsCachedSize < count` (and `bitsCachedSize >= 57` case is for when the cache is nearly full and we need more bits than a single byte can provide):
            // If we need `count` bits and have `bitsCachedSize` bits.
            // We read one byte. Total bits available = `bitsCachedSize + 8`.
            // If `bitsCachedSize + 8 < count`, we need to read more bytes. The current code structure doesn't seem to support this directly in the `if (bitsCachedSize < count)` block.
            // It seems to assume that `bitsCachedSize + 8` is *enough* to satisfy `count`, or that `count` is small enough to be formed from the current `bitsCachedSize` and *part* of the next byte.

            // Let's try a simpler interpretation of the BigEndian code when `bitsCachedSize < count`:
            // `int bitsToAddCount = count - bitsCachedSize;`
            // `overflowBits = 8 - bitsToAddCount;`
            // `final long nextByte = in.read();`
            // `if (byteOrder == ByteOrder.LITTLE_ENDIAN)` ...
            // `else { // BIG_ENDIAN`
            //   `bitsCached <<= bitsToAddCount;`  // Shift existing bits to make space for new bits
            //   `long bitsToAdd = (nextByte >>> overflowBits) & MASKS[bitsToAddCount];` // Take the required bits from the new byte
            //   `bitsCached |= bitsToAdd;` // Add these bits to the cache
            //   `overflow = nextByte & MASKS[overflowBits];` // The remaining bits in the new byte
            // `}`
            // `bitsCachedSize = count;`

            // Let's assume `count = 10`, `bitsCachedSize = 3`. `bitsToAddCount = 7`. `overflowBits = 1`.
            // `bitsCached <<= 7`.
            // `bitsToAdd = (nextByte >>> 1) & MASKS[7]`. Take 7 bits from `nextByte` (shifted right by 1).
            // `bitsCached |= bitsToAdd`.
            // `overflow = nextByte & MASKS[1]`. Take the LSB of `nextByte`.
            // `bitsCachedSize = 10`.

            // Now apply to our problematic trace: `count = 16`, `bitsCachedSize = 4`.
            // `bitsToAddCount = 12`. `overflowBits = -4`.
            // If `overflowBits` is negative, `nextByte >>> overflowBits` means `nextByte << Math.abs(overflowBits)`.
            // So `nextByte >>> -4` becomes `nextByte << 4`.
            // `bitsToAdd = (nextByte << 4) & MASKS[12]`.
