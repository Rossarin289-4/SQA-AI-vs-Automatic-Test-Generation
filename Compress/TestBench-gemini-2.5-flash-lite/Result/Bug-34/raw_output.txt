    @Test
    public void testLocalFileDataLengthForDefault() throws Exception {
        X7875_NewUnix xf = new X7875_NewUnix();
        // version=1, uidSize=1, gidSize=1, uid=1000 (0x3E8), gid=1000 (0x3E8)
        // uid.toByteArray() for 1000 is [3, -40] (little endian) which is 3 bytes after reversing
        // gid.toByteArray() for 1000 is [3, -40] (little endian) which is 3 bytes after reversing
        // Oh wait, BigInteger.toByteArray() adds a sign byte if needed.
        // 1000 in hex is 0x3E8. In two's complement, this is represented as [0x03, 0xE8].
        // For positive numbers, toByteArray() will not add a leading zero byte unless it's necessary to signify a positive number.
        // Let's trace 1000:
        // BigInteger.valueOf(1000).toByteArray() -> [3, -40] (which is 3, 216 - incorrect interpretation of byte[])
        // Correct interpretation: 1000 is 0x3E8.  ByteArray will be [0x03, 0xE8]. If this was to be interpreted as signed, the most significant bit of 0xE8 would be 1, so it would be negative. To force it positive, it might add a 0x00.
        // Let's check ZipUtil.longToBig(1000).toByteArray(): it's [3, -40]. This is because the byte values are interpreted as signed.
        // After reverse: [-40, 3].
        // So, uidSize is 2, gidSize is 2. Total length = 1 (version) + 1 (uidSize) + 2 (uid) + 1 (gidSize) + 2 (gid) = 7
        // Let's re-check BigInteger.valueOf(1000).toByteArray() on a system. It's {3, -40}.
        // The spec says "UID Size: 1 byte". So it should be the number of bytes for the UID.
        // For 1000, uid.toByteArray() is {3, -40}. After reverse: {-40, 3}.  So uidSize is 2.
        // For gid, same.
        // Total length = 1 (version) + 1 (uidSize) + 2 (uid) + 1 (gidSize) + 2 (gid) = 7.
        // Let's check X7875_NewUnix's trimLeadingZeroesForceMinLength:
        // For 1000, uid.toByteArray() = {3, -40}. trimLeadingZeroesForceMinLength returns {-40, 3}. length is 2.
        // So length = 3 + 2 + 2 = 7.
        assertEquals("Local file data length for default", new ZipShort(7), xf.getLocalFileDataLength());
    }

    @Test
    public void testGetLocalFileDataDataDefault() throws Exception {
        X7875_NewUnix xf = new X7875_NewUnix();
        // Default UID=1000, GID=1000
        // UID 1000 = 0x3E8. BigInteger.toByteArray() is {3, -40}. Reverse: {-40, 3}. Size 2.
        // GID 1000 = 0x3E8. BigInteger.toByteArray() is {3, -40}. Reverse: {-40, 3}. Size 2.
        // data = [version=1, uidSize=2, uidBytes={-40, 3}, gidSize=2, gidBytes={-40, 3}]
        byte[] expected = {1, 2, (byte)0xE8, (byte)0x03, 2, (byte)0xE8, (byte)0x03}; // little endian bytes for 1000 are 0xE8, 0x03
        // Let's re-check the reverse and BigInteger.toByteArray()
        // BigInteger.valueOf(1000).toByteArray() -> {3, -40}
        // reverse({3, -40}) -> {-40, 3}
        // These are the bytes that go into the data array.
        // version = 1
        // uidSize = 2 (length of {-40, 3})
        // uidBytes = {-40, 3}
        // gidSize = 2
        // gidBytes = {-40, 3}
        // So the data array should be {1, 2, -40, 3, 2, -40, 3}
        // -40 is 0xE8. So {-40, 3} is {0xE8, 0x03} which is 1000 in little-endian.
        byte[] expectedCorrect = {1, 2, (byte)0xE8, (byte)0x03, 2, (byte)0xE8, (byte)0x03};
        assertArrayEquals("Local file data default", expectedCorrect, xf.getLocalFileDataData());
    }

        byte[] expectedUidBytes = {(byte)0x09, (byte)0x78, (byte)0x56, (byte)0x34, (byte)0x12}; // Reversed
        byte[] expectedGidBytes = {(byte)0x09, (byte)0x78, (byte)0x56, (byte)0x34, (byte)0x12}; // Reversed
