package org.apache.commons.compress.archivers.tar;

import org.junit.Test;
import static org.junit.Assert.*;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.Reader;
import java.util.HashMap;
import java.util.Map;
import java.util.Map.Entry;
import org.apache.commons.compress.archivers.ArchiveEntry;
import org.apache.commons.compress.archivers.ArchiveInputStream;
import org.apache.commons.compress.utils.ArchiveUtils;

public class TarArchiveInputStreamTest {
    @Test
    public void testRecordSizeDefault() throws Exception {
        TarArchiveInputStream in = new TarArchiveInputStream(new java.io.ByteArrayInputStream(new byte[0]));
        assertEquals(512, in.getRecordSize());
        in.close();
    }

    @Test
    public void testRecordSizeCustom() throws Exception {
        TarArchiveInputStream in = new TarArchiveInputStream(new java.io.ByteArrayInputStream(new byte[0]), 1024, 256);
        assertEquals(256, in.getRecordSize());
        in.close();
    }

    @Test
    public void testAvailableInitially() throws Exception {
        TarArchiveInputStream in = new TarArchiveInputStream(new java.io.ByteArrayInputStream(new byte[0]));
        assertEquals(0, in.available());
        in.close();
    }

    @Test
    public void testNegativeSkip() throws Exception {
        TarArchiveInputStream in = new TarArchiveInputStream(new java.io.ByteArrayInputStream(new byte[0]));
        assertEquals(0L, in.skip(-1));
        in.close();
    }

    @Test
    public void testZeroSkip() throws Exception {
        TarArchiveInputStream in = new TarArchiveInputStream(new java.io.ByteArrayInputStream(new byte[0]));
        assertEquals(0L, in.skip(0));
        in.close();
    }

    @Test
    public void testResetDoesNotThrow() throws Exception {
        TarArchiveInputStream in = new TarArchiveInputStream(new java.io.ByteArrayInputStream(new byte[0]));
        in.reset();
        assertEquals(0, in.available());
        in.close();
    }

    @Test
    public void testEmptyArchiveHasNoNextEntry() throws Exception {
        TarArchiveInputStream in = new TarArchiveInputStream(new java.io.ByteArrayInputStream(new byte[0]));
        assertNull(in.getNextTarEntry());
        in.close();
    }

    @Test
    public void testEmptyArchiveNextEntryAlias() throws Exception {
        TarArchiveInputStream in = new TarArchiveInputStream(new java.io.ByteArrayInputStream(new byte[0]));
        assertNull(in.getNextEntry());
        in.close();
    }

    @Test
    public void testReadWithoutCurrentEntryReturnsEof() throws Exception {
        TarArchiveInputStream in = new TarArchiveInputStream(new java.io.ByteArrayInputStream(new byte[0]));
        assertEquals(-1, in.read(new byte[1], 0, 1));
        in.close();
    }

    @Test
    public void testCanReadNullEntry() throws Exception {
        TarArchiveInputStream in = new TarArchiveInputStream(new java.io.ByteArrayInputStream(new byte[0]));
        assertFalse(in.canReadEntryData(null));
        in.close();
    }

    @Test
    public void testCanReadNonTarEntry() throws Exception {
        TarArchiveInputStream in = new TarArchiveInputStream(new java.io.ByteArrayInputStream(new byte[0]));
        ArchiveEntry entry = new ArchiveEntry() {
            public String getName() { return "entry"; }
            public long getSize() { return 0; }
            public boolean isDirectory() { return false; }
            public java.util.Date getLastModifiedDate() { return new java.util.Date(0); }
        };
        assertFalse(in.canReadEntryData(entry));
        in.close();
    }

    @Test
    public void testCanReadOrdinaryTarEntry() throws Exception {
        TarArchiveInputStream in = new TarArchiveInputStream(new java.io.ByteArrayInputStream(new byte[0]));
        TarArchiveEntry entry = new TarArchiveEntry("entry");
        assertTrue(in.canReadEntryData(entry));
        in.close();
    }

    @Test
    public void testMatchesTooShortSignature() throws Exception {
        assertFalse(TarArchiveInputStream.matches(new byte[8], 8));
    }

    @Test
    public void testMatchesEmptySignature() throws Exception {
        assertFalse(TarArchiveInputStream.matches(new byte[0], 0));
    }

    @Test
    public void testCloseEmptyStream() throws Exception {
        TarArchiveInputStream in = new TarArchiveInputStream(new java.io.ByteArrayInputStream(new byte[0]));
        in.close();
        assertEquals(0, in.getRecordSize());
    }
}

### 1. SOURCE CODE ANALYSIS
The public methods with logic include skip, entry retrieval, read, canReadEntryData, and matches; these tests cover the simple EOF, empty-input, and type-discrimination branches.
Only simple constructor and empty-stream outcomes are used because constructing a valid tar archive would require additional header-format behavior not established here.

### 2. TEST CASE DESIGN
- `testRecordSizeDefault`: default constructor; expected 512, the standard record size used by TarBuffer.
- `testRecordSizeCustom`: record size 256; expected 256 from the constructor's configured TarBuffer.
- `testAvailableInitially`: no current entry; expected 0 because `entrySize` and `entryOffset` start at zero.
- `testNegativeSkip`: skip -1; expected 0 because the skip loop runs only while the amount is positive.
- `testZeroSkip`: skip 0; expected 0 because the skip loop does not run.
- `testResetDoesNotThrow`: empty stream; expected available count remains 0 because reset is a no-op.
- `testEmptyArchiveHasNoNextEntry`: empty input; expected null when the buffer returns no record.
- `testEmptyArchiveNextEntryAlias`: empty input; expected null because `getNextEntry()` delegates to `getNextTarEntry()`.
- `testReadWithoutCurrentEntryReturnsEof`: read one byte before an entry exists; expected -1 because offset equals size.
- `testCanReadNullEntry`: null input; expected false because it is not a `TarArchiveEntry`.
- `testCanReadNonTarEntry`: non-tar archive entry; expected false because the method rejects other entry types.
- `testCanReadOrdinaryTarEntry`: ordinary entry; expected true because it is not GNU sparse.
- `testMatchesTooShortSignature`: length 8; expected false because the required signature length has not been reached.
- `testMatchesEmptySignature`: length 0; expected false by the same minimum-length condition.
- `testCloseEmptyStream`: close empty stream; expected record size 0 after close, as TarBuffer configuration is unchanged.

### 4. DEFECT DETECTION STRATEGY
The tests pin constructor configuration, early EOF and skip behavior, entry-type handling, and the signature-length guard. They do not establish behavior for populated tar archives.

### 6. LIMITATIONS
The tests do not build tar records, so they leave populated-entry reading, availability, and signature variants untested. Actual defect detection is unknown until the unchanged test suite is executed against the reference and the defective versions.
