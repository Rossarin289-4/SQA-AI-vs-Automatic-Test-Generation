    @Test
    public void testConvertLocalToUTC_DST_NonStrict() throws Exception {
        DateTimeZone newYork = DateTimeZone.forID("America/New_York");
        // DST transition: March 2nd, 2008 at 2:00 AM local time. Clocks jump to 3:00 AM.
        // 2:30 AM local time on this day does not exist.
        // In non-strict mode, this local time is interpreted as the later time after the DST adjustment.
        long localMillisGap = DateTimeUtils.parseDateTime("2008-03-02T02:30:00").getMillis(); 
        // The offset for 02:30 on March 2nd 2008 in America/New_York is -04:00 (DST).
        // If we were to try to convert 02:30, the system would use the offset that corresponds to the later time.
        // The earlier time would be -05:00, the later time would be -04:00.
        // The code chooses the later offset for non-strict.
        // So, 02:30 - (-04:00) = 02:30 + 4 hours = 06:30 UTC
        // However, if it's a gap, it means the local time 02:00-02:59:59.999 never happened.
        // The conversion should result in a UTC time that, when converted back to local time with the *correct* offset, becomes a valid time.
        // Let's test an instant that *should* map to the DST time.
        long localMillisDST = DateTimeUtils.parseDateTime("2008-03-02T03:30:00").getMillis(); // This time exists.
        long utcMillisDST = newYork.convertLocalToUTC(localMillisDST, false);
        assertEquals(DateTimeUtils.parseDateTime("2008-03-02T07:30:00Z").getMillis(), utcMillisDST);

        // Now test the gap with non-strict, it should resolve to the offset for the *later* time.
        // A local time of 02:30 would be ambiguous. The code uses the offset of the later time.
        // For NY, the DST gap happens between 2am and 3am. The offset goes from -5h to -4h.
        // If we query for 2:30, it falls into the gap. Non-strict should use the offset of the later time (-4h).
        // So, 2:30 local minus (-4h offset) = 6:30 UTC.
        // The local time 02:30 in America/New_York on March 2nd, 2008, is interpreted as 6:30 UTC.
        // The `originalInstantUTC` parameter is used for cases where the offset might change.
        // When non-strict and in a gap, the offset used is the one that corresponds to the *later* local time.
        // For the NY DST gap, the later time is during DST, offset -04:00.
        long instantForGap = DateTimeUtils.parseDateTime("2008-03-02T02:30:00").getMillis(); // The local time which is in the gap.
        long utcResult = newYork.convertLocalToUTC(instantForGap, false, instantForGap); // Using instantForGap as originalInstantUTC
        // The expected UTC is the local time minus the *later* offset.
        // 02:30 local is 2:30 AM. Offset is -4 hours.
        // 02:30 - (-4:00) = 06:30 UTC.
        assertEquals(DateTimeUtils.parseDateTime("2008-03-02T06:30:00Z").getMillis(), utcResult);
    }
    
    @Test
    public void testConvertLocalToUTC_DST_NonStrict_usingOriginal() throws Exception {
        DateTimeZone newYork = DateTimeZone.forID("America/New_York");
        // DST transition: March 2nd, 2008 at 2:00 AM local time. Clocks jump to 3:00 AM.
        // The original UTC instant was before the DST change.
        long originalInstantUTC = DateTimeUtils.parseDateTime("2008-03-02T06:00:00Z").getMillis(); // This is 2 AM EST.
        long localMillis = DateTimeUtils.parseDateTime("2008-03-02T02:30:00").getMillis(); // This is the local time in the gap.
        
        // The original offset was -5 hours.
        // The conversion should use the offset of the originalInstantUTC if it can.
        // If the original offset works, it should return that.
        // In this case, originalInstantUTC (06:00Z) corresponds to 01:00 EST.
        // The local time provided is 02:30.
        // This logic path is for when the offset *used* for calculation is different from the original offset.
        // Here, the offset of originalInstantUTC is -18000000.
        // instantUTC = localMillis - offsetOriginal = 02:30 - (-5h) = 07:30 UTC.
        // offsetLocalFromOriginal = getOffset(07:30 UTC) which is -04:00 (3.5 hours after transition).
        // offsetLocalFromOriginal (-14400000) != offsetOriginal (-18000000).
        // So it calls convertLocalToUTC(instantLocal, strict)
        
        long utcResult = newYork.convertLocalToUTC(localMillis, false, originalInstantUTC);
        
        // Since non-strict and it's a gap, it will resolve to the later offset.
        // The local time 02:30 is in the gap. Non-strict resolves it to the DST offset (-04:00).
        // 02:30 local - (-04:00) = 06:30 UTC.
        assertEquals(DateTimeUtils.parseDateTime("2008-03-02T06:30:00Z").getMillis(), utcResult);
    }
