    @Test
    public void testGetMillisKeepLocal_differentZone() throws Exception {
        DateTimeZone london = DateTimeZone.forID("Europe/London");
        DateTimeZone ny = DateTimeZone.forID("America/New_York");
        long utcInstant = 1678886400000L; // A fixed UTC instant for predictability

        // Expected conversion:
        // UTC 2023-03-15T12:00:00Z
        // London: 2023-03-15T12:00:00Z (UTC)
        // NY: 2023-03-15T07:00:00 EST (UTC-05:00)
        
        // Convert UTC to London local time
        long londonLocal = london.convertUTCToLocal(utcInstant); // Should be 12:00:00
        
        // Convert London local time back to UTC in NY zone
        long nyUtc = ny.convertLocalToUTC(londonLocal, false, utcInstant);

        // We expect the NY UTC time to be different from the original UTC time
        // because we are preserving the local time (12:00:00 London time = 07:00:00 NY time).
        // The original UTC instant was 12:00:00 UTC.
        // The local time in London was 12:00:00.
        // This local time in NY is 07:00:00.
        // The UTC for 07:00:00 EST is 12:00:00 UTC.
        // This specific test case should result in the same UTC, as it's a fixed time.
        // Let's pick an instant that highlights the difference.
        
        // Let's try an instant that might have DST changes.
        // March 15, 2023, 12:00:00 UTC
        utcInstant = 1678886400000L; 
        
        // London: 2023-03-15 12:00:00 GMT (UTC+0) -> DST is not active yet
        // NY: 2023-03-15 07:00:00 EST (UTC-5)
        
        // convertUTCToLocal(utcInstant) in London should be 1678886400000L + 0 = 1678886400000L
        londonLocal = london.convertUTCToLocal(utcInstant);
        assertEquals(utcInstant, londonLocal);
        
        // convertLocalToUTC(londonLocal, false, utcInstant) in NY zone.
        // local time is 1678886400000L (which is 12:00:00 London time).
        // In NY, this local time is 07:00:00 EST.
        // The UTC for 07:00:00 EST is 07:00:00 + 5 hours = 12:00:00 UTC.
        // So it should remain the same UTC value.
        nyUtc = ny.convertLocalToUTC(londonLocal, false, utcInstant);
        assertEquals(utcInstant, nyUtc);

        // Let's try a time that is after DST start in NY
        // March 15, 2023, 17:00:00 UTC (which is 12:00 PM EST)
        utcInstant = 1678904400000L; 
        
        // London: 2023-03-15 17:00:00 GMT (UTC+0)
        // NY: 2023-03-15 12:00:00 EDT (UTC-4)
        
        londonLocal = london.convertUTCToLocal(utcInstant);
        assertEquals(utcInstant, londonLocal);

        // convertLocalToUTC(londonLocal, false, utcInstant) in NY zone.
        // local time is 1678904400000L (which is 17:00:00 London time).
        // In NY, this local time is 13:00:00 EDT.
        // The UTC for 13:00:00 EDT is 13:00:00 - 4 hours = 09:00:00 UTC.
        nyUtc = ny.convertLocalToUTC(londonLocal, false, utcInstant);
        // 09:00:00 UTC is 1678897200000L
        assertEquals(1678897200000L, nyUtc); 
        
        // The getMillisKeepLocal method should perform this conversion.
        // Original UTC instant: 1678904400000L (17:00 UTC)
        // We want to know what this instant is in London, but preserve the LOCAL time of NY.
        // Local time in NY for 1678904400000L UTC is 13:00 EDT.
        // The UTC for 13:00 EDT is 09:00 UTC.
        long expectedNYMillis = 1678897200000L;
        assertEquals(expectedNYMillis, london.getMillisKeepLocal(ny, utcInstant));
    }

    // Helper method to create a mock Provider that returns a fixed zone
    private static class MockFixedProvider implements Provider {
        private final DateTimeZone fixedZone;
        private final Set<String> ids;

        MockFixedProvider(DateTimeZone zone) {
            this.fixedZone = zone;
            this.ids = new java.util.HashSet<>();
            this.ids.add(zone.getID());
            this.ids.add("UTC"); // Ensure UTC is always available
        }

        @Override
        public DateTimeZone getZone(String id) {
            if (fixedZone.getID().equals(id)) {
                return fixedZone;
            }
            if ("UTC".equals(id)) {
                return DateTimeZone.UTC;
            }
            return null;
        }

        @Override
        public Set<String> getAvailableIDs() {
            return ids;
        }
    }
