    @Test
    public void testConvertLocalToUTC_strict_withGap() throws Exception {
        // Use a timezone with a DST gap, e.g., Pacific/Auckland, where in September
        // clocks jump forward by one hour (e.g., from 02:00 to 03:00).
        // A local time of 02:30 would fall into the gap.
        DateTimeZone nzZone = DateTimeZone.forID("Pacific/Auckland");
        // Create a local instant that falls into a DST gap.
        // Assuming DST starts on the last Sunday of September at 2 AM.
        // In 2023, it was Sept 24th. Local time 02:00 becomes 03:00.
        // So, 02:30 local time doesn't exist.
        long localInstantInGap = nzZone.convertUTCToLocal(
            ISOChronology.getInstanceUTC().getDateTimeMillis(2023, 9, 24, 14, 30, 0, 0) // UTC time that becomes 2:30 AM in Auckland on Sept 24, 2023
        ); // This should be approximately 2023-09-24T02:30:00.000

        // Test that strict conversion throws an exception.
        try {
            nzZone.convertLocalToUTC(localInstantInGap, true);
            fail("Expected IllegalArgumentException for strict conversion during a gap.");
        } catch (IllegalArgumentException expected) {
            // Correctly caught exception.
        }
    }
    
    @Test
    public void testConvertLocalToUTC_nonStrict_withGap() throws Exception {
        DateTimeZone nzZone = DateTimeZone.forID("Pacific/Auckland");
        // Local time that falls into a DST gap (e.g., 02:30 AM after DST starts)
        long localInstantInGap = nzZone.convertUTCToLocal(
            ISOChronology.getInstanceUTC().getDateTimeMillis(2023, 9, 24, 14, 30, 0, 0)
        ); 

        // Test non-strict conversion. It should adjust the time to the later hour.
        long utcInstant = nzZone.convertLocalToUTC(localInstantInGap, false);
        
        // The non-strict conversion should map the gap time to the hour after the gap.
        // If 2:30 AM local time is in a gap, it's likely mapped to 3:30 AM local time.
        // Let's verify the offset. The original offset at the UTC time used to create localInstantInGap
        // would be the standard offset. After the DST transition, the offset increases.
        // The non-strict method will attempt to use the offset that resolves to the later time.
        long utcInstantExpected = nzZone.getMillisKeepLocal(DateTimeZone.UTC, localInstantInGap);
        
        // The key is that convertLocalToUTC(..., false) will map the 'gap' time to the equivalent time
        // that occurs *after* the DST transition.
        // So if the local time 02:30 is skipped, convertLocalToUTC(02:30, false) should yield the UTC
        // that corresponds to 03:30 local time.
        // Let's find the UTC time for 03:30 on that day.
        long expectedUTC_afterGap = nzZone.convertUTCToLocal(
            ISOChronology.getInstanceUTC().getDateTimeMillis(2023, 9, 24, 15, 30, 0, 0) // UTC time that becomes 3:30 AM in Auckland on Sept 24, 2023
        );
        assertEquals(expectedUTC_afterGap, utcInstant);
    }

        DateTimeZone zone1 = DateTimeZone.forOffsetHours(1); // UTC+1
        DateTimeZone zone2 = DateTimeZone.forOffsetHours(2); // UTC+2
        long utcInstant = 1000000L;
        long localInstantZone1 = zone1.convertUTCToLocal(utcInstant);
        long convertedInstantZone2 = zone1.getMillisKeepLocal(zone2, utcInstant);
        
        // The local time should be the same in zone2 as it was in zone1.
        // localInstantZone1 = utcInstant + offset1
        // convertedInstantZone2 = localInstantZone1 - offset2
        // convertedInstantZone2 = (utcInstant + offset1) - offset2
        long expected = utcInstant + zone1.getOffset(utcInstant) - zone2.getOffset(utcInstant);
        assertEquals(expected, convertedInstantZone2);
    }

    @Test
    public void testNextTransition_basic() throws Exception {
        DateTimeZone londonZone = DateTimeZone.forID("Europe/London");
        long instant = ISOChronology.getInstanceUTC().getDateTimeMillis(2023, 1, 1, 12, 0, 0, 0); // January 1st
