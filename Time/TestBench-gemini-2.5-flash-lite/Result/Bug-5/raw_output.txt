    @Test
    public void testConstructorWithDurationMillis() throws Exception {
        // 1 year = 365 days * 24 hours/day = 8760 hours
        // 1 month = 30 days * 24 hours/day = 720 hours (approx, for simplicity in test)
        // Using a duration that is roughly one year and a bit
        long durationMillis = (365L * DateTimeConstants.MILLIS_PER_DAY) + (10L * DateTimeConstants.MILLIS_PER_HOUR);
        Period period = new Period(durationMillis);
        // The constructor with only duration splits into precise fields (time fields)
        // 365 days * 24 hours/day = 8760 hours
        // 10 hours
        // Total = 8770 hours
        assertEquals(0, period.getYears());
        assertEquals(0, period.getMonths());
        assertEquals(0, period.getWeeks());
        assertEquals(0, period.getDays());
        assertEquals(8770, period.getHours()); // 8760 + 10
        assertEquals(0, period.getMinutes());
        assertEquals(0, period.getSeconds());
        assertEquals(0, period.getMillis());
    }

    @Test
    public void testToStandardWeeks() throws Exception {
        Period period = new Period(0, 0, 2, 3, 4, 5, 6, 7); // 2 weeks, 3 days, 4 hours, 5 minutes, 6 seconds, 7 millis
        // 2 weeks * 7 days/week = 14 days
        // Total days = 14 + 3 = 17 days
        // 17 days * 24 hours/day = 408 hours
        // 4 hours
        // Total hours = 408 + 4 = 412 hours
        // 412 hours * 60 minutes/hour = 24720 minutes
        // 5 minutes
        // Total minutes = 24720 + 5 = 24725 minutes
        // 24725 minutes * 60 seconds/minute = 1483500 seconds
        // 6 seconds
        // Total seconds = 1483500 + 6 = 1483506 seconds
        // 1483506 seconds * 1000 ms/second = 1483506000 ms
        // 7 millis
        // Total millis = 1483506007 ms
        // Total weeks = 1483506007 ms / (7 * 24 * 60 * 60 * 1000) ms/week = 1483506007 / 604800000 ~= 2.45
        // The method returns Weeks.weeks(fieldValue), and here it is the weeks field + milliseconds converted to weeks.
        // So it's 2 weeks + (3 days, 4 hours, 5 mins, 6 secs, 7 millis) converted to weeks.
        // 3 days = 3*24*60*60*1000 ms = 259,200,000 ms
        // 4 hours = 4*60*60*1000 ms = 14,400,000 ms
        // 5 minutes = 5*60*1000 ms = 300,000 ms
        // 6 seconds = 6*1000 ms = 6,000 ms
        // 7 millis = 7 ms
        // Total millis = 259200000 + 14400000 + 300000 + 6000 + 7 = 273,906,007 ms
        // Total weeks from this duration = 273906007 / 604800000 ~= 0.45
        // So total weeks = 2 + 0.45 = 2.45.
        // The toStandardWeeks() method is designed to sum up all fields into a total number of weeks.
        // 2 weeks + 3 days + 4 hours + 5 mins + 6 secs + 7 millis
        // = 2 weeks + (3 * 24 + 4) hours + 5 mins + 6 secs + 7 millis
        // = 2 weeks + 76 hours + 5 mins + 6 secs + 7 millis
        // = 2 weeks + (76*60 + 5) mins + 6 secs + 7 millis
        // = 2 weeks + 4565 mins + 6 secs + 7 millis
        // = 2 weeks + (4565*60 + 6) secs + 7 millis
        // = 2 weeks + 273906 secs + 7 millis
        // = 2 weeks + (273906 * 1000 + 7) ms
        // = 2 weeks + 273906007 ms
        // Convert millis to weeks: 273906007 / (7 * 24 * 60 * 60 * 1000) = 273906007 / 604800000 = 0.4529...
        // Total weeks = 2 + 0.4529... = 2.4529...
        // safeToInt will truncate.
        assertEquals(2, period.toStandardWeeks().getWeeks());
    }

    @Test
    public void testNormalizedStandard() throws Exception {
        Period period = new Period(1, 13, 0, 40, 25, 61, 61, 1001); // 1 year, 13 months, 40 days, 25 hours, 61 minutes, 61 seconds, 1001 millis
        // Normalized should sum up fields based on standard definitions:
        // 1 year + 13 months = 2 years 1 month
        // 40 days = 5 weeks 5 days
        // 25 hours = 1 day 1 hour
        // 61 minutes = 1 hour 1 minute
        // 61 seconds = 1 minute 1 second

        // Combining:
        // Years: 2
        // Months: 1

        // Hours: 1

        // 6 days = 6 days
        // 1 hour
        // 2 minutes
        // 2 seconds
        // 1 millis
        // Total: 2 years, 1 month, 6 days, 1 hour, 2 minutes, 2 seconds, 1 millis
