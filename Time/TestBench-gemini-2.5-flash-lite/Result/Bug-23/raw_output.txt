    @Test
    public void testGetOffsetFromLocal_dst_transition_overlap() throws Exception {
        // This test is tricky as it depends on specific DST rules.
        // Let's assume a zone where DST ends and an hour is repeated.
        // We want to test the scenario where the local time is ambiguous.
        // Example: A zone that goes from +02:00 to +01:00 at 2 AM.
        // The hour 1 AM to 2 AM happens twice.
        // If we pass a local instant within this overlap, getOffsetFromLocal should return the offset for the *later* (standard) time.
        
        // We need a zone with DST. Europe/London is a good candidate.
        // DST ends on the last Sunday of October. In 2023, it was Oct 29th.
        // At 2:00 AM, clocks went back to 1:00 AM.
        // The hour 01:00:00 to 01:59:59 happened twice.
        // We need to pick an instant that falls into this repeated hour.
        // Let's test a local time of 2023-10-29 01:30:00.
        
        DateTimeZone london = DateTimeZone.forID("Europe/London");
        
        // Construct the local instant carefully.
        // Use a fixed chronology to avoid unexpected DST shifts during construction.
        org.joda.time.chrono.ISOChronology iso = org.joda.time.chrono.ISOChronology.getInstance(london);
        
        // Instant for 2023-10-29 01:30:00 in London, *before* the DST change (assuming it's the first occurrence)
        // The actual calculation of local time to millis is complex due to DST.
        // A simpler approach is to test the output of convertLocalToUTC.
        
        // Let's test the convertLocalToUTC, which uses getOffsetFromLocal internally.
        // If a local time is ambiguous, convertLocalToUTC with strict=true and originalInstantUTC might behave differently.
        
        // Test a time that *should* be ambiguous.
        // Example: In Europe/London, 2023-10-29 01:30:00 GMT occurs twice.
        // The first occurrence is 2023-10-29 01:30:00 BST (UTC+1). UTC instant: 2023-10-29 00:30:00 UTC.
        // The second occurrence is 2023-10-29 01:30:00 GMT (UTC+0). UTC instant: 2023-10-29 01:30:00 UTC.
        
        // getOffsetFromLocal should return the offset that corresponds to the *standard* time (UTC+0).
        long localMillis_ambiguous = new org.joda.time.DateTime(2023, 10, 29, 1, 30, 0, london).getMillis();
        
        // The offset for the first occurrence (BST) is +1 hour.
        // The offset for the second occurrence (GMT) is +0 hours.
        // getOffsetFromLocal should return 0 for this ambiguous local time.
        int offset = london.getOffsetFromLocal(localMillis_ambiguous);
        assertEquals(0, offset); // Expecting standard offset during overlap
    }

    @Test
    public void testGetOffsetFromLocal_dst_transition_gap() throws Exception {
        // Test a time that should fall into a DST gap.
        // Example: A zone that goes from +01:00 to +02:00 at 2 AM.
        // The hour 2 AM to 3 AM is skipped.
        // getOffsetFromLocal should return the offset that occurs *after* the gap.
        
        // Europe/London: DST starts on the last Sunday of March. In 2023, it was March 26th.
        // At 1:00 AM GMT (UTC+0), clocks sprang forward to 2:00 AM BST (UTC+1).
        // The hour 01:00:00 to 01:59:59 GMT is skipped.
        
        DateTimeZone london = DateTimeZone.forID("Europe/London");
        
        // Construct a local instant within the gap, e.g., 2023-03-26 01:30:00 BST.
        // This local time does not exist in London for that date.
        long localMillis_gap = new org.joda.time.DateTime(2023, 3, 26, 1, 30, 0, london).getMillis();
        
        // The offset for the hour immediately after the gap (2:30 AM BST) is +1 hour.
        // getOffsetFromLocal should return +1 hour.
        int offset = london.getOffsetFromLocal(localMillis_gap);
        assertEquals(DateTimeConstants.MILLIS_PER_HOUR, offset);
    }

        DateTimeZone zone = DateTimeZone.forID("Europe/London");

        long localMillis = new org.joda.time.DateTime(2023, 10, 29, 1, 30, 0, zone).getMillis();
        
        long utcMillis = zone.convertLocalToUTC(localMillis, false);
        
        // The expected UTC instant is 2023-10-29 01:30:00 UTC.
        long expectedUtcMillis = new org.joda.time.DateTime(2023, 10, 29, 1, 30, 0, DateTimeZone.UTC).getMillis();
        assertEquals(expectedUtcMillis, utcMillis);
