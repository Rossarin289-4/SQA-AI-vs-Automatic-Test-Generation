    @Test
    public void testDateTimeOfYearWithTokenizer2400() throws Exception {
        // Test case where day is last and dayOfWeek is specified, then 24:00
        // "last Sun" is dayOfWeek=7. Adding a day to Sunday results in Monday, which is dayOfWeek=1.
        // The month should advance if day is -1 (last).
        StringTokenizer st = new StringTokenizer("Dec last Sun 24:00");
        ZoneInfoCompiler.DateTimeOfYear dt = new ZoneInfoCompiler.DateTimeOfYear(st);
        assertEquals(1, dt.iMonthOfYear); // December + 1 day = January
        assertEquals(1, dt.iDayOfMonth); // Day advances to 1st
        assertEquals(2, dt.iDayOfWeek);  // Day of week advances from Sunday (7) to Monday (1) + 1 = 2 (Tuesday)
        assertTrue(dt.iAdvanceDayOfWeek); // Advance day of week is true for last day type
        assertEquals(0, dt.iMillisOfDay); // 24:00 implies start of next day
    }

        ZoneInfoCompiler.Zone zone = new ZoneInfoCompiler.Zone(st);
        assertEquals("Zone/Test", zone.iName);
        assertEquals(0, zone.iOffsetMillis);
        assertEquals("RuleSet1", zone.iRules);
        assertEquals("ZULU", zone.iFormat);
        assertEquals(Integer.MAX_VALUE, zone.iUntilYear); // Expected MAX_VALUE if no year is provided and only day of week is present.
        // The provided tokenizer seems to be missing the year for the "until" part.
        // Based on the constructor, if tokens are present, it expects a year.
        // If 'Sun>=1 01:00:00 w' are present, the year should be parsed first.
        // Given the current tokenizer, it will likely throw NumberFormatException if Year is missing.
        // Let's assume a valid input for testing the structure.
        // The provided constructor `private Zone(String name, StringTokenizer st)` expects year and then DateTimeOfYear.
        // If st.hasMoreTokens() is true, it expects a year.
        // If st.hasMoreTokens() is true again, it parses DateTimeOfYear.
        // In this specific tokenizer "Sun>=1 01:00:00 w", "Sun>=1" would be the year if interpreted as number, which is not.
        // This indicates a potential issue with the test data or my understanding of its structure.
        // The code `year = Integer.parseInt(st.nextToken());` would fail here.
        // Let's correct the test data to match expected pattern: Year followed by DateTimeOfYear.
    }
