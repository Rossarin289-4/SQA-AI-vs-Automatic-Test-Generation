        char[] next = buffer.finishCurrentSegment();
        next[0] = 'b';
        buffer.setCurrentLength(1);

    public void testTextBufferEnsureNotSharedAndRelease() throws Exception {

        buffer.ensureNotShared();

        assertEquals("", buffer.contentsAsString());
