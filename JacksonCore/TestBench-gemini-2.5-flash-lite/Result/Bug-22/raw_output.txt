    // Test case: Filter that explicitly denies a property
    @Test
    public void testNextToken_filterDeniesProperty() throws Exception {
        MockJsonParser delegate = new MockJsonParser();
        
        // Mock structure: {"key1": "value1", "key2": "value2"}
        // Filter will deny "key2"
        TokenFilter denyingFilter = new TokenFilter() {
            @Override
            public TokenFilter includeProperty(String name) {
                if ("key2".equals(name)) {
                    return TokenFilter. நோக்க; // Deny
                }
                return TokenFilter.INCLUDE_ALL;
            }
        };
        
        FilteringParserDelegate parser = createFilteringParser(delegate, denyingFilter, false, false);

        // Simulate reading "key1"
        delegate.setCurrentToken(JsonToken.FIELD_NAME);
        delegate.setCurrentName("key1");
        assertEquals(JsonToken.FIELD_NAME, parser.nextToken());
        assertEquals("key1", parser.getCurrentName());

        // Simulate reading "value1" (should be included)
        delegate.setCurrentToken(JsonToken.VALUE_STRING);
        delegate.setTextValue("value1");
        assertEquals(JsonToken.VALUE_STRING, parser.nextToken());
        assertEquals("value1", parser.getText());

        // Simulate reading "key2" (should be filtered out)
        delegate.setCurrentToken(JsonToken.FIELD_NAME);
        delegate.setCurrentName("key2");
        
        // The nextToken call for "key2" should result in delegate skipping children
        // and nextToken returning null or the next token after the skipped children.
        // This mock doesn't fully simulate skipChildren, but we expect the value to be filtered.
        // For this test, we'll assume the nextToken call after a denied field name
        // might return the value or skip it. Let's test that it's not returned directly.
        JsonToken token2 = parser.nextToken();
        // Depending on the internal logic, it might skip the value and return null,
        // or return a different token. The key is that "key2" and its value are effectively ignored.
        // For simplicity, let's assume the next token after a denied field name might be END_OBJECT
        // if it's the last field, or the next field.
        // A more robust test would require simulating the delegate's behavior after skipChildren.
        // For now, we'll check that it doesn't return "key2" as the current token.
        assertNotEquals(JsonToken.FIELD_NAME, token2); 
        // If the mock returns null after a denied field, that's also a valid test.
        // Let's check if it returns null to signify skipping.
        // The actual behavior is more complex, involving skipping children.
        // We can test the outcome of the filter logic.
        // If key2 is denied, delegate.nextToken() will be called, and then skipChildren will be called.
        // Let's set up the delegate to return the next token after key2/value2.
        delegate.setCurrentToken(JsonToken.END_OBJECT); // Assume this is next after "key2":"value2"
        JsonToken tokenAfterDenied = parser.nextToken();
        // The actual behavior of nextToken after a denied field name is to call delegate.nextToken()
        // and then delegate.skipChildren(). The returned token is the result of these calls.
        // If "key2":"value2" is denied, the parser should skip "value2" and then process the next token.
        // The logic in `_nextToken2` and `_nextTokenWithBuffering` handles this.
        // If `includeProperty` returns ` நோக்க`, the code calls `delegate.nextToken()` and `delegate.skipChildren()`.
        // So the next token returned by `parser.nextToken()` will be whatever `delegate.nextToken()` returns *after* skipping.
        // For this mock, let's assume after skipping "value2", the next token from delegate is END_OBJECT.
        assertEquals(JsonToken.END_OBJECT, tokenAfterDenied);
    }

    // Test case: Multiple matches allowed, first match returns INCLUDE_ALL
    @Test
    public void testNextToken_allowMultipleMatches_includeAll() throws Exception {
        MockJsonParser delegate = new MockJsonParser();
        delegate.setCurrentToken(JsonToken.VALUE_STRING);
        delegate.setTextValue("first match");
        
        // A filter that returns INCLUDE_ALL for the first token and then continues
        TokenFilter filter = new TokenFilter() {
            private int callCount = 0;
            @Override
            public TokenFilter includeProperty(String name) {
                if (++callCount == 1) return TokenFilter.INCLUDE_ALL;
                return TokenFilter. நோக்க;
            }
            @Override
            public boolean includeValue(JsonParser p) throws IOException {
                if (callCount == 1) return true;
                return false;
            }
        };
        
        FilteringParserDelegate parser = createFilteringParser(delegate, filter, false, true); // allowMultipleMatches = true
        
        JsonToken token1 = parser.nextToken();
        assertEquals(JsonToken.VALUE_STRING, token1);
        assertEquals("first match", parser.getText());
        assertEquals(1, parser._matchCount);

        // Now, assume the next token is a field name, and this time the filter should deny.
        delegate.setCurrentToken(JsonToken.FIELD_NAME);
        delegate.setCurrentName("secondField");
        JsonToken token2 = parser.nextToken(); // This should call the filter again
        
        // Since allowMultipleMatches is true, the filter is still checked.
        // The filter logic for the second call should return TokenFilter. நோக்க,
        // causing the second field/value to be skipped.
        // The next token returned should be whatever comes after "secondField" and its value.
        // Let's assume delegate returns END_OBJECT next.
        delegate.setCurrentToken(JsonToken.END_OBJECT);
        JsonToken token3 = parser.nextToken();
        assertEquals(JsonToken.END_OBJECT, token3);
        assertEquals(1, parser._matchCount); // Match count should remain 1
    }

        // Filter that returns INCLUDE_ALL on first call
        TokenFilter filter = TokenFilter.INCLUDE_ALL;
        
        FilteringParserDelegate parser = createFilteringParser(delegate, filter, false, false); // allowMultipleMatches = false
        
        JsonToken token1 = parser.nextToken();
        assertEquals(JsonToken.VALUE_STRING, token1);
        assertEquals("first match", parser.getText());
        assertEquals(1, parser._matchCount); // First match increments match count

        // Now, expect nextToken to return null because allowMultipleMatches is false and we've had a match.
        // The internal logic has a check `if (!_allowMultipleMatches && (_currToken != null) && (_exposedContext == null))`
        // and if `_currToken.isScalarValue() && !_headContext.isStartHandled() && !_includePath && (_itemFilter == TokenFilter.INCLUDE_ALL)`
        // it returns null. This test case needs careful setup to hit that path.
        
        // Let's simulate a scalar value after the first match.
        delegate.setCurrentToken(JsonToken.VALUE_NUMBER_INT); // This is a scalar
        delegate.setIntValue(123);
        
        // The `_currToken` is already set to VALUE_STRING from the previous call.
        // The logic checks `_currToken != null` which is true.
        // `_exposedContext` is null.
        // `_currToken.isScalarValue()` is true.
        // `_headContext.isStartHandled()` is false (assuming root level).
        // `!_includePath` is true.
        // `_itemFilter == TokenFilter.INCLUDE_ALL` is true.
        // Thus, it should return null.

        // To properly test this, the previous token must be cleared, and the new token set.
        parser.clearCurrentToken(); // Clear the previous VALUE_STRING
        
        // Now call nextToken again with the new scalar value
        JsonToken token2 = parser.nextToken(); 
        assertNull(token2); // Expect null due to _allowMultipleMatches = false and prior match.
        assertEquals(1, parser._matchCount); // Match count should not change
    }

The tests rely on a heavily mocked `JsonParser` (`MockJsonParser`) to simulate token streams. This mock is simplified and may not perfectly replicate all edge cases of a real `JsonParser`'s behavior, particularly around `skipChildren()` and complex token sequences.
