        return clazz.getMethod(methodName, parameterTypes);

    // Helper method to get a TypeVariable
    private TypeVariable<?> getTypeVariable(Class<?> clazz, String variableName) {
        TypeVariable<?>[] typeVariables = clazz.getTypeParameters();
        for (TypeVariable<?> tv : typeVariables) {
            if (tv.getName().equals(variableName)) {
                return tv;
            }
        }
        return null;
    }

    // Helper method to create a ParameterizedType (simplified for testing)
    private ParameterizedType createParameterizedType(Class<?> rawType, Type... actualTypeArguments) {
        return new ParameterizedType() {
            @Override
            public Type[] getActualTypeArguments() {
                return actualTypeArguments;
            }

            @Override
            public Type getRawType() {
                return rawType;
            }

            @Override
            public Type getOwnerType() {
                return null;
            }

            @Override
            public String toString() {
                return rawType.getSimpleName() + "<" + Arrays.deepToString(actualTypeArguments) + ">";
            }

            @Override
            public boolean equals(Object o) {
                if (this == o) return true;
                if (o == null || getClass() != o.getClass()) return false;
                ParameterizedType that = (ParameterizedType) o;
                return Objects.equals(getActualTypeArguments(), Arrays.asList(that.getActualTypeArguments())) &&
                       Objects.equals(getRawType(), that.getRawType()) &&
                       Objects.equals(getOwnerType(), that.getOwnerType());
            }

            @Override
            public int hashCode() {
                return Objects.hash(getActualTypeArguments(), getRawType(), getOwnerType());
            }
        };
    }

    // Helper method to create a WildcardType
    private WildcardType createWildcardType(Type[] upperBounds, Type[] lowerBounds) {
        return new WildcardType() {
            @Override
            public Type[] getUpperBounds() {
                return upperBounds;
            }

            @Override
            public Type[] getLowerBounds() {
                return lowerBounds;
            }

            @Override
            public String toString() {
                return "?" + (upperBounds.length > 0 ? " extends " + Arrays.deepToString(upperBounds) : "") +
                       (lowerBounds.length > 0 ? " super " + Arrays.deepToString(lowerBounds) : "");
            }

            @Override
            public boolean equals(Object o) {
                if (this == o) return true;
                if (o == null || getClass() != o.getClass()) return false;
                WildcardType that = (WildcardType) o;
                return Objects.equals(getUpperBounds(), Arrays.asList(that.getUpperBounds())) &&
                       Objects.equals(getLowerBounds(), Arrays.asList(that.getLowerBounds()));
            }

            @Override
            public int hashCode() {
                return Objects.hash(getUpperBounds(), getLowerBounds());
            }
        };
    }

    @Test
    public void testResolveGenericReturnTypeForGenericMethodParam() throws Exception {
        Method method = getMethod(TestInterface.class, "getGenericMethodParam", Object.class);
        // This is tricky as getGenericMethodParam has its own type parameter V.
        // The genericReturnType is a TypeVariable 'V'.
        Type genericReturnType = method.getGenericReturnType();
        GenericMetadataSupport metadata = new GenericMetadataSupport.TypeVariableReturnType(new FromClassGenericMetadataSupport(TestInterface.class), method.getTypeParameters(), (TypeVariable<?>) genericReturnType);

        // The raw type of this TypeVariableReturnType should be the type variable itself.
        assertEquals(method.getTypeParameters()[0], metadata.rawType());

        // The actual type arguments map should resolve the type variable 'V' based on the method's declaration.
        // In this case, V extends T.
        TypeVariable<?> vVar = method.getTypeParameters()[0]; // V
        TypeVariable<?> tVar = getTypeVariable(TestInterface.class, "T");

        // The contextualActualTypeParameters should contain the resolution for V.
        // Since V extends T, and T is a type parameter of TestInterface, the resolution should be T.
        // However, TypeVariableReturnType's contextualActualTypeParameters are populated from its source and method type parameters.
        // Let's inspect how it populates.
        // It registers method's type parameters: V.
        // It registers bounds of V: T. So, contextualActualTypeParameters will map V to T.
        assertEquals(tVar, metadata.contextualActualTypeParameters.get(vVar));
        assertEquals(tVar, metadata.getActualTypeArgumentFor(vVar));
    }
