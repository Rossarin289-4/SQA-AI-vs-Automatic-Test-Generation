    @Test
    public void testComplexGenericClass_twoTypeParams() throws Exception {
        GenericMetadataSupport metadata = GenericMetadataSupport.inferFrom(ComplexGenericClass.class);
        Method method = ComplexGenericClass.class.getMethod("methodTwoTypeParams");
        GenericMetadataSupport returnMetadata = metadata.resolveGenericReturnType(method);
        // The method is <S, T extends S> T. The return type is T.
        // The generic declaration for this method has two type parameters S and T.
        // T is bounded by S.
        // This is complex to assert directly. We check if it returns a TypeVariable.
        assertTrue(returnMetadata.contextualActualTypeParameters.isEmpty()); // Context from ComplexGenericClass is not directly used for method-level type params.
        assertTrue(returnMetadata.actualTypeArguments().isEmpty()); // No type args for the method's own type params here.
        // The raw type of the return should be extractable from the type variable T.
        // This is hard to test without more context on how `extractRawTypeOf` works.
        // Let's test a simpler aspect: the type variable itself.
        // The return type is T. T is a TypeVariable.
        // The `resolveGenericReturnType` should return a `TypeVariableReturnType`.
        // We can't easily assert the type of the returned object without breaking encapsulation.
        // Instead, let's assert that `rawType()` throws an exception for this scenario, or returns Object.
        // The current implementation of `TypeVariableReturnType.rawType()` may throw.
        // Let's test the direct type bounds instead.
        TypeVariable<?>[] methodTypeParams = method.getTypeParameters();
        assertEquals(2, methodTypeParams.length); // S and T
        TypeVariable<?> typeT = null;
        for(TypeVariable<?> tv : methodTypeParams) {
            if (tv.getName().equals("T")) {
                typeT = tv;
                break;
            }
        }
        assertNotNull(typeT);

        // The metadata for the return type T should reflect its bounds.
        // This is usually handled by `TypeVariableReturnType`.
        // We can't directly instantiate `TypeVariableReturnType` here.
        // We can check the result of `resolveGenericReturnType`.
        // The `rawType()` method on `TypeVariableReturnType` should resolve to something.
        // For `<S, T extends S> T`, `T`'s bound is `S`. `S` has no explicit bounds.
        // So `T`'s effective bound is `Object`.
        // Thus `rawType()` should return `Object.class`.
        assertEquals(Object.class, returnMetadata.rawType());
    }

    public void testExtractRawTypeOf_typeVariableReferringToContext() {
        class DummyTVReturnType extends GenericMetadataSupport.TypeVariableReturnType {
            public DummyTVReturnType(GenericMetadataSupport source, TypeVariable[] typeParameters, TypeVariable typeVariable) {
                super(source, typeParameters, typeVariable);

                return super.extractRawTypeOf(type);
            }

        // E.g., `class Outer<X> { class Inner<Y extends X> { Y method(); } }`
        // Infer metadata for `Outer<String>.Inner<String>`
        class Outer<X> {
            class Inner<Y extends X> {
                Y method() { return null; }
            }
        }
        TypeVariable<?> outerX = Outer.class.getTypeParameters()[0];
        TypeVariable<?> innerY = Outer.Inner.class.getTypeParameters()[0];

        // Create ParameterizedType for Outer<String>
        Type parameterizedOuterString = new org.mockito.internal.util.reflection.ParameterizedTypeImpl(Outer.class, new Type[]{String.class});

        // Create ParameterizedType for Inner<String> which is a member of Outer<String>
        // This requires careful construction of the generic type.
        // `Outer.Inner.class.getGenericSuperclass()` might not be what we need directly for `inferFrom`.
        // Instead, we infer from `Outer.Inner`. But `Outer.Inner` is an inner class, and `inferFrom` expects a top-level class or a raw type.

        // Let's simplify: Infer from `Outer<String>.Inner`.
        // This requires creating a `ParameterizedType` for `Outer<String>` and then using its `getActualTypeArguments()` to get `String`,
        // then using that to resolve the type of `Inner`.

        // `inferFrom` takes a `Type`.
        // If we infer from `Outer.Inner.class`, it doesn't have the context of `Outer<String>`.
        // We need to provide a `ParameterizedType` that represents `Outer<String>.Inner<String>`.

        // Let's simulate the resolution process:
        // Assume we have context where X = String.
        // Inner<Y extends X> means Inner<Y extends String>.
        // If we infer `Outer<String>.Inner<String>`, then Y = String.

        // Let's infer from `Outer.Inner.class` and then manually set the context.
        GenericMetadataSupport dummySource = GenericMetadataSupport.inferFrom(Object.class);
        // Manually create context for Outer<String>.Inner
        Map<TypeVariable, Type> context = new HashMap<>();
        // X is String
        context.put(outerX, String.class);
        // Y is String
        context.put(innerY, String.class);

        // Now create a dummy TypeVariableReturnType that uses this context.
        // The method returns Y. So the typeVariable is `innerY`.
        TypeVariable<?> yTypeVar = innerY; // The type variable being returned.

        // To pass this to DummyTVReturnType, we need a `source` that has this context.
        // `FromClassGenericMetadataSupport` can populate context.
        GenericMetadataSupport sourceWithContext = new GenericMetadataSupport.FromClassGenericMetadataSupport(Outer.Inner.class);
        // Manually add context for Outer<String>
        sourceWithContext.contextualActualTypeParameters.put(outerX, String.class);
        sourceWithContext.contextualActualTypeParameters.put(innerY, String.class); // Y in Inner<Y extends X> maps to String

        // Now test extracting raw type of `innerY` using this context.
        DummyTVReturnType tvReturnType = new DummyTVReturnType(sourceWithContext, Outer.Inner.class.getTypeParameters(), innerY);
        assertEquals(String.class, tvReturnType.callExtractRawTypeOf(innerY));
    }
